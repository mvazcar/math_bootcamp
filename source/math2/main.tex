For nonempty sets $A$ and $B$, the \textbf{Cartesian product} is
\[
  A\times B:=\{(a,b):a\in A,\ b\in B\}.
\]
\end{definitionbox}

$B:=\{y:0\leq y\leq1\text{ or }2\leq y\leq3\}$, the product is not a rectangle.

\begin{frame}{Functions}
\begin{definitionbox}[1.1.6 Definition]
Let $A$ and $B$ be sets. Then a \textbf{function} from $A$ to $B$ is a set $f$ of ordered
pairs in $A\times B$ such that for each $a\in A$ there exists a unique $b\in B$ with
$(a,b)\in f$.

\vspace{.4em}
$D(f):=A$ is the \textbf{domain}; $R(f)$, the set of second elements, is the
\textbf{range}, with $R(f)\subseteq B$.
\end{definitionbox}

\end{frame}

\begin{frame}{Injective, Surjective, and Bijective}
\begin{definitionbox}[1.1.9 Definition]
Let $f:A\to B$ be a function.
\statementitem{a}{$f$ is \textbf{injective} (\textbf{one-to-one}) if
$x_1\neq x_2$ implies $f(x_1)\neq f(x_2)$.}
\statementitem{b}{$f$ is \textbf{surjective} (maps $A$ \textbf{onto} $B$) if
$f(A)=B$; that is, if $R(f)=B$.}
\statementitem{c}{$f$ is \textbf{bijective} if it is both.}
\end{definitionbox}

\vspace{.4em}
Such an $f$ is called an \emph{injection}, a \emph{surjection}, or a
\emph{bijection}.
\end{frame}

\begin{frame}{Proofs - Injective and Surjective}
\begin{itemize}
  \item In order to prove that a function $f$ is injective, we must establish that:
  \[
    \text{for all }x_1,x_2\text{ in }A,\quad
    \text{if }f(x_1)=f(x_2),\text{ then }x_1=x_2.
  \]
  To do this we assume that $f(x_1)=f(x_2)$ and show that $x_1=x_2$.

  \item To prove that a function $f$ is surjective, we must show that for any $b\in B$
  there exists at least one $x\in A$ such that $f(x)=b$.
\end{itemize}
\end{frame}

\begin{frame}{Composition and Inverse Images}
\begin{definitionbox}[1.1.12 Definition]
If $f:A\to B$ and $g:B\to C$, the \textbf{composite function}
$g\circ f:A\to C$ is defined by
\[
(g\circ f)(x):=g\bigl(f(x)\bigr)\qquad\text{for all }x\in A.
\]
\end{definitionbox}

\begin{definitionbox}[1.1.7 Definition]
For $H\subseteq B$, the \textbf{inverse image} of $H$ under $f$ is
\[
f^{-1}(H):=\{x\in A:f(x)\in H\}.
\]
\end{definitionbox}

\end{frame}

\begin{frame}{Cardinality - Finite and Infinite Sets}
\begin{definitionbox}[1.3.1 Definition]
\statementitem{a}{$\varnothing$ has $0$ \textbf{elements}.}
\statementitem{b}{$S$ has $n$ \textbf{elements} if there is a bijection
$\mathbb{N}_n:=\{1,2,\ldots,n\}\to S$.}
\statementitem{c}{$S$ is \textbf{finite} if it is empty or has $n$ elements for
some $n\in\mathbb{N}$; \textbf{infinite} otherwise.}
\end{definitionbox}

\begin{definitionbox}[1.3.6 Definition]
\statementitem{a}{$S$ is \textbf{denumerable} (\textbf{countably infinite}) if
there is a bijection $\mathbb{N}\to S$.}
\statementitem{b}{$S$ is \textbf{countable} if it is finite or denumerable;
\textbf{uncountable} otherwise.}
\end{definitionbox}
\end{frame}


\begin{frame}{Cardinalities of Infinite Sets}
\emph{Many of our intuitions, and even formal notions, break at
infinity.}

\begin{definitionbox}[Comparing Sizes]
Two sets have the same cardinality when there is a bijection between them.
This definition applies to both finite and infinite sets.
\end{definitionbox}

\begin{itemize}
    \item For finite sets, cardinality counts the elements;
    for infinite sets, it grades the size of the infinity.
    \item $\mathbb{N}$, $\mathbb{Z}$, and $\mathbb{Q}$ are countably infinite:
    each can be put in bijection with $\mathbb{N}$.
    \item $\mathbb{R}$ is uncountable: there is no bijection
    $\mathbb{N}\to\mathbb{R}$.
\end{itemize}

\centering
Cantor's diagonal argument proves the second claim:
\href{https://www.youtube.com/watch?v=YIZd23zGV3M}{video}.
\end{frame}

\begin{frame}{$\mathbb{R}$ Is a Complete Ordered Field}
\begin{definitionbox}[Ordered Field]
A \textbf{field} supports addition and multiplication. A \textbf{total order}
compares every pair and is compatible with those operations.
\[
\begin{gathered}
a<b \;\Longrightarrow\; a+c<b+c,\\
0<a\text{ and }0<b \;\Longrightarrow\; 0<ab.
\end{gathered}
\]
Totality gives \textbf{trichotomy}: exactly one of
$a<b$, $a=b$, $a>b$ holds.
\end{definitionbox}

\vspace{.5em}
Both $\mathbb{Q}$ and $\mathbb{R}$ are ordered fields. Only $\mathbb{R}$ is
\textbf{complete}: it has no gaps.
\end{frame}


\begin{frame}{Bounds}
\begin{definitionbox}[2.3.1 Bounds]
Let $\varnothing\ne S\subseteq\mathbb{R}$.
\[
\begin{aligned}
u\text{ is an upper bound}
&\quad\Longleftrightarrow\quad s\leq u &&\text{for every }s\in S,\\
w\text{ is a lower bound}
&\quad\Longleftrightarrow\quad w\leq s &&\text{for every }s\in S.
\end{aligned}
\]
$S$ is \textbf{bounded} if it has both an upper and a lower bound.
\end{definitionbox}

\vspace{.5em}
\textbf{Example.}\quad For $S=(0,1)$: upper bounds $=[1,\infty)$, lower bounds $=(-\infty,0]$.

\begin{columns}[t,onlytextwidth]
\begin{column}{\tightcolumnwidth}
\begin{definitionbox}[2.3.2 Supremum = least upper bound]
\[
u=\sup S
\]
\begin{enumerate}
    \item $s\leq u$ for every $s\in S$.
    \item $u\leq v$ for every upper bound $v$.
\end{enumerate}
\centering
$u\in S\Rightarrow u=\max S$;\\[-.1em]
$u\notin S\Rightarrow\text{no maximum}$.\par
\end{definitionbox}
\end{column}
\hspace{\twocolumngutter}
\begin{column}{\tightcolumnwidth}
\begin{definitionbox}[Infimum = greatest lower bound]
\[
w=\inf S
\]
\begin{enumerate}
    \item $w\leq s$ for every $s\in S$.
    \item $t\leq w$ for every lower bound $t$.
\end{enumerate}
\centering
$w\in S\Rightarrow w=\min S$;\\[-.1em]
$w\notin S\Rightarrow\text{no minimum}$.\par
\end{definitionbox}
\end{column}

\begin{frame}{Completeness of the Real Numbers}
\begin{theorembox}[2.3.6 Completeness Property of $\mathbb{R}$]
For every nonempty $S\subseteq\mathbb{R}$,
\[
S\text{ bounded above}
\quad\Longrightarrow\quad
\sup S\in\mathbb{R}.
\]
Consequently, if $S$ is bounded below, then $\inf S\in\mathbb{R}$.
\end{theorembox}

\begin{block}{Two Different Words}
A \emph{binary relation} is complete when it ranks every pair: for all $x,y$,
$x\succeq y$ or $y\succeq x$. Nothing is left uncompared.

The \emph{real numbers} are complete in an unrelated sense: no bounded set is left
without a supremum. Nothing is left unfilled.
\end{block}

\centering Same word, unrelated properties. Days 1 and 3 use the first; Day 2 uses
the second.
\end{frame}

\textbf{2.4.3}\quad
\emph{If $x\in\mathbb{R}$, then there exists $n_x\in\mathbb{N}$ such that $x\leq n_x$.}

\textbf{2.4.8 Density Theorem}\quad
\emph{If $x$ and $y$ are any real numbers with $x<y$, then there exists a rational number

\vspace{.25em}
\textbf{2.4.9 Corollary}\quad
\emph{If $x$ and $y$ are real numbers with $x<y$, then there exists an irrational number

\begin{frame}{What Makes Something a Distance}
\begin{definitionbox}[Properties of $\lVert x-y\rVert$]
The \textbf{Euclidean distance} between $x$ and $y$ is $\lVert x-y\rVert$, where
\[
\lVert z\rVert:=\sqrt{\textstyle\sum_{\ell=1}^{L}z_\ell^2}.
\]
For every $x,y,z\in\mathbb{R}^L$ it satisfies:
\begin{enumerate}
    \item \textbf{Nonnegativity.} $\lVert x-y\rVert\geq0$, and $=0$ only when $x=y$.
    \item \textbf{Symmetry.} $\lVert x-y\rVert=\lVert y-x\rVert$.
    \item \textbf{Triangle inequality.} $\lVert x-z\rVert\leq\lVert x-y\rVert+\lVert y-z\rVert$.
\end{enumerate}
\end{definitionbox}

Any function on pairs with these three properties is a \textbf{distance}.
Everything that follows --- balls, convergence, continuity --- uses only
these three.

\vspace{.35em}
\centering The third is the one that does the work: no detour through $y$ is shorter
than going straight.
\end{frame}

\begin{frame}{The Open Ball}
\begin{definitionbox}[Open Ball]
For $x\in\mathbb{R}^L$ and $\varepsilon>0$,
\[
B_\varepsilon(x):=\{y\in\mathbb{R}^L:\lVert y-x\rVert<\varepsilon\}.
\]
\end{definitionbox}

\begin{frame}{Open, Closed, Bounded}
\begin{definitionbox}[Sets in $\mathbb{R}^L$]
Let $D\subseteq\mathbb{R}^L$.
\begin{itemize}
    \item $D$ is \textbf{open} if every $x\in D$ has some
    $\varepsilon>0$ with $B_{\varepsilon}(x)\subseteq D$.
    \item $D$ is \textbf{closed} if its complement
    $\mathbb{R}^L\setminus D$ is open.
    \item $D$ is \textbf{bounded} if $D\subseteq B_{M}(0)$
    for some $M>0$.
\end{itemize}
\end{definitionbox}

All three rest on the same measurement: the ball $B_{\varepsilon}(x)$, and so
the distance $\lVert y-x\rVert$.
\end{frame}

\begin{frame}{Sequences}
\begin{definitionbox}[Sequence]
A \textbf{sequence} in $\mathbb{R}^L$ is a function $x:\mathbb{N}\to\mathbb{R}^L$.
We write $x_n$ for $x(n)$, and $(x_n)$ or $(x_1,x_2,x_3,\ldots)$ for the sequence itself.
\end{definitionbox}

A list of points, one for each natural number, in order. Bartle 3.1.1 is the case
$L=1$. The one question we ask of a sequence: where is it heading?
\end{frame}

\begin{frame}{Convergence and Divergence}
\begin{definitionbox}[Convergence]
$(x_n)$ \textbf{converges} to $x\in\mathbb{R}^L$ if
\[
\forall\varepsilon>0\;\exists N\in\mathbb{N}\;\forall n\geq N,
\qquad \lVert x_n-x\rVert<\varepsilon .
\]
We write $x_n\to x$. A sequence that converges to no point is \textbf{divergent}.
\end{definitionbox}

In words: every ball around $x$, however small, eventually holds every term.
Convergence in $\mathbb{R}^L$ is convergence in each coordinate.

\begin{theorembox}[Limits Keep Weak Inequalities]
If $a_n\to a$ in $\mathbb{R}$ and $a_n\geq c$ for every $n$, then $a\geq c$.
\end{theorembox}
\end{frame}

\begin{frame}{Closed Sets Contain Their Limits}
\begin{theorembox}[Sequential Test for Closed Sets]
Let $F\subseteq\mathbb{R}^L$. Then $F$ is closed if and only if
\[
x_n\in F\ \text{for every }n,\qquad x_n\to x
\quad\Longrightarrow\quad x\in F.
\]
\end{theorembox}

A closed set cannot be left by taking limits: whatever a sequence inside it
approaches is inside it too. This is how closedness is checked in practice,
and how Day 3 will check it for contour sets.
\end{frame}

\begin{frame}{Continuity of a Function}
\begin{definitionbox}[Sequential Definition]
Let $X\subseteq\mathbb{R}^L$, $Y\subseteq\mathbb{R}^K$, and $f:X\to Y$.
The function $f$ is \emph{continuous at $x\in X$} if, for every sequence
$(x_n)$ in $X$,
\[
x_n\to x \quad\Longrightarrow\quad f(x_n)\to f(x).
\]
It is continuous on $X$ if it is continuous at every $x\in X$.
\end{definitionbox}

\vspace{.5em}
\textbf{Keep the objects separate.}\quad This is continuity of a
\emph{function}. Day 3 defines continuity of a \emph{preference relation}.
\end{frame}

\begin{frame}{Compactness in $\mathbb{R}^L$}
\begin{theorembox}[Heine--Borel and Extreme Values]
Let $D\subseteq\mathbb{R}^L$. Then
\[
D\text{ is compact}
\quad\Longleftrightarrow\quad
D\text{ is closed and bounded}.
\]
If $D$ is nonempty and compact and $f:D\to\mathbb{R}$ is continuous, then
there exist $x_{\min},x_{\max}\in D$ such that
\[
f(x_{\min})\leq f(x)\leq f(x_{\max})
\qquad\text{for every }x\in D.
\]
\end{theorembox}

\end{frame}

\textbf{Is rationality enough?}\quad On finite or countable $X$, yes. On $\mathbb{R}_+^L$
it is not: continuity is needed too. That is Day 3.
\end{frame}

\begin{frame}{Utility Is Ordinal}
\begin{propositionbox}[Invariance to Increasing Transformations]
If $u$ represents $\succeq$ and $\phi:\mathbb{R}\to\mathbb{R}$ is strictly increasing,
then $\phi\circ u$ represents $\succeq$ as well.
\end{propositionbox}

The numbers carry no information beyond the order they induce. Any relabelling
that preserves that order represents the same preferences, so differences and
ratios of utilities mean nothing.
\end{frame}

\begin{frame}{Representation and Choice}
\begin{definitionbox}[Representation Preserves Maximizers]
Suppose $u:X\to\mathbb{R}$ represents $\succeq$. For every nonempty menu
$B\subseteq X$ on which a best alternative exists,
\[
C_{\succeq}(B)
=\{x\in B:x\succeq y\text{ for every }y\in B\}
=\operatorname*{arg\,max}_{x\in B}u(x).
\]
If $\phi:\mathbb{R}\to\mathbb{R}$ is strictly increasing, then
\[
\operatorname*{arg\,max}_{x\in B}u(x)
=\operatorname*{arg\,max}_{x\in B}(\phi\circ u)(x).
\]
\end{definitionbox}

\end{frame}
