\begin{frame}{Notation}
\begin{definitionbox}[Fixed Throughout]
\[
D\subseteq\mathbb{R}^n,
\qquad
f:D\to\mathbb{R},
\qquad
\max\{f(x)\mid x\in D\}.
\]
One dimension: $f'(x)$, $f''(x)$.

\vspace{.6em}
Several dimensions, the \textbf{gradient} and the \textbf{Hessian}:
\[
\nabla f(x):=\begin{pmatrix} f_1(x)\\ \vdots\\ f_n(x) \end{pmatrix},
\qquad
H_f(x):=\begin{pmatrix}
f_{11}(x) & \cdots & f_{1n}(x)\\
\vdots & \ddots & \vdots\\
f_{n1}(x) & \cdots & f_{nn}(x)
\end{pmatrix}.
\]
\vspace{-.4em}
$f_i=\partial f/\partial x_i$ and $f_{ij}=\partial^2 f/\partial x_i\,\partial x_j$.
\end{definitionbox}
\end{frame}

\begin{frame}{The Optimization Problem}
\begin{definitionbox}[Sundaram 2.1]
Given $f:\mathbb{R}^n\to\mathbb{R}$ and $D\subseteq\mathbb{R}^n$,
\[
\max\{f(x)\mid x\in D\}
\qquad\text{or}\qquad
\min\{f(x)\mid x\in D\}.
\]
$f$ is the \textbf{objective}, $D$ the \textbf{feasible set}.
\end{definitionbox}

\vspace{.5em}
\centering Two questions: does a solution exist, and how do we find it?
\end{frame}

\begin{frame}{Maximum Value or Maximizing Choice}
\begin{definitionbox}[Two Different Objects]
\[
\max_{x\in D} f(x)\in\mathbb{R},
\qquad
\operatorname*{arg\,max}_{x\in D} f(x)=\{x\in D: f(x)\geq f(y)\ \forall y\in D\}.
\]
The first is a number; the second is a set, possibly empty.
\end{definitionbox}
\end{frame}

\begin{frame}{Increasing Transformations}
\begin{propositionbox}
If $\phi:\mathbb{R}\to\mathbb{R}$ is strictly increasing, then
\[
\operatorname*{arg\,max}_{x\in D} f(x)
=\operatorname*{arg\,max}_{x\in D} (\phi\circ f)(x).
\]
\end{propositionbox}

\vspace{.5em}
\centering Utility is ordinal (Day 2): the objective may be rescaled freely.
\end{frame}

\begin{frame}{Existence: The Weierstrass Theorem}
\begin{theorembox}[Sundaram 3.1]
Let $D\subseteq\mathbb{R}^n$ be \textbf{compact}, and let
$f:D\to\mathbb{R}$ be \textbf{continuous} on $D$. Then $f$
attains a maximum and a minimum on $D$.
\[
D\ \text{compact},\ f\ \text{continuous}
\quad\Longrightarrow\quad
\operatorname*{arg\,max}_{x\in D}f(x)\neq\varnothing.
\]
\end{theorembox}

\vspace{.5em}
\centering Compact means closed and bounded (Day 2, Heine--Borel).
\end{frame}

\begin{frame}{The First-Order Condition}
\begin{theorembox}[Sundaram 4.1]
Suppose $x^*\in\operatorname{int}D\subseteq\mathbb{R}^n$ is a local
maximum of $f$ on $D$, and $f$ is differentiable at $x^*$. Then
\[
\nabla f(x^*)=0.
\]
The same holds if $x^*$ is a local minimum.
\end{theorembox}

\vspace{.5em}
\centering Interior is an assumption: at a boundary point the condition fails.
\end{frame}

\begin{frame}{Stationary Points and the Boundary}
\begin{definitionbox}[Osborne 5.1.1]
Let $f(x)=x^2$ on $D=[-1,2]$. The only stationary point, $x=0$, is the global
minimum; the global maximum is the boundary point $x=2$, where $f'(2)\neq0$.
\end{definitionbox}

\begin{frame}{Critical Points}
\begin{definitionbox}[Critical Point]
$x^*$ is a \textbf{critical point} of $f$ on $D$ if $\nabla f(x^*)=0$.
\end{definitionbox}

\centering
\includegraphics[width=0.88\linewidth,height=142pt,keepaspectratio]{critical-points.pdf}

The first-order condition is necessary, never sufficient (Sundaram 4.2).
\end{frame}

\begin{frame}{Second-Order Conditions}
\begin{theorembox}[Sundaram 4.3]
Let $f$ be $C^2$ and let $x^*\in\operatorname{int}D$ be a critical point.
\[
x^*\ \text{local maximum}\ \Longrightarrow\ H_f(x^*)\preceq0,
\]
\[
\nabla f(x^*)=0,\ H_f(x^*)\prec0\ \Longrightarrow\ x^*\ \text{strict local maximum}.
\]
\end{theorembox}

\vspace{.5em}
\centering Semidefinite is necessary; definite is sufficient.
\end{frame}

\begin{frame}{The Determinant Test in $\mathbb{R}^2$}
\begin{propositionbox}
For $H=H_f(x^*)$ with entries $f_{11},f_{12},f_{22}$,
\[
\begin{aligned}
f_{11}<0,\ \det H>0 &\;\Longrightarrow\; \text{local maximum},\\
f_{11}>0,\ \det H>0 &\;\Longrightarrow\; \text{local minimum},\\
\det H<0 &\;\Longrightarrow\; \text{saddle}.
\end{aligned}
\]
\end{propositionbox}
\end{frame}

\begin{frame}{Definiteness and Concavity}
\begin{theorembox}[Sundaram 7.10]
Let $D\subseteq\mathbb{R}^n$ be open and convex and let $f$ be $C^2$. Then
\[
f\ \text{concave on}\ D
\quad\Longleftrightarrow\quad
H_f(x)\preceq0\ \ \text{for every}\ x\in D,
\]
and $H_f(x)\prec0$ for every $x\in D$ implies $f$ strictly concave.
\end{theorembox}

\vspace{.5em}
\centering At one point definiteness is a local test; at every point it is
concavity.
\end{frame}

\begin{frame}{Equality Constraints}
\begin{definitionbox}[Sundaram 5.1]
For $f:\mathbb{R}^n\to\mathbb{R}$ and $g:\mathbb{R}^n\to\mathbb{R}^k$,
\[
\max f(x)
\quad\text{subject to}\quad
g_i(x)=0,\quad i=1,\ldots,k.
\]
\end{definitionbox}

\vspace{.5em}
\centering The constraint removes $k$ degrees of freedom.
\end{frame}

\begin{frame}{The Theorem of Lagrange}
\begin{theorembox}[Sundaram 5.1]
Let $f$ and $g_i$ be $C^1$, and let $x^*$ be a local maximum or minimum of $f$ on
\[
D=U\cap\{x\mid g_i(x)=0,\ i=1,\ldots,k\},
\qquad U\ \text{open}.
\]
If $\nabla g_1(x^*),\ldots,\nabla g_k(x^*)$ are linearly independent, there is
$\lambda^*\in\mathbb{R}^k$ with
\[
\nabla f(x^*)+\sum_{i=1}^{k}\lambda_i^*\,\nabla g_i(x^*)=0.
\]
\end{theorembox}
\end{frame}

\begin{frame}{Lagrange in Two Variables}
\begin{definitionbox}[Osborne 6.1]
Maximize $f(x,y)=xy$ subject to $x+y=6$. With
$L(x,y)=xy-\lambda(x+y-6)$ the conditions
\[
y-\lambda=0,
\qquad
x-\lambda=0,
\qquad
x+y=6
\]
give $(x^*,y^*)=(3,3)$ and $\lambda^*=3$.
\end{definitionbox}
\end{frame}

\begin{frame}{Regularity}
\begin{definitionbox}[Constraint Qualification]
The rank condition $\nabla g_1(x^*),\ldots,\nabla g_k(x^*)\ \text{are linearly independent}$ says the constraint gradients
are linearly independent at $x^*$.
\end{definitionbox}

\begin{frame}{The Cobb--Douglas Consumer}
\begin{definitionbox}[Osborne 6.1]
Maximize $x^{a}y^{b}$ subject to $px+y=m$, with $a,b,p,m>0$. From
\[
a x^{a-1}y^{b}=\lambda p,
\qquad
b x^{a}y^{b-1}=\lambda,
\qquad
px+y=m
\]
we get
\[
x^*=\frac{am}{(a+b)p},
\qquad
y^*=\frac{bm}{a+b}.
\]
\end{definitionbox}

\vspace{.4em}
\centering Fixed income shares, independent of $p$ and $m$.
\end{frame}

\begin{frame}{Inequality Constraints}
\begin{definitionbox}[Sundaram 6.1]
For $f:\mathbb{R}^n\to\mathbb{R}$ and $h:\mathbb{R}^n\to\mathbb{R}^l$,
\[
\max f(x)
\quad\text{subject to}\quad
h_i(x)\geq0,\quad i=1,\ldots,l.
\]
A constraint is \textbf{effective} at $x^*$ if $h_i(x^*)=0$.
\end{definitionbox}

\vspace{.5em}
\centering Which constraints bind is not given in advance; it is part of the
solution.
\end{frame}

\begin{frame}{The Karush--Kuhn--Tucker Theorem}
\begin{theorembox}[Sundaram 6.1]
Let $f$ and $h_i$ be $C^1$, and let $x^*$ be a local maximum of $f$ on
\[
D=U\cap\{x\mid h_i(x)\geq0,\ i=1,\ldots,l\},
\qquad U\ \text{open}.
\]
If the effective gradients $\{\nabla h_i(x^*)\}_{i\in E}$ are linearly
independent, there is $\lambda^*\in\mathbb{R}^l$ with
\[
\lambda_i^*\geq0,
\qquad
\lambda_i^*h_i(x^*)=0,
\qquad
\nabla f(x^*)+\sum_{i=1}^{l}\lambda_i^*\,\nabla h_i(x^*)=0.
\]
\end{theorembox}
\end{frame}

\begin{frame}{Complementary Slackness}
\begin{definitionbox}[Two Regimes]
For each constraint, $\lambda_i^*h_i(x^*)=0$ forces one of
\[
\begin{aligned}
h_i(x^*)=0 &: \ \text{the constraint binds, and }\lambda_i^*\geq0;\\
h_i(x^*)>0 &: \ \text{the constraint is slack, and }\lambda_i^*=0.
\end{aligned}
\]
\end{definitionbox}

\vspace{.5em}
\centering A slack constraint has no shadow value.
\end{frame}

\begin{frame}{From Local to Global}
\begin{theorembox}[Sundaram 7]
Let $D$ be convex and $f$ concave on $D$. Then every local
maximum of $f$ is a global maximum, and the set of maximizers is convex.
\end{theorembox}

\vspace{.5em}
\centering For concave $f$ the first-order condition is also sufficient.
\end{frame}

\begin{frame}{Uniqueness}
\begin{theorembox}[Sundaram 8]
Let $D$ be convex and $f$ \emph{strictly} quasiconcave. Then $f$ has
at most one maximizer on $D$.
\end{theorembox}

\vspace{.5em}
\centering Day 3 promised this: strictly convex preferences give a single
demand.
\end{frame}

\begin{frame}{A Binding Budget}
\begin{propositionbox}
If $\succeq$ is locally nonsatiated and $p\gg0$, every solution of
\[
\max\ u(x)\quad\text{subject to}\quad p\cdot x\leq I
\]
satisfies $p\cdot x=I$.
\end{propositionbox}

\vspace{.5em}
\centering Day 3 promised this too: the inequality becomes an equality.
\end{frame}

\begin{frame}{The Consumer Problem}
\begin{definitionbox}[Utility Maximization]
Given prices $p\gg0$ and income $I>0$,
\[
\max_{x\in\mathbb{R}_+^L} u(x)
\quad\text{subject to}\quad
p\cdot x\leq I,
\]
over the budget set $B(p,I)=\{x\in\mathbb{R}_+^L:p\cdot x\leq I\}$, which is
compact and convex.
\end{definitionbox}

\vspace{.5em}
\centering Compact and continuous give existence; strictly convex preferences
give uniqueness.
\end{frame}
