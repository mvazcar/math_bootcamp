\begin{frame}{Optimization Has Two Separate Questions}
  Consider the scalar problem
  \[
    \max\{f(x):x\in D\},
    \qquad D\subseteq\mathbb{R}.
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Existence}
        \begin{minipage}[t][2.10cm][t]{\dimexpr\linewidth-.76em\relax}
          Does some feasible point actually attain the largest value?

          \vspace{.45em}
          This is a question about the feasible set and the objective together.
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Identification}
        \begin{minipage}[t][2.10cm][t]{\dimexpr\linewidth-.76em\relax}
          If a maximizer exists, what conditions must it satisfy?

          \vspace{.45em}
          Derivatives generate candidates; they do not create a solution.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.55em}
  Day 1 wrote choice as an \emph{argmax}. Today we study what makes that expression meaningful.
  \sources{Sundaram, Sections 2.1, 3.1, and 4.1.}
\end{frame}

\begin{frame}{The Maximum and the Maximizers Are Different Objects}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Maximum Value]
        \begin{minipage}[t][2.15cm][t]{\dimexpr\linewidth-.76em\relax}
          If the largest value is attained, then
          \[
            \max\{f(x):x\in D\}=f(x^*).
          \]
          The maximum is a number.
        \end{minipage}
      \end{definitionbox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Maximizers]
        \begin{minipage}[t][2.15cm][t]{\dimexpr\linewidth-.76em\relax}
          \[
            \operatorname*{arg\,max}_{x\in D}f(x)
          \]
          is the set of feasible $x$ such that $f(x)\geq f(y)$ for every $y\in D$.
        \end{minipage}
      \end{definitionbox}
    \end{column}
  \end{columns}

  \vspace{.45em}
  The maximum records how high the objective goes; the argmax records where it gets there.
  \sources{Sundaram, Section 2.1.}
\end{frame}

\begin{frame}{The Argmax Can Be Empty, a Singleton, or Many Points}
  \renewcommand{\arraystretch}{1.38}
  \centering
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{Problem} & \textbf{Argmax} & \textbf{What happens}\\
    \midrule
    $\max_{x\in(0,1)}x$ & $\varnothing$ & the supremum is not attained\\
    $\max_{x\in[-1,1]}(-x^2)$ & $\{0\}$ & one maximizer\\
    $\max_{x\in[0,1]}0$ & $[0,1]$ & many maximizers\\
    \bottomrule
  \end{tabular}

  \vspace{.75em}
  \begin{block}{Existence and Uniqueness Are Separate}
    An empty argmax means no maximizer exists. A nonempty argmax may contain one point or many points.
  \end{block}
  \sources{Sundaram, Section 2.1.}
\end{frame}

\begin{frame}{Continuity and Compactness Guarantee Extrema}
  \begin{theorembox}[Extreme Value Theorem: Practical Form]
    If the choice set $D\subseteq\mathbb{R}$ is nonempty and compact and the
    objective $f:D\to\mathbb{R}$ is continuous, then there are points
    $x^{\max},x^{\min}\in D$ such that
    \[
      f(x^{\max})\geq f(x)\geq f(x^{\min})
      \qquad\text{for every }x\in D.
    \]
  \end{theorembox}

  \vspace{.25em}
  \begin{block}{In One Dimension}
    Compact means closed and bounded. Thus a closed bounded interval $[a,b]$
    is the canonical compact choice set.
  \end{block}

  \vspace{.20em}
  \centering
  \textcolor{slideblue}{\bfseries Continuous objective $+$ compact choice set
  $\Longrightarrow$ maximum and minimum are attained.}

  \vspace{.10em}
  These conditions guarantee existence, not uniqueness.
  \sources{Sundaram, Theorem 3.1 and the discussion preceding it.}
\end{frame}

\begin{frame}{Unconstrained Means Interior to the Feasible Set}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Local Maximum]
        \begin{minipage}[t][2.70cm][t]{\dimexpr\linewidth-.76em\relax}
          $x^*$ is a local maximizer if there is some $\varepsilon>0$ such that
          \[
            f(x^*)\geq f(x)
          \]
          whenever $x\in D$ and $|x-x^*|<\varepsilon$.
        \end{minipage}
      \end{definitionbox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Global Maximum]
        \begin{minipage}[t][2.70cm][t]{\dimexpr\linewidth-.76em\relax}
          $x^*$ is a global maximizer if
          \[
            f(x^*)\geq f(x)
          \]
          for every $x\in D$.
        \end{minipage}
      \end{definitionbox}
    \end{column}
  \end{columns}

  \vspace{.55em}
  An \emph{unconstrained} first-order calculation assumes that $x^*$ is an interior point:
  sufficiently small movements both left and right are feasible.
  \sources{Sundaram, Sections 4.1 and 4.2.}
\end{frame}

\begin{frame}{A Nonzero Slope Leaves an Improving Direction}
  \centering
  \includegraphics[width=.88\textwidth]{foc-slope.pdf}

  \vspace{.2em}
  If $f'(x^*)>0$, moving slightly right raises $f$; if $f'(x^*)<0$, moving slightly left raises $f$.
  \sources{Sundaram, Figure 4.2 and Theorem 4.1.}
\end{frame}

\begin{frame}{Why the First-Order Condition Is Necessary}
  \setlength{\abovedisplayskip}{.30em}
  \setlength{\belowdisplayskip}{.30em}
  \begin{propositionbox}[First-Order Necessary Condition]
    If $x^*$ is an interior local maximizer and $f$ is differentiable at $x^*$,
    then $f'(x^*)=0$.
  \end{propositionbox}

  \statementitem{1}{For sufficiently small $h>0$, local optimality gives
  \[
    \frac{f(x^*+h)-f(x^*)}{h}\leq0.
  \]}
  \statementitem{2}{For sufficiently small $h<0$, the numerator is still nonpositive but the denominator is negative, so
  \[
    \frac{f(x^*+h)-f(x^*)}{h}\geq0.
  \]}
  \statementitem{3}{Differentiability makes the two one-sided limits equal. Thus
  $f'(x^*)\leq0$ and $f'(x^*)\geq0$, which implies $f'(x^*)=0$.}
  \sources{Sundaram, Theorem 4.1 and Section 4.5.}
\end{frame}

\begin{frame}{A Critical Point Is Only a Candidate}
  A point satisfying $f'(x)=0$ is a \emph{critical point}. The condition alone does not identify its role.

  \vspace{.5em}
  \renewcommand{\arraystretch}{1.40}
  \centering
  \begin{tabular}{@{}ccc@{}}
    \toprule
    \textbf{Function} & \textbf{At $x=0$} & \textbf{Conclusion}\\
    \midrule
    $f(x)=-x^2$ & $f'(0)=0$ & strict maximum\\
    $f(x)=x^2$ & $f'(0)=0$ & strict minimum\\
    $f(x)=x^3$ & $f'(0)=0$ & neither\\
    \bottomrule
  \end{tabular}

  \vspace{.8em}
  \begin{block}{Boundary Warning}
    A boundary maximizer need not satisfy $f'(x)=0$, because movement may be feasible in only one direction.
  \end{block}
  \sources{Sundaram, Example 4.2.}
\end{frame}

\begin{frame}{The Second Derivative Gives a Local Test}
  Suppose $f$ is twice continuously differentiable on a neighborhood of $x^*$.

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{propositionbox}[Necessary at a Smooth Maximum]
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          If $x^*$ is an interior local maximizer and $f''(x^*)$ exists, then
          \[
            f''(x^*)\leq0.
          \]
        \end{minipage}
      \end{propositionbox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{propositionbox}[Sufficient for a Strict Local Maximum]
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          If
          \[
            f'(x^*)=0
            \quad\text{and}\quad
            f''(x^*)<0,
          \]
          then $x^*$ is a strict local maximizer.
        \end{minipage}
      \end{propositionbox}
    \end{column}
  \end{columns}

  \vspace{.55em}
  If $f''(x^*)=0$, the test is inconclusive:
  \[
    x^4\text{ has a minimum},
    \qquad -x^4\text{ has a maximum},
    \qquad x^3\text{ has neither at }0.
  \]
  \sources{Sundaram, Theorem 4.3 and Examples 4.4--4.5.}
\end{frame}

\begin{frame}{Concavity Turns a Local Condition into a Global Conclusion}
  \begin{definitionbox}[Concavity on an Interval]
    For every $x,y$ in the interval and every $t\in[0,1]$,
    \[
      f(tx+(1-t)y)\geq t f(x)+(1-t)f(y).
    \]
  \end{definitionbox}

  \begin{propositionbox}[Global First-Order Test]
    If $f$ is differentiable and concave on an interval, then
    \[
      f'(x^*)=0
      \quad\Longrightarrow\quad
      x^*\text{ is a global maximizer.}
    \]
    If $f$ is strictly concave, there is at most one maximizer.
  \end{propositionbox}

  For a twice differentiable scalar function, $f''(x)\leq0$ throughout an interval is a convenient test for concavity.
  \sources{Sundaram, Sections 7.2--7.4, especially Theorems 7.14--7.15.}
\end{frame}

\begin{frame}{Existence, Candidates, and Comparison Do Different Jobs}
  For a continuous function on a closed bounded interval $[a,b]$:

  \begin{enumerate}
    \item \textbf{Existence:} Weierstrass guarantees that a maximizer exists.
    \item \textbf{Interior candidates:} solve $f'(x)=0$ for $x\in(a,b)$.
    \item \textbf{Boundary candidates:} include $a$ and $b$.
    \item \textbf{Verification:} compare the objective at every candidate.
  \end{enumerate}

  \vspace{.7em}
  \begin{block}{Why First-Order Conditions Are Not Enough}
    They narrow the search. They neither guarantee that a maximizer exists nor establish that a candidate is globally best.
  \end{block}
  \sources{Sundaram, Sections 4.1--4.4.}
\end{frame}

\begin{frame}{An Equality Constraint Leaves One Free Variable}
  Consider
  \[
    \max_{x,y} f(x,y)
    \qquad\text{subject to}\qquad
    g(x,y)=0.
  \]
  Suppose that, near a candidate $(x^*,y^*)$, the constraint determines
  $y=q(x)$. Then the constrained problem becomes
  \[
    \max_x F(x),
    \qquad
    F(x)=f(x,q(x)).
  \]

  Differentiating the constraint $g(x,q(x))=0$ gives
  \[
    g_x(x,q(x))+g_y(x,q(x))q'(x)=0,
    \qquad
    q'(x)=-\frac{g_x}{g_y},
  \]
  provided $g_y\neq0$. Here $g_x=\partial g/\partial x$ and $g_y=\partial g/\partial y$.
  \sources{Sundaram, Section 5.1 and the proof idea for Theorem 5.1.}
\end{frame}

\begin{frame}{The Chain Rule Produces the Tangency Condition}
  \setlength{\abovedisplayskip}{.25em}
  \setlength{\belowdisplayskip}{.25em}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.52\textwidth}
      At an interior constrained optimum,
      \[
        \begin{aligned}
          0=F'(x^*)
          &=f_x+f_yq'(x^*)\\
          &=f_x-f_y\frac{g_x}{g_y}.
        \end{aligned}
      \]
      Therefore,
      \[
        f_xg_y=f_yg_x.
      \]
      When the relevant derivatives are nonzero,
      \[
        \boxed{-\frac{f_x}{f_y}=-\frac{g_x}{g_y}.}
      \]
      The objective level curve and the constraint have the same slope.
    \end{column}
    \begin{column}{.46\textwidth}
      \centering
      \includegraphics[width=.88\linewidth]{lagrange-tangency.pdf}
    \end{column}
  \end{columns}
  \sources{Sundaram, Sections 5.1--5.2.}
\end{frame}

\begin{frame}{The Lagrangean Packages the Same Calculation}
  Define the Lagrangean
  \[
    L(x,y;\lambda)=f(x,y)+\lambda g(x,y).
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Stationarity}
        \begin{minipage}[t][2.70cm][t]{\dimexpr\linewidth-.76em\relax}
          \[
            \frac{\partial L}{\partial x}=f_x+\lambda g_x=0,
          \]
          \[
            \frac{\partial L}{\partial y}=f_y+\lambda g_y=0.
          \]
          Eliminating $\lambda$ gives $f_xg_y=f_yg_x$.
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Feasibility}
        \begin{minipage}[t][2.70cm][t]{\dimexpr\linewidth-.76em\relax}
          \[
            \frac{\partial L}{\partial\lambda}=g(x,y)=0.
          \]
          The candidate must remain on the constraint.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.15em}
  These equations generate candidates; they do not prove global optimality.
  \sources{Sundaram, Sections 5.1--5.3.}
\end{frame}

\begin{frame}{Regularity Makes the Multiplier Exist and Be Unique}
  \setlength{\abovedisplayskip}{.25em}
  \setlength{\belowdisplayskip}{.25em}
  \begin{theorembox}[Theorem of Lagrange: One Equality]
    Let $U$ be an open neighborhood of $(x^*,y^*)$, and suppose $f$ and $g$ are continuously differentiable on $U$. If $(x^*,y^*)$ is a local optimum on
    $U\cap\{(x,y):g(x,y)=0\}$, and at least one of $g_x(x^*,y^*)$ and
    $g_y(x^*,y^*)$ is nonzero, then there is a multiplier $\lambda^*$ satisfying
    \[
      f_x(x^*,y^*)+\lambda^*g_x(x^*,y^*)=0,
      \qquad
      f_y(x^*,y^*)+\lambda^*g_y(x^*,y^*)=0.
    \]
  \end{theorembox}

  The same regularity condition makes $\lambda^*$ unique \emph{at the fixed optimizer} $(x^*,y^*)$.

  \begin{block}{What the Theorem Does Not Say}
    It does not establish that an optimizer exists, and it does not make the optimizer unique.
  \end{block}
  \sources{Sundaram, Theorem 5.1 and Section 5.2.2.}
\end{frame}

\begin{frame}{The Lagrange Procedure Starts Before the Lagrangean}
  \begin{enumerate}
    \item \textbf{Establish existence.} Show that the original constrained problem has a global optimum.
    \item \textbf{Check regularity.} Verify that $g_x$ and $g_y$ are not both zero at feasible candidates.
    \item \textbf{Generate candidates.} Solve $L_x=0$, $L_y=0$, and $L_\lambda=0$.
    \item \textbf{Verify.} Compare feasible candidates, or use concavity to establish global optimality.
  \end{enumerate}

  \vspace{.65em}
  \begin{block}{Two Ways the Cookbook Can Fail}
    If no optimum exists, there is nothing to characterize. If regularity fails, an optimum may exist without any multiplier solving the stationarity equations.
  \end{block}
  \sources{Sundaram, Sections 5.2--5.4.}
\end{frame}

\begin{frame}{A Multiplier Can Measure the Value of a Constraint Change}
  Let the constraint depend on a scalar parameter $c$:
  \[
    g(x,y)+c=0.
  \]
  Suppose the optimizer $(x(c),y(c))$ and its multiplier $\lambda(c)$ vary smoothly, and define the optimized value
  \[
    V(c)=f(x(c),y(c)).
  \]

  With the sign convention $L=f+\lambda[g+c]$,
  \[
    \boxed{V'(c)=\lambda(c).}
  \]

  \begin{itemize}
    \item $\lambda(c)$ is the marginal change in optimized value from increasing $c$.
    \item This interpretation needs the additional smoothness just stated.
    \item The number changes if the objective or the constraint is rescaled.
  \end{itemize}
  \sources{Sundaram, Section 5.2.3.}
\end{frame}

\begin{frame}{Kuhn--Tucker Adds a Boundary Condition}
  Consider the one-dimensional inequality problem
  \[
    \max_{x\geq0}f(x),
    \qquad
    L(x;\lambda)=f(x)+\lambda x.
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Interior Candidate}
        \begin{minipage}[t][1.45cm][t]{\dimexpr\linewidth-.76em\relax}
          If $x^*>0$, both directions are feasible:
          \[
            f'(x^*)=0,
            \qquad \lambda^*=0.
          \]
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Boundary Candidate}
        \begin{minipage}[t][1.45cm][t]{\dimexpr\linewidth-.76em\relax}
          If $x^*=0$, feasible moves go only right:
          \[
            f'(0)\leq0,
            \qquad \lambda^*=-f'(0)\geq0.
          \]
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  Both cases satisfy
  \[
    f'(x^*)+\lambda^*=0,
    \qquad x^*\geq0,
    \qquad \lambda^*\geq0,
    \qquad \lambda^*x^*=0.
  \]
  For a concave $f$, these conditions are also sufficient for global optimality.
  \sources{Sundaram, Sections 6.1--6.2.}
\end{frame}

\begin{frame}{Time for Economics}
  \begin{itemize}
    \item The objective $f$ becomes utility or profit.
    \item The feasible interval becomes a budget set or a technological restriction.
    \item Equality constraints describe choices on a binding budget.
    \item Nonnegativity creates corners, where ordinary tangency is inappropriate.
  \end{itemize}

  \vspace{.7em}
  \begin{block}{The Order of Reasoning Does Not Change}
    Establish existence, generate all relevant candidates, and then verify which candidate is globally best.
  \end{block}
  \sources{Sundaram, Sections 5.5 and 6.3.}
\end{frame}

\begin{frame}{Consumer Demand Is an Argmax Problem}
  A consumer chooses two nonnegative quantities $x_1$ and $x_2$, has income $I>0$, and faces prices $p_1,p_2>0$.

  \begin{definitionbox}[Budget Set]
    \[
      B(p_1,p_2,I)
      =\{(x_1,x_2):x_1,x_2\geq0,\ p_1x_1+p_2x_2\leq I\}.
    \]
  \end{definitionbox}

  \begin{definitionbox}[Consumer Problem]
    \[
      (x_1^*,x_2^*)\in
      \operatorname*{arg\,max}_{(x_1,x_2)\in B(p_1,p_2,I)}u(x_1,x_2).
    \]
  \end{definitionbox}

  The set of maximizing pairs is the demand correspondence; it becomes a function when the maximizer is unique.
  \sources{Sundaram, Sections 2.2--2.3 and 5.5.1.}
\end{frame}

\begin{frame}{The Assumptions from Day 3 Now Do Separate Jobs}
  \renewcommand{\arraystretch}{1.34}
  \centering
  \begin{tabular}{@{}p{.37\textwidth}p{.56\textwidth}@{}}
    \toprule
    \textbf{Assumption} & \textbf{Consequence}\\
    \midrule
    $p_1,p_2>0$ and $I>0$ & $0\leq x_1\leq I/p_1$ and $0\leq x_2\leq I/p_2$; the budget set is closed and bounded.\\
    Continuity of $u$ & At least one affordable bundle maximizes utility.\\
    Strict quasiconcavity of $u$ & At most one bundle maximizes utility on the convex budget set.\\
    Local nonsatiation & Every optimal bundle exhausts the budget.\\
    \bottomrule
  \end{tabular}

  \vspace{.75em}
  Existence, uniqueness, and budget exhaustion are three different conclusions.
  \sources{Sundaram, Examples 3.6 and 5.5.1; Math Bootcamp Day 3.}
\end{frame}

\begin{frame}{Local Nonsatiation Makes the Budget Bind}
  \begin{propositionbox}[Budget Exhaustion]
    If preferences are locally nonsatiated and $(x_1^*,x_2^*)$ solves the consumer problem, then
    \[
      p_1x_1^*+p_2x_2^*=I.
    \]
  \end{propositionbox}

  \statementitem{1}{Suppose instead that $p_1x_1^*+p_2x_2^*<I$. There is positive spending slack.}
  \statementitem{2}{Because prices are finite, every sufficiently nearby nonnegative bundle remains affordable.}
  \statementitem{3}{Local nonsatiation gives a sufficiently nearby bundle $(y_1,y_2)$ that is strictly preferred.}
  \statementitem{4}{That bundle is affordable and better, contradicting optimality.}

  \vspace{.45em}
  Therefore the budget inequality can be replaced by an equality when solving this problem.
  \sources{Sundaram, Section 5.5.1; Math Bootcamp Day 3.}
\end{frame}

\begin{frame}{A Binding Budget Reduces Consumer Choice to One Variable}
  \setlength{\abovedisplayskip}{.22em}
  \setlength{\belowdisplayskip}{.22em}
  Budget exhaustion gives
  \[
    \begin{gathered}
      x_2=\frac{I-p_1x_1}{p_2},
      \qquad 0\leq x_1\leq\frac{I}{p_1},\\[-.1em]
      \phi(x_1)=u\!\left(x_1,\frac{I-p_1x_1}{p_2}\right).
    \end{gathered}
  \]
  At a differentiable interior optimum $x_1^*\in(0,I/p_1)$,
  \[
    0=\phi'(x_1^*)
    =u_1(x_1^*,x_2^*)-\frac{p_1}{p_2}u_2(x_1^*,x_2^*),
  \]
  where $u_i=\partial u/\partial x_i$. Hence
  \[
    \boxed{\frac{u_1(x_1^*,x_2^*)}{p_1}
    =\frac{u_2(x_1^*,x_2^*)}{p_2}.}
  \]
  \sources{Sundaram, Section 5.5.1; the derivation mirrors Section 5.1.}
\end{frame}

\begin{frame}{The Consumer Lagrangean Gives the Same Equations}
  \setlength{\abovedisplayskip}{.30em}
  \setlength{\belowdisplayskip}{.30em}
  Write
  \[
    \begin{aligned}
      L(x_1,x_2;\lambda)
      &=u(x_1,x_2)+\lambda(I-p_1x_1-p_2x_2),\\
      u_1(x_1^*,x_2^*)&=\lambda^*p_1,\\
      u_2(x_1^*,x_2^*)&=\lambda^*p_2,\\
      p_1x_1^*+p_2x_2^*&=I.
    \end{aligned}
  \]

  \begin{block}{Interior Interpretation}
    Each consumed good yields the same marginal utility per unit of money:
    $u_1/p_1=u_2/p_2=\lambda^*$.
  \end{block}
  This tangency condition applies only when both quantities are strictly positive and utility is differentiable.
  \sources{Sundaram, Section 5.5.1.}
\end{frame}

\begin{frame}{Cobb--Douglas Has an Interior Solution}
  Let
  \[
    u(x_1,x_2)=x_1^\alpha x_2^{1-\alpha},
    \qquad 0<\alpha<1.
  \]

  \begin{enumerate}
    \item Utility equals zero on either axis.
    \item With $I>0$ and positive prices, an affordable bundle with $x_1,x_2>0$ exists and gives positive utility.
    \item Therefore no optimal bundle lies on an axis: $x_1^*,x_2^*>0$.
  \end{enumerate}

  Thus we may first restrict attention to $x_1,x_2>0$. Only then may we use the strictly increasing transformation
  \[
    \log u(x_1,x_2)=\alpha\log x_1+(1-\alpha)\log x_2.
  \]
  \sources{Sundaram, Theorem 2.2 and Section 5.5.1.}
\end{frame}

\begin{frame}{Cobb--Douglas Demand Spends Fixed Income Shares}
  The first-order conditions for the transformed objective are
  \[
    \frac{\alpha}{x_1}=\lambda p_1,
    \qquad
    \frac{1-\alpha}{x_2}=\lambda p_2,
    \qquad
    p_1x_1+p_2x_2=I.
  \]
  Multiplying the first two equations by $x_1$ and $x_2$ gives
  \[
    \alpha=\lambda p_1x_1,
    \qquad
    1-\alpha=\lambda p_2x_2.
  \]
  Adding them and using the budget gives $\lambda=1/I$. Therefore,
  \[
    \boxed{x_1^*=\frac{\alpha I}{p_1},
    \qquad
    x_2^*=\frac{(1-\alpha)I}{p_2}.}
  \]
  The consumer spends the shares $\alpha$ and $1-\alpha$ of income on the two goods.
  The log transformation preserves the bundle, but not the numerical multiplier.
  \sources{Sundaram, Section 5.5.1.}
\end{frame}

\begin{frame}{Perfect Substitutes Require a Comparison, Not a Tangency}
  Let
  \[
    u(x_1,x_2)=a x_1+b x_2,
    \qquad a,b>0.
  \]
  Utility per unit of money is constant: compare $a/p_1$ with $b/p_2$.
  Write $x(p_1,p_2,I)$ for the set of utility-maximizing affordable pairs.

  \begin{definitionbox}[Demand]
    \[
      x(p_1,p_2,I)=
      \begin{cases}
        \{(I/p_1,0)\}, & a/p_1>b/p_2,\\[.25em]
        \{(0,I/p_2)\}, & a/p_1<b/p_2,\\[.25em]
        \{(x_1,x_2):x_1,x_2\geq0,\ p_1x_1+p_2x_2=I\},
          & a/p_1=b/p_2.
      \end{cases}
    \]
  \end{definitionbox}

  The first two cases are corners; the equality case has many maximizers.
  \sources{Sundaram, Chapters 2 and 6; standard perfect-substitutes application.}
\end{frame}

\begin{frame}{Leontief Utility Is Solved at the Kink}
  \setlength{\abovedisplayskip}{.28em}
  \setlength{\belowdisplayskip}{.28em}
  Let
  \[
    u(x_1,x_2)=\min\{a x_1,b x_2\},
    \qquad a,b>0.
  \]
  Unmatched units do not raise utility, so the optimal bundle satisfies
  \[
    a x_1=b x_2.
  \]
  Local nonsatiation also makes the budget bind:
  \[
    p_1x_1+p_2x_2=I.
  \]
  Solving these two scalar equations gives
  \[
    \boxed{x_1^*=\frac{bI}{bp_1+ap_2},
    \qquad
    x_2^*=\frac{aI}{bp_1+ap_2}.}
  \]

  The relevant point is a kink, so ordinary derivative tangency does not apply.
  \sources{Standard Leontief consumer problem; direct geometric reasoning at the kink.}
\end{frame}

\begin{frame}{A Static Firm Compares Marginal Product with the Wage}
  Normalize the output price to one. A one-period firm chooses labor $\ell\geq0$:
  \[
    \max_{\ell\geq0}\{f(\ell)-w\ell\}.
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Existence}
        \begin{minipage}[t][2.20cm][t]{\dimexpr\linewidth-.76em\relax}
          A solution exists, for example, if labor is restricted to a closed bounded interval and profit is continuous.
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Identification}
        \begin{minipage}[t][2.20cm][t]{\dimexpr\linewidth-.76em\relax}
          For differentiable, concave $f$:
          \[
            \ell^*>0\Rightarrow f'(\ell^*)=w,
          \]
          while $f'_+(0)\leq w$ supports $\ell^*=0$.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.35em}
  Here $f'_+(0)$ is the right derivative. The problem is static: hiring, production, and payment all occur in the same period.
  \sources{Sundaram, Chapters 3, 4, and 6; static profit-maximization application.}
\end{frame}

\begin{frame}{Linear Production Reopens the Existence Question}
  Let one unit of labor produce $A>0$ units of output:
  \[
    f(\ell)=A\ell,
    \qquad
    \pi(\ell)=(A-w)\ell.
  \]
  Marginal product is constant, so there is no interior tangency to find.

  \begin{definitionbox}[Profit Maximizers without a Capacity Constraint]
    \[
      \operatorname*{arg\,max}_{\ell\geq0}(A-w)\ell
      =
      \begin{cases}
        \{0\}, & A<w,\\[.2em]
        [0,\infty), & A=w,\\[.2em]
        \varnothing, & A>w.
      \end{cases}
    \]
  \end{definitionbox}

  With a capacity $0\leq\ell\leq\bar\ell$, the case $A>w$ has the solution $\ell^*=\bar\ell$.
  \sources{Sundaram, Sections 2.1, 3.1, and 4.1; linear production application.}
\end{frame}

\begin{frame}{First-Order Conditions Narrow the Search}
  \begin{enumerate}
    \item Write the objective and the feasible set.
    \item Establish that a maximizer exists; ask separately whether it is unique.
    \item Classify the possible solution: interior, equality-constrained, boundary, or kink.
    \item Generate every candidate with derivatives, the Lagrangean, Kuhn--Tucker, or direct comparison.
    \item Verify feasibility and global optimality.
  \end{enumerate}

  \vspace{.75em}
  \begin{block}{The Discipline of Optimization}
    \centering
    \Large Existence $\longrightarrow$ candidates $\longrightarrow$ verification.
  \end{block}
  \sources{Sundaram, Chapters 3--6.}
\end{frame}
