\begin{frame}{The Maximum and the Maximizers Are Different Objects}
  Consider
  \[
    \max\{f(x)\mid x\in D\}.
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Maximum Value]
        If a solution exists, then
        \[
          \max\{f(x)\mid x\in D\}=f(x^*)\in\mathbb R.
        \]
        This is a number.
      \end{definitionbox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Maximizers]
        \[
          \begin{gathered}
            \arg\max\{f(x)\mid x\in D\}\\[-.2em]
            =\{x\in D:f(x)\geq f(y)\ \forall y\in D\}.
          \end{gathered}
        \]
        This is a set of choices.
      \end{definitionbox}
    \end{column}
  \end{columns}

  \vspace{.5em}
  The argmax may be empty, contain one point, or contain many points.
  \sources{Sundaram, Section 2.1; Osborne, Optimization: definitions.}
\end{frame}

\begin{frame}{Existence Comes Before Calculation}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{.315\textwidth}
      \centering
      \textbf{Open feasible set}
      \[
        \max_{x\in(0,1)}x
      \]
      Supremum $1$, but no maximizer.
    \end{column}
    \begin{column}{.315\textwidth}
      \centering
      \textbf{Unbounded objective}
      \[
        \max_{x\geq0}x
      \]
      No finite maximum.
    \end{column}
    \begin{column}{.315\textwidth}
      \centering
      \textbf{Discontinuity}
      \[
        f(x)=\begin{cases}x,&x<\frac12,\\x-1,&x\geq\frac12.
        \end{cases}
      \]
      On $[0,1]$, the supremum is not attained.
    \end{column}
  \end{columns}

  \vspace{.8em}
  \begin{theorembox}[Extreme Value Theorem]
    If $D\subseteq\mathbb R^n$ is nonempty and compact and
    $f:D\to\mathbb R$ is continuous, then $f$ attains a maximum and a minimum on $D$.
  \end{theorembox}
  \sources{Sundaram, Theorem 3.1; Osborne, Existence of an optimum.}
\end{frame}

\begin{frame}{Why the First-Order Condition Is Necessary}
  \setlength{\abovedisplayskip}{.35em}
  \setlength{\belowdisplayskip}{.35em}
  \begin{propositionbox}[First-Order Necessary Condition]
    If $x^*\in\operatorname{int}D\subseteq\mathbb R$ is a local maximizer of $f$
    on $D$ and $f$ is differentiable at $x^*$, then $f'(x^*)=0$.
  \end{propositionbox}

  \vspace{.05em}
  \statementitem{1}{For sufficiently small $h>0$, local optimality gives
  \[
    f(x^*+h)\leq f(x^*)
    \quad\Longrightarrow\quad
    \frac{f(x^*+h)-f(x^*)}{h}\leq0.
  \]}
  \statementitem{2}{For sufficiently small $h<0$, the numerator is still
  nonpositive but the denominator is negative, so
  \[
    \frac{f(x^*+h)-f(x^*)}{h}\geq0.
  \]}
  \statementitem{3}{The two one-sided limits are equal because $f$ is
  differentiable. Hence $f'(x^*)\leq0$ and $f'(x^*)\geq0$, so $f'(x^*)=0$.}
  \sources{Osborne, Proposition 5.1.1; Sundaram, Theorem 4.1 and Section 4.5.}
\end{frame}

\begin{frame}{The Gradient Tests Every Direction at Once}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.51\textwidth}
      For a direction $d\in\mathbb R^n$,
      \[
        f(x+td)-f(x)\approx t\,\nabla f(x)\cdot d.
      \]
      If $x^*$ is an interior local maximum, define
      \[
        \phi_d(t)=f(x^*+td).
      \]
      Then $t=0$ locally maximizes $\phi_d$, so
      \[
        \phi_d'(0)=\nabla f(x^*)\cdot d=0
        \qquad\text{for every }d.
      \]
      Therefore,
      \[
        \boxed{\nabla f(x^*)=0.}
      \]
    \end{column}
    \begin{column}{.47\textwidth}
      \centering
      \includegraphics[width=.96\linewidth]{gradient-directions.pdf}
    \end{column}
  \end{columns}
  \sources{Osborne, Proposition 5.1.2; Sundaram, Section 4.5, directional-derivative proof.}
\end{frame}

\begin{frame}{Curvature Turns Candidates into Conclusions}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{propositionbox}[Local Conclusion]
        Suppose $f$ is twice differentiable and
        \[
          \nabla f(x^*)=0.
        \]
        If $\nabla^2f(x^*)$ is negative definite, then $x^*$ is a strict local maximizer.
      \end{propositionbox}
      Negative semidefiniteness at $x^*$ is necessary for a smooth local maximum,
      but may not be sufficient.
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{propositionbox}[Global Conclusion]
        Suppose $D$ is convex and $f$ is differentiable and concave on $D$.
        Then
        \[
          \begin{gathered}
            \nabla f(x^*)=0\quad\Longrightarrow\\[-.15em]
            x^*\in\arg\max\{f(x)\mid x\in D\}.
          \end{gathered}
        \]
      \end{propositionbox}
      If $f$ is strictly concave, the maximizer is unique whenever it exists.
    \end{column}
  \end{columns}

  \vspace{.5em}
  First-order conditions generate candidates; curvature and concavity tell us
  what those candidates mean.
  \sources{Sundaram, Theorems 4.3, 7.14, and 7.15; Osborne, Local optima and
  global conditions.}
\end{frame}

\begin{frame}{A Constraint Removes Directions}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.50\textwidth}
      Consider
      \[
        \max_x f(x)
        \qquad\text{subject to}\qquad g(x)=0.
      \]
      A direction $d$ tangent to the constraint satisfies, to first order,
      \[
        \nabla g(x)\cdot d=0.
      \]
      At a constrained optimum, the objective must not increase along any such
      feasible direction.

      \vspace{.5em}
      The gradient $\nabla g(x)$ is normal to the constraint curve.
    \end{column}
    \begin{column}{.48\textwidth}
      \centering
      \includegraphics[width=.96\linewidth]{constraint-tangent.pdf}
    \end{column}
  \end{columns}
  \sources{Osborne, Equality constraints: motivation; Chong and Zak, tangent and
  normal spaces, printed pp. 368--378.}
\end{frame}

\begin{frame}{If the Curves Cross, Improvement Remains Possible}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.45\textwidth}
      If the feasible curve crosses an objective level curve, one of the two
      feasible directions leads toward a higher level.

      \vspace{.7em}
      At the marked point,
      \[
        \nabla f(x)\cdot d>0.
      \]
      A small feasible movement in direction $d$ raises the objective, so the
      point cannot be optimal.
    \end{column}
    \begin{column}{.53\textwidth}

\end{frame}

\begin{frame}{At a Smooth Constrained Optimum, the Normals Align}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.45\textwidth}
      At tangency, the objective level curve and the feasible curve share the
      same tangent direction.

      Their normal vectors must therefore be parallel:
      \[
        \boxed{\nabla f(x^*)+\lambda^*\nabla g(x^*)=0.}
      \]
      Equivalently, every feasible tangent direction $d$ satisfies
      \[
        \nabla f(x^*)\cdot d=0.
      \]
    \end{column}
    \begin{column}{.53\textwidth}

      \includegraphics[width=.90\linewidth]{lagrange-tangency.pdf}
    \end{column}
  \end{columns}

  Figures 19.10 and 19.12.}
\end{frame}

\begin{frame}{The Lagrangean Packages Tangency and Feasibility}
  For
  \[
    \max_x f(x)
    \qquad\text{subject to}\qquad g(x)=0,
  \]
  define
  \[
    L(x;\lambda)=f(x)+\lambda g(x).
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Stationarity}
        \[
          \begin{aligned}
            \nabla_x L(x^*;\lambda^*)&=0,\\
            \nabla f(x^*)&+\lambda^*\nabla g(x^*)=0.
          \end{aligned}
        \]
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Feasibility}
        \[
          g(x^*)=0.
        \]
        The candidate must satisfy the constraint.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.4em}
  These equations generate candidates. Concavity or direct comparison is still
  needed to establish global optimality.
  \sources{Sundaram, Theorem 5.1 and Section 5.3; Osborne, Equality constraints:
  necessary conditions.}
\end{frame}

\begin{frame}{Regular Constraints Make the Multiplier Well Defined}
  \begin{theorembox}[Theorem of Lagrange]
    Let $f,g:\mathbb R^n\to\mathbb R$ be continuously differentiable and let $U$
    be open. If $x^*$ is a local optimum of $f$ on
    $D=U\cap\{x:g(x)=0\}$ and $\nabla g(x^*)\neq0$, then there exists a unique
    $\lambda^*\in\mathbb R$ such that
    \[
      \nabla f(x^*)+\lambda^*\nabla g(x^*)=0.
    \]
  \end{theorembox}

  For $g=(g_1,\ldots,g_k)$, let $J_g$ denote its Jacobian. The constraint
  qualification is $\rho(J_g(x^*))=k$.

  \vspace{.25em}
  \begin{itemize}
    \item Regularity guarantees existence of the multiplier at a local optimum.
    \item Independence makes its coefficients unique at that fixed $x^*$.
    \item It does not guarantee that the optimizer itself is unique.
  \end{itemize}
  \sources{Sundaram, Theorem 5.1 and Section 5.2.2; Chong and Zak, Theorem 19.3.}
\end{frame}

\begin{frame}{A Multiplier Measures the Value of Relaxing a Constraint}
  Let $g(x;c)=g(x)+c$, let $x^*(c)$ solve
  $\max\{f(x)\mid g(x;c)=0\}$, and define $F(c)=f(x^*(c))$.
  If $x^*(c)$ varies smoothly, then $\boxed{F'(c)=\lambda^*(c)}$.

  \begin{columns}[c,onlytextwidth]
    \begin{column}{.49\textwidth}
      \begin{itemize}
        \item $\lambda^*$ is first a coefficient aligning two normal vectors.
        \item With the additional smoothness, it is the marginal optimized value
        of relaxing the constraint.
        \item Its value depends on the chosen representation of the objective.
      \end{itemize}
    \end{column}
    \begin{column}{.49\textwidth}
      \centering
      \includegraphics[width=.98\linewidth]{multiplier-relaxation.pdf}
    \end{column}
  \end{columns}
  \sources{Sundaram, Section 5.2.3; Osborne, Interpretation of Lagrange multipliers.}
\end{frame}

\begin{frame}{Inequality Constraints Are Effective or Slack}
  Write each inequality as $h_i(x)\geq0$. At a candidate $x^*$:

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Slack Constraint}
        \[
          h_i(x^*)>0.
        \]
        The constraint is slack. Small changes in its boundary do not affect the
        local choice, so $\lambda_i^*=0$.
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Effective Constraint}
        \[
          h_i(x^*)=0.
        \]
        The boundary restricts feasible directions and may carry a positive multiplier.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.25em}
  Both cases are summarized by
  \[
    \lambda_i^*\geq0,\qquad h_i(x^*)\geq0,\qquad
    \lambda_i^* h_i(x^*)=0.
  \]
  \sources{Sundaram, Sections 6.1.1 and 6.1.3; Osborne, Kuhn--Tucker conditions.}
\end{frame}

\begin{frame}{The Kuhn--Tucker Conditions Are One-Sided}
  \begin{theorembox}[Theorem of Kuhn and Tucker]
    Let $f,h_1,\ldots,h_\ell$ be continuously differentiable, let $U$ be open, and
    define $D=U\cap\{x\in\mathbb R^n:h_i(x)\geq0,\ i=1,\ldots,\ell\}$.
    Suppose $x^*$ is a local maximum of $f$ on $D$. Let
    $E=\{i:h_i(x^*)=0\}$ and suppose $\rho(J_{h_E}(x^*))=|E|$. Then there exists
    $\lambda^*\in\mathbb R^\ell$ such that
    \[
      \text{[KT-1]}\quad
      \lambda_i^*\geq0,\qquad \lambda_i^*h_i(x^*)=0,
      \quad i=1,\ldots,\ell,
    \]
    \[
      \text{[KT-2]}\quad
      \nabla f(x^*)+\sum_{i=1}^{\ell}\lambda_i^*\nabla h_i(x^*)=0.
    \]
  \end{theorembox}

  Only effective normals contribute; they generate a nonnegative cone, not a linear span.
  \sources{Sundaram, Theorem 6.1; Osborne, Kuhn--Tucker conditions; Chong and
  Zak, Figure 20.1.}
\end{frame}

\begin{frame}{Consumer Choice Is the Same Optimization Problem}
  A consumer chooses $x\in\mathbb R_+^n$, has income $I>0$, and faces prices
  $p\in\mathbb R_{++}^n$.

  \begin{definitionbox}[Walrasian Budget Set]
    \[
      B(p,I)=\{x\in\mathbb R_+^n:p\cdot x\leq I\}.
    \]
  \end{definitionbox}

  \begin{definitionbox}[Walrasian Demand Correspondence]
    \[
      x(p,I)=\arg\max\{u(x)\mid x\in B(p,I)\}.
    \]
    A selected optimum is written $x^*\in x(p,I)$.
  \end{definitionbox}

  Demand is a correspondence because several bundles may maximize utility.
  \sources{Sundaram, Sections 2.2--2.3; Graduate Microeconomics I, Lecture 5,
  slides 3--7.}
\end{frame}

\end{frame}

\begin{frame}{Local Nonsatiation Makes the Budget Bind}
  \begin{propositionbox}[Walras' Law for Individual Demand]
    If preferences are locally nonsatiated and $x^*\in x(p,I)$, then
    \[
      p\cdot x^*=I.
    \]
  \end{propositionbox}

  \statementitem{1}{Suppose instead that $p\cdot x^*<I$. There is positive
  spending slack.}
  \statementitem{2}{Because prices are finite, every sufficiently close bundle
  remains affordable.}
  \statementitem{3}{Local nonsatiation gives a sufficiently close bundle $y$
  with $y\succ x^*$.}
  \statementitem{4}{Then $y\in B(p,I)$ and $y\succ x^*$, contradicting
  $x^*\in x(p,I)$.}

  \vspace{.5em}
  Hence the inequality constraint may be treated as the equality
  $p\cdot x=I$ when solving a locally nonsatiated consumer problem.
  \sources{Graduate Microeconomics I, Lecture 5, slide 9; Day 3 local
  nonsatiation.}
\end{frame}

\begin{frame}{A Money-Neutral Reallocation Is a Feasible Direction}
  Move $\delta>0$ units of money from good $j$ to good $i$:
  \[
    \Delta x_i=\frac{\delta}{p_i},
    \qquad
    \Delta x_j=-\frac{\delta}{p_j}.
  \]
  The change in expenditure is zero:
  \[
    p_i\Delta x_i+p_j\Delta x_j=\delta-\delta=0.
  \]

  To first order, the utility change is
  \[
    \Delta u
    \approx
    \delta\left(
      \frac{\partial u(x)/\partial x_i}{p_i}
      -
      \frac{\partial u(x)/\partial x_j}{p_j}
    \right).
  \]

  If the expression in parentheses is positive, the current bundle is not optimal.
  \sources{Dixit, Optimization in Economic Theory, printed pp. 4--8; notation
  aligned with Sundaram, Chapter 5.}
\end{frame}

\begin{frame}{At an Interior Optimum, No Money Switch Pays}
  If both goods are consumed in strictly positive amounts, both reallocation
  directions are feasible. Therefore,
  \[
    \boxed{
      \frac{\partial u(x^*)/\partial x_i}{p_i}
      =
      \frac{\partial u(x^*)/\partial x_j}{p_j}
      =\lambda^*.}
  \]

  Equivalently,
  \[
    \nabla u(x^*)=\lambda^*p,
    \qquad
    \frac{\partial u(x^*)/\partial x_i}{\partial u(x^*)/\partial x_j}
    =\frac{p_i}{p_j}.
  \]

  \begin{block}{Interpretation}
    Each consumed good produces the same marginal utility per unit of money.
    Under the smooth value-function assumptions,
    $\lambda^*=\partial v(p,I)/\partial I$.
  \end{block}
  \sources{Sundaram, Sections 5.2.3 and 5.5.1; Dixit, printed pp. 4--8;
  Graduate Microeconomics I, Lecture 5, slides 19--20.}
\end{frame}

\begin{frame}{A Monotone Transformation Simplifies Cobb--Douglas}
  Let
  \[
    u(x_1,x_2)=x_1^\alpha x_2^{1-\alpha},
    \qquad 0<\alpha<1.
  \]
  Because $\log$ is strictly increasing,
  \[
    \arg\max\{u(x)\mid x\in B(p,I)\}
    =
    \arg\max\{\alpha\log x_1+(1-\alpha)\log x_2\mid x\in B(p,I)\}.
  \]

  With local nonsatiation, use the binding budget and write
  \[
    L(x;\lambda)
    =\alpha\log x_1+(1-\alpha)\log x_2
     +\lambda(I-p_1x_1-p_2x_2).
  \]

  The maximizing bundle is unchanged by the transformation; the numerical value
  of the multiplier is not.
  \sources{Sundaram, Theorem 2.2 and Section 5.5.1; Osborne, Optimization:
  definitions and Example 6.1.1.3.}
\end{frame}

\begin{frame}{Cobb--Douglas Demand Allocates Fixed Shares of Income}
  The first-order conditions are
  \[
    \frac{\alpha}{x_1}=\lambda^* p_1,
    \qquad
    \frac{1-\alpha}{x_2}=\lambda^* p_2,
    \qquad
    p_1x_1+p_2x_2=I.
  \]
  Multiplying the first two equations by $x_1$ and $x_2$ gives
  \[
    \alpha=\lambda^* p_1x_1,
    \qquad
    1-\alpha=\lambda^* p_2x_2.
  \]
  Summing and using the budget yields $\lambda^*=1/I$, so
  \[
    \boxed{
      x_1(p,I)=\frac{\alpha I}{p_1},
      \qquad
      x_2(p,I)=\frac{(1-\alpha)I}{p_2}.}
  \]

  The consumer spends the shares $\alpha$ and $1-\alpha$ of income on the two goods.
  \sources{Sundaram, Section 5.5.1; Dixit, Cobb--Douglas example, printed
  pp. 19--25; Osborne, Example 6.1.1.3.}
\end{frame}

\begin{frame}{Perfect Substitutes Turn Tangency into a Comparison}
  Let $u(x_1,x_2)=a x_1+b x_2$, where $a,b>0$. Marginal utility per unit of
  money is constant:
  \[
    \frac{a}{p_1}
    \qquad\text{and}\qquad
    \frac{b}{p_2}.
  \]

  \begin{definitionbox}[Walrasian Demand]
    \[
      x(p,I)=
      \begin{cases}
        \{(I/p_1,0)\}, & a/p_1>b/p_2,\\[.25em]
        \{(0,I/p_2)\}, & a/p_1<b/p_2,\\[.25em]
        \{x\in\mathbb R_+^2:p\cdot x=I\}, & a/p_1=b/p_2.
      \end{cases}
    \]
  \end{definitionbox}

  Unless the two returns are equal, the optimum is a corner rather than a tangency.
  \sources{Dixit, corner solutions, printed pp. 6--8; standard consumer theory.}
\end{frame}

\begin{frame}{Leontief Utility Is Solved at the Kink}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.56\textwidth}
      Let
      \[
        u(x_1,x_2)=\min\{a x_1,b x_2\},
        \qquad a,b>0.
      \]
      Unmatched units do not raise utility, so an optimum satisfies
      \[
        a x_1=b x_2.
      \]
      Together with $p_1x_1+p_2x_2=I$,
      \[
        \boxed{
        x_1(p,I)=\frac{bI}{bp_1+ap_2},
        \qquad
        x_2(p,I)=\frac{aI}{bp_1+ap_2}.}
      \]
    \end{column}
    \begin{column}{.42\textwidth}
      \centering
      \includegraphics[width=.98\linewidth]{leontief-kink.pdf}
    \end{column}
  \end{columns}

  Ordinary smooth tangency conditions do not apply at the economically relevant kink.

\end{frame}

\begin{frame}{Kuhn--Tucker for the Consumer Problem}
  \setlength{\abovedisplayskip}{.35em}
  \setlength{\belowdisplayskip}{.35em}
  Write the constraints as $h_i(x)=x_i$ for $i=1,\ldots,n$ and
  $h_{n+1}(x)=I-p\cdot x$. With nonnegative multipliers,
  \[
    L(x;\lambda)
    =u(x)+\sum_{i=1}^n\lambda_i x_i
    +\lambda_{n+1}(I-p\cdot x).
  \]
  With local nonsatiation, the budget binds. For every good, stationarity and
  complementary slackness give
  \[
    \frac{\partial u(x^*)}{\partial x_i}
      +\lambda_i^*-\lambda_{n+1}^*p_i=0,
    \qquad
    \lambda_i^*x_i^*=0.
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      If $x_i^*>0$, then $\lambda_i^*=0$ and
      \[
        \frac{\partial u(x^*)/\partial x_i}{p_i}=\lambda_{n+1}^*.
      \]
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      If $x_i^*=0$, then
      \[
        \frac{\partial u(x^*)/\partial x_i}{p_i}\leq\lambda_{n+1}^*.
      \]
    \end{column}
  \end{columns}
  \sources{Sundaram, Sections 5.1, 6.1, and 6.3.1; Graduate Microeconomics I,
  Lecture 5, slides 14--20.}
\end{frame}

\begin{frame}{Every Case Asks the Same Question}
  \begin{block}{Central Question}
    \centering
    \Large Is there a feasible direction that raises the objective?
  \end{block}

  \vspace{.45em}
  \begin{enumerate}
    \item Write the objective and the feasible set.
    \item Establish existence and uniqueness separately.
    \item Identify whether the candidate is interior, on a smooth boundary, at a
    corner, or at a kink.
    \item Generate candidates using first-order conditions, the Lagrangean
    method, Kuhn--Tucker, or geometry.
    \item Verify feasibility, complementary slackness, and global optimality.
  \end{enumerate}

  \vspace{.4em}
  \centering
  \textcolor{slideblue}{\bfseries First-order conditions reduce the search;
  assumptions and verification finish it.}
  \sources{Sundaram, Chapters 4--6; Osborne, Optimization summary.}
\end{frame}
