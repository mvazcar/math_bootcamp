\begin{frame}{Every Dated Constraint Has Its Own Multiplier}
  An infinite problem has one constraint at each date, so its formal
  Lagrangean has one multiplier at each date:
  \[
    \max_{z=(z_t)_{t=0}^{\infty}} F(z)
    \quad\text{s.t.}\quad
    g_t(z)=0, t=0,1,\ldots,
    \qquad
    L(z;\lambda)=F(z)+\sum_{t=0}^{\infty}\lambda_t g_t(z).
  \]

  \centering
  \begin{tikzpicture}
    \foreach \x/\lab/\mul in {0/$g_0(z)=0$/$\lambda_0$,3.15/$g_1(z)=0$/$\lambda_1$,
      6.30/$\cdots$/$\cdots$,9.45/$g_t(z)=0$/$\lambda_t$}{
      \node[object,text width=2.0cm] at (\x,0) {\lab};
      \node[text=slideblue] at (\x,-.80) {\mul};
    }
  \end{tikzpicture}

  \vspace{.10em}
  \begin{block}{What Is Formal Here?}
    A general multiplier theorem for sequence spaces is beyond our scope. We
    use this expression to generate date-by-date candidates; feasibility,
    concavity, and a model-specific transversality condition do the verification.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 51--53 and 77; Sundaram, Chapter 5.}
\end{frame}

\begin{frame}{Next Period's State Appears in Two Neighboring Constraints}
  To keep the notation concrete, call the linking choice $a_{t+1}$: saving
  chosen at $t$ and carried into $t+1$. It appears twice:
  \[
    g_t(\ldots,a_{t+1})=0,
    \qquad
    g_{t+1}(a_{t+1},\ldots)=0.
  \]

  \centering
  \begin{tikzpicture}
    \node[object,text width=2.35cm] (left) at (0,0)
      {Date-$t$ constraint\\$g_t=0$};
    \node[object,fill=slidegold!9,draw=slidegold,text width=2.2cm] (state) at (4.0,0)
      {Saving today\\$a_{t+1}$};
    \node[object,text width=2.35cm] (right) at (8.0,0)
      {Date-$(t+1)$ constraint\\$g_{t+1}=0$};
    \draw[equality] (left) -- node[above,font=\small]{$\lambda_t$} (state);
    \draw[equality] (state) -- node[above,font=\small]{$\lambda_{t+1}$} (right);
  \end{tikzpicture}

  \vspace{.5em}
  Differentiating with respect to $a_{t+1}$ links adjacent multipliers.
  Eliminating them typically produces
  \[
    \Phi(a_t,a_{t+1},a_{t+2})=0.
  \]
  In the household problem, this recurrence will be the Euler equation.
  \sources{AdvancedMacro Book, printed pp. 51--53, 72--73, and 77--78.}
\end{frame}

\begin{frame}{The Euler Equation Alone Does Not Determine a Path}
  A recurrence alone does not determine a path:
  \[
    \Phi(a_t,a_{t+1},a_{t+2})=0,
    \qquad t=0,1,\ldots.
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Beginning of the Path}
        \begin{minipage}[t][1.75cm][t]{\dimexpr\linewidth-.76em\relax}
          The initial state is fixed:
          \[
            a_0\ \text{is given}.
          \]
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{End of the Path}
        \begin{minipage}[t][1.75cm][t]{\dimexpr\linewidth-.76em\relax}
          A model-specific condition controls the tail as $t\to\infty$.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.45em}
  The Euler equation is local. The initial state and the tail restriction select
  among candidate paths; neither proves that a path exists.
  \sources{AdvancedMacro Book, Appendix A, printed p. 86; Math Bootcamp Day 4.}
\end{frame}

\begin{frame}{The Household Ranks Entire Consumption Paths}
  \begin{definitionbox}[Lifetime Utility]
    For $c=(c_0,c_1,c_2,\ldots)$,
    \[
      U(c)=\sum_{t=0}^{\infty}\beta^t u(c_t),
      \qquad 0<\beta<1.
    \]
    The discount factor $\beta$ gives the relative weight on later utility.
    We assume that this sum is well-defined and finite on admissible paths.
  \end{definitionbox}

  \vspace{.2em}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \textbf{Monotonicity}
      \[
        u'(c_t)>0
      \]
      More consumption is preferred.
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \textbf{Strict concavity}
      \[
        u''(c_t)<0
      \]
      Marginal utility falls with consumption.
    \end{column}
  \end{columns}
  \sources{AdvancedMacro Book, printed pp. 41--49.}
\end{frame}

\begin{frame}{The Household Chooses a Consumption--Saving Path}
  Given the price paths $\left(w_t,r_t\right)_{t=0}^{\infty}$ and initial assets
  $a_0$, the household solves
  \[
    \max_{\left(c_t,a_{t+1}\right)_{t=0}^{\infty}}
    \sum_{t=0}^{\infty}\beta^t u(c_t)
  \]
  subject to
  \[
    c_t+a_{t+1}=w_t+(1+r_t)a_t,
    \qquad c_t\geq0,
    \qquad a_{t+1}\geq0,
    \qquad t=0,1,\ldots,
  \]
  \[
    a_0\ \text{given}.
  \]

  \begin{block}{Admissibility and Price Taking}
    \textbf{Admissibility:} $a_{t+1}\geq0$ forbids borrowing; it is stronger than
    a no-Ponzi condition.\par\smallskip
    \textbf{Price taking:} the household treats $(w_t,r_t)$ as given.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 51--52.}
\end{frame}

\begin{frame}{Each Dated Budget Constraint Has Its Own Shadow Value}
  At an interior path, write the formal Lagrangean
  \[
    L=
    \sum_{t=0}^{\infty}\beta^t u(c_t)
    +
    \sum_{t=0}^{\infty}
    \lambda_t\left[w_t+(1+r_t)a_t-c_t-a_{t+1}\right].
  \]

  The first-order conditions imply
  \[
    \beta^t u'(c_t)=\lambda_t,
    \qquad t=0,1,\ldots,
  \]
  \[
    \lambda_t=(1+r_{t+1})\lambda_{t+1},
    \qquad t=0,1,\ldots.
  \]

  \begin{block}{Interpretation}
    $\lambda_t$ is the increase in lifetime utility from one additional unit of
    the consumption good available at date $t$.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 51--53.}
\end{frame}

\begin{frame}{Tangency Across Dates Gives the Euler Equation}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.45\textwidth}
      The household treats consumption at $t$ and $t+1$ as two different goods.
      At an interior optimum,
      \[
        \frac{\beta u'(c_{t+1})}{u'(c_t)}
        =
        \frac{1}{1+r_{t+1}}.
      \]

      \vspace{.45em}
      \begin{block}{Same Mathematics as Day 4}
        The MRS equals the relative price. The only new feature is that the
        goods carry date labels.
      \end{block}
    \end{column}
    \begin{column}{.53\textwidth}
      \centering
      \includegraphics[width=.98\linewidth]{intertemporal-choice.pdf}
    \end{column}
  \end{columns}
  \sources{AdvancedMacro Book, printed pp. 53 and 57--60; Lecture 3, slides 4--6.}
\end{frame}

\begin{frame}{Saving One More Unit: Marginal Cost Equals Marginal Benefit}
  Combine the two first-order conditions:
  \[
    \boxed{
      u'(c_t)
      =
      \beta(1+r_{t+1})u'(c_{t+1}).
    }
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Cost of Saving One More Unit}
        \begin{minipage}[t][1.75cm][t]{\dimexpr\linewidth-.76em\relax}
          One unit is forgone today, reducing lifetime utility by
          \[
            \beta^t u'(c_t).
          \]
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Benefit One Date Later}
        \begin{minipage}[t][1.75cm][t]{\dimexpr\linewidth-.76em\relax}
          Tomorrow it yields $1+r_{t+1}$ units. Lifetime utility rises by
          \[
            \beta^{t+1}(1+r_{t+1})u'(c_{t+1}).
          \]
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.25em}
  At an interior optimum, these two marginal effects are equal.
  \sources{AdvancedMacro Book, printed p. 53.}
\end{frame}

\begin{frame}{The Euler Equation Is Local; the TVC Disciplines the Tail}
  For the interior candidate path in this model, impose the model-specific
  transversality condition
  \[
    \boxed{
      \lim_{t\to\infty}\beta^t u'(c_t)a_{t+1}=0.
    }
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Euler Equation}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          Compares the marginal cost and benefit of moving one unit between two
          adjacent dates.
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Transversality Condition}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          Rules out tails that retain resources with positive shadow value.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.35em}
  Neither condition proves existence. Feasibility, concavity, and a separate
  existence argument remain necessary.
  \sources{AdvancedMacro Book, printed pp. 49--53, 75--78, and 86.}
\end{frame}

\begin{frame}{The Firm Transforms Capital and Labor into Output}
  In every date $t$, the firm rents capital $k_t$, hires labor $\ell_t$, and
  produces the single final good:
  \[
    y_t=A_t f(k_t,\ell_t)
    =
    A_t k_t^\alpha\ell_t^{1-\alpha},
    \qquad 0<\alpha<1.
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Technology}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          \raggedright
          $A_t$ is total factor productivity. It scales output at every input
          pair; $f$ remains the standard Cobb--Douglas function.
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Constant Returns}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          \raggedright
          Under constant returns, any finite profit-maximizing plan earns zero
          profit. Positive profit would scale without bound.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.35em}
  Throughout, $\delta=0$: capital is fully preserved between dates.
  \sources{AdvancedMacro Book, printed pp. 61--65.}
\end{frame}

\begin{frame}{The Firm's Problem Is Static Each Date}
  Normalize the final-good price to one. Given $w_t$ and $r_t$, the firm solves
  \[
    \max_{k_t,\ell_t\geq0}
    \left\{
      A_t f(k_t,\ell_t)-w_t\ell_t-r_t k_t
    \right\}.
  \]

  At a positive-input optimum,
  \[
    w_t=
    A_t\frac{\partial f(k_t,\ell_t)}{\partial\ell_t}
    =A_t f_2(k_t,\ell_t),
    \qquad
    r_t=
    A_t\frac{\partial f(k_t,\ell_t)}{\partial k_t}
    =A_t f_1(k_t,\ell_t).
  \]

  \begin{block}{Why the Firm Problem Is Static}
    This is the one-period problem from Day 4, repeated at every date. The firm
    rents inputs, produces, and sells output within date $t$; the household owns
    the capital and chooses how much to carry into the next date.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 65--67.}
\end{frame}

\begin{frame}{Capital Earns $r_t$ and Returns Intact}
  For Cobb--Douglas production,
  \[
    w_t=(1-\alpha)A_t k_t^\alpha\ell_t^{-\alpha},
    \qquad
    r_t=\alpha A_t k_t^{\alpha-1}\ell_t^{1-\alpha}.
  \]

  The firm pays $r_t$ for one unit of capital. Since capital is returned intact,
  the household receives the gross payoff $1+r_t$.

  Constant returns and the first-order conditions imply
  \[
    A_t f(k_t,\ell_t)=w_t\ell_t+r_t k_t,
  \]
  so competitive profits are zero.
  \sources{AdvancedMacro Book, printed pp. 66--68, specialized to no depreciation.}
\end{frame}

\begin{frame}{Physical Feasibility Contains No Prices}
  \begin{definitionbox}[Allocation and Feasibility]
    A physical allocation is a sequence
    \[
      z=\left(c_t,k_{t+1},\ell_t\right)_{t=0}^{\infty}.
    \]
  \end{definitionbox}

  It is \textbf{feasible} if, for every $t=0,1,\ldots$,
  \[
    c_t\geq0,
    \qquad
    k_{t+1}\geq0,
    \qquad
    0\leq\ell_t\leq1,
  \]
  and
  \[
    \boxed{
      c_t+k_{t+1}
      \leq
      A_t k_t^\alpha\ell_t^{1-\alpha}+k_t.
    }
  \]

  Feasibility is physical and contains no prices. Monotonicity makes the
  resource inequality bind at an optimum.
  \sources{Lecture 3, slides 20--22; AdvancedMacro Book, printed pp. 68--69.}
\end{frame}

\begin{frame}{Optimal Choice May Be Set-Valued}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Household Correspondence]
        \begin{minipage}[t][2.75cm][t]{\dimexpr\linewidth-.76em\relax}
          At prices $(w,r)$, define
          \[
            H(w,r)=\{\text{household-optimal paths}\}.
          \]
          It may contain more than one path $(c,a)$.
        \end{minipage}
      \end{definitionbox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Firm Correspondence]
        \begin{minipage}[t][2.75cm][t]{\dimexpr\linewidth-.76em\relax}
          At prices $(w_t,r_t)$, define
          \[
            F_t(w_t,r_t)=\{\text{optimal input pairs at }t\}.
          \]
          It may contain more than one pair $(k,\ell)$.
        \end{minipage}
      \end{definitionbox}
    \end{column}
  \end{columns}

  \vspace{.35em}
  Under constant returns,
  \[
    \text{positive profit}\Longrightarrow F_t(w_t,r_t)=\varnothing,
    \qquad
    \text{zero profit}\Longrightarrow F_t(w_t,r_t)\text{ may be set-valued}.
  \]
  Equilibrium selects mutually compatible elements of $H$ and $F_t$.
  \sources{Lecture 3, slides 24 and 26--30; Macro1 GE Theory, slides 16--24.}
\end{frame}

\begin{frame}{Define Equilibrium Before Deriving Its Equations}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[DEFINITION]
        \begin{minipage}[t][3.80cm][t]{\dimexpr\linewidth-.76em\relax}
          \begin{itemize}
            \item Names the objects that constitute an equilibrium.
            \item States optimization and market clearing.
            \item Does not require derivatives or first-order conditions.
          \end{itemize}
        \end{minipage}
      \end{definitionbox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{propositionbox}[CHARACTERIZATION]
        \begin{minipage}[t][3.80cm][t]{\dimexpr\linewidth-.76em\relax}
          \begin{itemize}
            \item Derives equations every equilibrium must satisfy.
            \item Uses first-order conditions or KKT under their hypotheses.
            \item Needs extra conditions for sufficiency, existence, or uniqueness.
          \end{itemize}
        \end{minipage}
      \end{propositionbox}
    \end{column}
  \end{columns}
  \sources{AdvancedMacro Book, printed pp. 70--74; Macro1 GE Theory,
  slides 16--17 and 24--25.}
\end{frame}

\begin{frame}{Equilibrium Requires Optimality and Market Clearing}
  \begin{definitionbox}[Competitive Equilibrium]
    Given $a_0$, a competitive equilibrium is a collection
    \[
      \left((c_t,a_{t+1}),(k_t,\ell_t),(w_t,r_t)\right)_{t=0}^{\infty}
    \]
    with $(w_t,r_t)\in\mathbb{R}_{++}^2$ for every $t$, such that:
    \begin{enumerate}
      \item $(c,a)\in H(w,r)$;
      \item $(k_t,\ell_t)\in F_t(w_t,r_t)$ for every $t$;
      \item markets clear at every date:
      \[
        a_t=k_t,
        \qquad
        \ell_t=1,
        \qquad
        c_t+k_{t+1}=A_t f(k_t,\ell_t)+k_t.
      \]
    \end{enumerate}
  \end{definitionbox}

  \sources{AdvancedMacro Book, Definition 14, printed p. 70, specialized to no
  depreciation; Macro1 GE Theory, slides 23--24; Lecture 3, slides 24--30.}
\end{frame}

\begin{frame}{Why Goods-Market Clearing Follows}
  Suppose the household and firm optimize and the asset and labor markets clear.

  \statementitem{1}{Substitute $a_t=k_t$ and $a_{t+1}=k_{t+1}$ into the
  household budget:
  \[
    c_t+k_{t+1}=w_t+(1+r_t)k_t.
  \]}
  \statementitem{2}{Subtract the capital carried into date $t$:
  \[
    c_t+k_{t+1}-k_t=w_t+r_t k_t.
  \]}
  \statementitem{3}{Since $\ell_t=1$ and constant returns imply zero profit,
  \[
    w_t\ell_t+r_t k_t=A_t f(k_t,\ell_t).
  \]}
  \statementitem{4}{Therefore the goods market clears:
  \[
    c_t+k_{t+1}=A_t f(k_t,\ell_t)+k_t.
  \]}
  \sources{AdvancedMacro Book, Theorem 15, printed pp. 71--72.}
\end{frame}

\begin{frame}{Firm Optimality Makes Prices Functions of Capital}
  To characterize equilibrium, set $A_t=1$ and use $\ell_t=1$.

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{General Technology}
        \begin{minipage}[t][2.05cm][t]{\dimexpr\linewidth-.76em\relax}
          \[
            w_t=f_\ell(k_t,1),
          \]
          \[
            r_t=f_k(k_t,1).
          \]
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Cobb--Douglas Technology}
        \begin{minipage}[t][2.05cm][t]{\dimexpr\linewidth-.76em\relax}
          \[
            w_t=(1-\alpha)k_t^\alpha,
          \]
          \[
            r_t=\alpha k_t^{\alpha-1}.
          \]
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.45em}
  Firm optimality and market clearing remove prices as independent unknowns.
  \sources{AdvancedMacro Book, printed pp. 72--73.}
\end{frame}

\begin{frame}{Feasibility Recovers Consumption from Capital}
  With $\ell_t=1$ and $A_t=1$, the resource constraint is
  \[
    c_t+k_{t+1}=k_t^\alpha+k_t.
  \]
  Hence,
  \[
    \boxed{
      c_t
      =
      k_t^\alpha+k_t-k_{t+1}.
    }
  \]

  \vspace{.3em}
  \centering
  \begin{tikzpicture}
    \node[object,text width=2.5cm] (current) at (0,0)
      {Current state\\$k_t$};
    \node[object,text width=2.8cm] (choice) at (3.75,0)
      {Next capital\\$k_{t+1}$};
    \node[object,fill=slideblue!8,text width=3.2cm] (cons) at (8.0,0)
      {Consumption $c_t$\\
      $k_t^\alpha+k_t-k_{t+1}$};
    \draw[route] (current) -- (choice);
    \draw[route] (choice) -- (cons);
  \end{tikzpicture}

  \vspace{.55em}
  Once the capital path is known, the allocation and prices can be recovered.
  \sources{AdvancedMacro Book, printed pp. 72--73.}
\end{frame}

\begin{frame}{Three Substitutions Produce the Capital Recurrence}
  \setlength{\abovedisplayskip}{.28em}
  \setlength{\belowdisplayskip}{.28em}
  \statementitem{1}{Firm optimality replaces the return:
  \[
    r_{t+1}=\alpha k_{t+1}^{\alpha-1}.
  \]}
  \statementitem{2}{Feasibility replaces consumption at both dates:
  \[
    c_j=k_j^\alpha+k_j-k_{j+1},
    \qquad j=t,t+1.
  \]}
  \statementitem{3}{Insert both expressions into the household Euler equation:
  \[
    \boxed{
      \begin{aligned}
        u'\!\left(k_t^\alpha+k_t-k_{t+1}\right)
        ={}&\beta\left[1+\alpha k_{t+1}^{\alpha-1}\right]\\
        &\times u'\!\left(k_{t+1}^\alpha+k_{t+1}-k_{t+2}\right).
      \end{aligned}
    }
  \]}

  The result is a second-order difference equation in
  $(k_t,k_{t+1},k_{t+2})$.

  \sources{AdvancedMacro Book, equation (4.40), printed pp. 72--73.}
\end{frame}

\begin{frame}{Solving the Euler System Does Not Prove Existence}
  \begin{block}{Necessary Direction}
    \[
      \text{interior equilibrium}
      \quad\Longrightarrow\quad
      \text{capital recurrence}.
    \]
  \end{block}

  \begin{block}{Converse Direction}
    \[
      \begin{gathered}
        \text{capital recurrence}+\text{feasibility}+\text{initial condition}\\
        {}+\text{transversality}+\text{sufficient optimality conditions}
      \end{gathered}
      \quad\Longrightarrow\quad
      \text{equilibrium}.
    \]
  \end{block}

  \vspace{.30em}
  \centering
  Neither implication proves that a solution exists or that it is unique.
  \sources{AdvancedMacro Book, printed pp. 72--75 and 86.}
\end{frame}

\begin{frame}{Efficiency Asks Whether Any Feasible Path Is Better}
  \begin{definitionbox}[Efficiency]
    A feasible allocation
    $z^*=\left(c_t^*,k_{t+1}^*,\ell_t^*\right)_{t=0}^{\infty}$ is efficient if
    there is no feasible $\widetilde z$ such that
    $U(\widetilde c)>U(c^*)$.
  \end{definitionbox}

  Efficiency uses only preferences and physical feasibility: no prices and no
  ownership rule. With one household, the planner selects the efficient path by
  maximizing the household's lifetime utility over feasible allocations.
  \sources{Lecture 3, slides 21--22; Macro1 GE Theory, slides 34--40.}
\end{frame}

\begin{frame}{The Planner Chooses the Best Feasible Path}
  With $A_t=1$, the planner solves
  \[
    \max_{\left(c_t,k_{t+1},\ell_t\right)_{t=0}^{\infty}}
    \sum_{t=0}^{\infty}\beta^t u(c_t)
  \]
  subject to
  \[
    c_t+k_{t+1}
    =
    k_t^\alpha\ell_t^{1-\alpha}+k_t,
    \qquad t=0,1,\ldots,
  \]
  \[
    c_t\geq0,
    \qquad
    k_{t+1}\geq0,
    \qquad
    0\leq\ell_t\leq1,
    \qquad
    k_0>0\ \text{given}.
  \]

  Since leisure does not enter utility and production increases in labor,
  $\ell_t=1$. At an interior optimum,
  \[
    \lim_{t\to\infty}\beta^t u'(c_t)k_{t+1}=0.
  \]
  \sources{AdvancedMacro Book, printed pp. 75--78, specialized to no
  depreciation; Macro1 GE Theory, slides 24--26.}
\end{frame}

\begin{frame}{The Planner Has the Same Euler Equation}
  For the reduced problem, define
  \[
    L=
    \sum_{t=0}^{\infty}\beta^t u(c_t)
    +
    \sum_{t=0}^{\infty}
    \lambda_t
    \left[
      k_t^\alpha+k_t-c_t-k_{t+1}
    \right].
  \]

  The first-order conditions give
  \[
    \lambda_t=\beta^t u'(c_t),
    \qquad t=0,1,\ldots,
  \]
  \[
    \lambda_t
    =
    \lambda_{t+1}
    \left[
      \alpha k_{t+1}^{\alpha-1}+1
    \right],
    \qquad t=0,1,\ldots.
  \]
  Therefore,
  \[
    \boxed{
      u'(c_t)
      =
      \beta
      \left[
        \alpha k_{t+1}^{\alpha-1}+1
      \right]
      u'(c_{t+1}).
    }
  \]
  \sources{AdvancedMacro Book, printed pp. 77--78.}
\end{frame}

\begin{frame}{The Market and Planner Choose the Same Allocation}
  Assume lifetime utility is well-defined, the relevant optima are interior,
  the problems are concave, and both transversality conditions hold.

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{theorembox}[Equilibrium $\Longrightarrow$ Planner]
        \begin{minipage}[t][2.85cm][t]{\dimexpr\linewidth-.76em\relax}
          Every equilibrium allocation is a planner optimum:
          \[
            \text{equilibrium}
            \Longrightarrow
            \text{planner optimum}.
          \]
          Markets attain an efficient allocation.
        \end{minipage}
      \end{theorembox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{theorembox}[Planner $\Longrightarrow$ Equilibrium]
        \begin{minipage}[t][2.85cm][t]{\dimexpr\linewidth-.76em\relax}
          Every interior planner optimum can be supported by prices:
          \[
            \text{planner optimum}
            \Longrightarrow
            \text{equilibrium}.
          \]
          Marginal products construct the prices.
        \end{minipage}
      \end{theorembox}
    \end{column}
  \end{columns}

  \vspace{.20em}
  For an infinite horizon, these hypotheses are part of the theorem; the Euler
  equations alone are not sufficient.
  \sources{AdvancedMacro Book, Theorems 18--19, printed pp. 78--79.}
\end{frame}

\begin{frame}{Infinite-Horizon Existence Requires a Separate Theorem}
  This is a problem over sequences, so the finite-dimensional Extreme Value
  Theorem does not apply directly. A rigorous existence proof must specify:
  \statementitem{1}{a topology on the path space;}
  \statementitem{2}{compactness or sequential compactness of the feasible set;}
  \statementitem{3}{continuity and uniform control of the discounted tail.}

  \begin{block}{Two Separate Conclusions}
    \[
      \begin{aligned}
        \text{planner solution exists}+\text{supporting-price theorem}
        &\Longrightarrow \text{equilibrium exists},\\[.35em]
        \begin{gathered}
          \text{planner solution is unique},\\[-.1em]
          \text{equilibrium}\Longrightarrow\text{planner solution}
        \end{gathered}
        &\Longrightarrow
        \begin{gathered}
          \text{at most one}\\[-.1em]
          \text{equilibrium allocation}.
        \end{gathered}
      \end{aligned}
    \]
  \end{block}
  \sources{AdvancedMacro Book, Appendix A, printed p. 86; Macro1 GE Theory,
  slides 3 and 24--26.}
\end{frame}
