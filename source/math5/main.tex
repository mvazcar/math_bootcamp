\begin{frame}{Every Date Contributes a Constraint and a Multiplier}
  Consider an infinite-path problem and its formal Lagrangean:
  \[
    \max_{z=(z_t)_{t=0}^{\infty}} F(z)
    \quad\text{s.t.}\quad
    g_t(z)=0, t=0,1,\ldots,
    \qquad
    L(z;\lambda)=F(z)+\sum_{t=0}^{\infty}\lambda_t g_t(z).
  \]

  \centering
  \begin{tikzpicture}
    \foreach \x/\lab/\mul in {0/$g_0(z)=0$/$\lambda_0$,3.15/$g_1(z)=0$/$\lambda_1$,
      6.30/$\cdots$/$\cdots$,9.45/$g_t(z)=0$/$\lambda_t$}{
      \node[object,text width=2.0cm] at (\x,0) {\lab};
      \node[text=slideblue] at (\x,-.80) {\mul};
    }
  \end{tikzpicture}

  \vspace{.10em}
  \begin{block}{Regularity and Interpretation}
    Assume an infinite-horizon multiplier theorem supplies $\lambda$.
    Under its regularity conditions, the multiplier sequence is unique. For
    $g_t(z)+b_t=0$ and differentiable $V$,
    $\lambda_t=\partial V(b)/\partial b_t$.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 51--53 and 77; Sundaram, Chapter 5.}
\end{frame}

\begin{frame}{A Linking Choice Creates a Recurrence}
  A variable carried from $t$ to $t+1$ appears in two neighboring constraints:
  \[
    g_t(\ldots,s_{t+1})=0,
    \qquad
    g_{t+1}(s_{t+1},\ldots)=0.
  \]

  \centering
  \begin{tikzpicture}
    \node[object,text width=2.35cm] (left) at (0,0)
      {Date-$t$ constraint\\$g_t=0$};
    \node[object,fill=slidegold!9,draw=slidegold,text width=2.2cm] (state) at (4.0,0)
      {Linking choice\\$s_{t+1}$};
    \node[object,text width=2.35cm] (right) at (8.0,0)
      {Date-$(t+1)$ constraint\\$g_{t+1}=0$};
    \draw[equality] (left) -- node[above,font=\small]{$\lambda_t$} (state);
    \draw[equality] (state) -- node[above,font=\small]{$\lambda_{t+1}$} (right);
  \end{tikzpicture}

  \vspace{.5em}
  Differentiating with respect to $s_{t+1}$ links adjacent multipliers.
  Eliminating them typically produces
  \[
    \Phi(s_t,s_{t+1},s_{t+2})=0.
  \]
  This is a second-order difference equation.
  \sources{AdvancedMacro Book, printed pp. 51--53, 72--73, and 77--78.}
\end{frame}

\begin{frame}{An Infinite Path Needs a Boundary at Infinity}
  A recurrence alone does not determine a path:
  \[
    \Phi(s_t,s_{t+1},s_{t+2})=0,
    \qquad t=0,1,\ldots.
  \]
  It needs an initial condition and a boundary condition at infinity:
  \[
    s_0\ \text{given},
    \qquad
    \lim_{t\to\infty}\lambda_t s_{t+1}=0.
  \]

  \begin{theorembox}[Infinite-Horizon Logic]
    \begin{itemize}
      \item Existence requires an argument about feasible sequences and tails.
      \item Strict concavity gives at most one solution.
      \item Regularity gives first-order conditions at every date.
      \item Concavity and the boundary condition verify global optimality.
    \end{itemize}
  \end{theorembox}
  \sources{AdvancedMacro Book, Appendix A, printed p. 86; Math Bootcamp Day 4.}
\end{frame}

\begin{frame}{Time for Economics}
  \begin{itemize}
    \item The vector $z$ becomes a path of consumption, assets, and capital.
    \item Each $g_t$ becomes a dated budget or resource constraint.
    \item The linking choice $s_{t+1}$ becomes saving or next-period capital.
    \item When the perturbed value is differentiable, $\lambda_t$ becomes the
    shadow value of resources at date $t$.
    \item The recurrence becomes an Euler equation.
  \end{itemize}

  \vspace{.65em}
  \begin{block}{The Same Mathematical Structure}
    Dynamic economics applies ordinary constrained optimization to choices that
    are linked across dates.
  \end{block}
  \sources{AdvancedMacro Book, Chapters 4--5; Math Bootcamp Day 4.}
\end{frame}

\begin{frame}{Infinite Horizon Adds a Boundary at Infinity}
  The model uses dates $t=0,1,\ldots$ and assumes:
  \begin{itemize}
    \item a deterministic closed economy with no government;
    \item one representative price-taking household and firm;
    \item $u$ is strictly increasing, strictly concave, and twice
    differentiable; $f$ is increasing, concave, and constant returns to scale;
    \item $k_0=a_0>0$ and $\delta=0$.
  \end{itemize}

  \begin{block}{Transversality Condition}
    \[
      \lim_{t\to\infty}\beta^t u'(c_t)a_{t+1}=0.
    \]
    The Euler equation is local; this condition controls the tail.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 49--53, 75--78, and 86.}
\end{frame}

\begin{frame}{The Household Ranks Entire Consumption Paths}
  \begin{definitionbox}[Lifetime Utility]
    For $c=(c_0,c_1,c_2,\ldots)$,
    \[
      U(c)=\sum_{t=0}^{\infty}\beta^t u(c_t),
      \qquad 0<\beta<1.
    \]
    The discount factor $\beta$ gives the relative weight on later utility.
  \end{definitionbox}

  \vspace{.2em}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \textbf{Monotonicity}
      \[
        u'(c_t)>0
      \]
      More consumption is preferred.
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \textbf{Strict concavity}
      \[
        u''(c_t)<0
      \]
      Marginal utility falls with consumption.
    \end{column}
  \end{columns}
  \sources{AdvancedMacro Book, printed pp. 41--49.}
\end{frame}

\begin{frame}{The Household Chooses an Infinite Path at Given Prices}
  Given the price paths $\left(w_t,r_t\right)_{t=0}^{\infty}$ and initial assets
  $a_0$, the household solves
  \[
    \max_{\left(c_t,a_{t+1}\right)_{t=0}^{\infty}}
    \sum_{t=0}^{\infty}\beta^t u(c_t)
  \]
  subject to
  \[
    c_t+a_{t+1}=w_t+(1+r_t)a_t,
    \qquad c_t\geq0,
    \qquad a_{t+1}\geq0,
    \qquad t=0,1,\ldots,
  \]
  \[
    a_0\ \text{given}.
  \]

  \begin{block}{Admissibility and Price Taking}
    Nonnegative assets rule out Ponzi finance. The household chooses quantities
    while treating every wage and interest rate as given.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 51--52.}
\end{frame}

\begin{frame}{One Shadow Value Is Needed for Every Date}
  At an interior path, write the formal Lagrangean
  \[
    L=
    \sum_{t=0}^{\infty}\beta^t u(c_t)
    +
    \sum_{t=0}^{\infty}
    \lambda_t\left[w_t+(1+r_t)a_t-c_t-a_{t+1}\right].
  \]

  The first-order conditions imply
  \[
    \beta^t u'(c_t)=\lambda_t,
    \qquad t=0,1,\ldots,
  \]
  \[
    \lambda_t=(1+r_{t+1})\lambda_{t+1},
    \qquad t=0,1,\ldots.
  \]

  \begin{block}{Interpretation}
    $\lambda_t$ is the increase in lifetime utility from one additional unit of
    the consumption good available at date $t$.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 51--53.}
\end{frame}

\begin{frame}{The Euler Equation Balances Sacrifice and Return}
  Combine the two first-order conditions:
  \[
    \boxed{
      u'(c_t)
      =
      \beta(1+r_{t+1})u'(c_{t+1}).
    }
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Cost of Saving One More Unit}
        \begin{minipage}[t][1.75cm][t]{\dimexpr\linewidth-.76em\relax}
          One unit is forgone today, reducing lifetime utility by
          \[
            \beta^t u'(c_t).
          \]
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Benefit One Date Later}
        \begin{minipage}[t][1.75cm][t]{\dimexpr\linewidth-.76em\relax}
          Tomorrow it yields $1+r_{t+1}$ units. Lifetime utility rises by
          \[
            \beta^{t+1}(1+r_{t+1})u'(c_{t+1}).
          \]
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.25em}
  At an interior optimum, these two marginal effects are equal.
  \sources{AdvancedMacro Book, printed p. 53.}
\end{frame}

\begin{frame}{The Euler Equation Is Ordinary Tangency Across Dates}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.45\textwidth}
      The household treats consumption at $t$ and $t+1$ as two different goods.
      At an interior optimum,
      \[
        \frac{\beta u'(c_{t+1})}{u'(c_t)}
        =
        \frac{1}{1+r_{t+1}}.
      \]

      \vspace{.45em}
      \begin{block}{Same Mathematics as Day 4}
        The MRS equals the relative price. The only new feature is that the
        goods carry date labels.
      \end{block}
    \end{column}

      \includegraphics[width=.98\linewidth]{intertemporal-choice.pdf}
    \end{column}
  \end{columns}
  \sources{AdvancedMacro Book, printed pp. 53 and 57--60; Lecture 3, slides 4--6.}
\end{frame}

\begin{frame}{The Firm Transforms Capital and Labor into Output}
  In every date $t$, the firm rents capital $k_t$, hires labor $\ell_t$, and
  produces the single final good:
  \[
    y_t=A_t f(k_t,\ell_t)
    =
    A_t k_t^\alpha\ell_t^{1-\alpha},
    \qquad 0<\alpha<1.
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Technology}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          $A_t$ is total factor productivity. It shifts output without changing
          the capital--labor ratio inside $f$.
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Constant Returns}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          Scaling both inputs by one factor scales output by that factor. Price
          taking then implies zero profits.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.35em}
  Throughout, $\delta=0$: capital is fully preserved between dates.
  \sources{AdvancedMacro Book, printed pp. 61--65.}
\end{frame}

\begin{frame}{The Firm Solves a Static Problem at Every Date}
  Given the wage $w_t$ and rental rate $r_t$, the firm solves
  \[
    \max_{k_t,\ell_t\geq0}
    \left\{
      A_t f(k_t,\ell_t)-w_t\ell_t-r_t k_t
    \right\}.
  \]

  At a positive-input optimum,
  \[
    w_t=
    A_t\frac{\partial f(k_t,\ell_t)}{\partial\ell_t}
    =A_t f_2(k_t,\ell_t),
    \qquad
    r_t=
    A_t\frac{\partial f(k_t,\ell_t)}{\partial k_t}
    =A_t f_1(k_t,\ell_t).
  \]

  \begin{block}{Why the Firm Problem Is Static}
    This is the one-period problem from Day 4, repeated at every date. The firm
    rents inputs, produces, and sells output within date $t$; the household owns
    the capital and chooses how much to carry into the next date.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 65--67.}
\end{frame}

\begin{frame}{The Same Rental Rate Links Firm and Household}
  For Cobb--Douglas production,
  \[
    w_t=(1-\alpha)A_t k_t^\alpha\ell_t^{-\alpha},
    \qquad
    r_t=\alpha A_t k_t^{\alpha-1}\ell_t^{1-\alpha}.
  \]

  The firm pays $r_t$ for one unit of capital. Since capital is returned intact,
  the household receives the gross payoff $1+r_t$.

  Constant returns and the first-order conditions imply
  \[
    A_t f(k_t,\ell_t)=w_t\ell_t+r_t k_t,
  \]
  so competitive profits are zero.
  \sources{AdvancedMacro Book, printed pp. 66--68, specialized to no depreciation.}
\end{frame}

\begin{frame}{DEFINITION: Allocation and Feasibility}
  \begin{definitionbox}[Allocation]
    A physical allocation is a sequence
    \[
      z=\left(c_t,k_{t+1},\ell_t\right)_{t=0}^{\infty}.
    \]
  \end{definitionbox}

  It is \textbf{feasible} if, for every $t=0,1,\ldots$,
  \[
    c_t\geq0,
    \qquad
    k_{t+1}\geq0,
    \qquad
    0\leq\ell_t\leq1,
  \]
  and
  \[
    \boxed{
      c_t+k_{t+1}
      \leq
      A_t k_t^\alpha\ell_t^{1-\alpha}+k_t.
    }
  \]

  Feasibility is physical and contains no prices. Monotonicity makes the
  resource inequality bind at an optimum.
  \sources{Lecture 3, slides 20--22; AdvancedMacro Book, printed pp. 68--69.}
\end{frame}

\begin{frame}{Optimal Choice May Be Set-Valued}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}

        \begin{minipage}[t][2.75cm][t]{\dimexpr\linewidth-.76em\relax}
          At prices $(w,r)$, define
          \[
            H(w,r)=\{\text{household-optimal paths}\}.
          \]
          It may contain more than one path $(c,a)$.
        \end{minipage}
      \end{definitionbox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Firm Correspondence]
        \begin{minipage}[t][2.75cm][t]{\dimexpr\linewidth-.76em\relax}
          At prices $(w_t,r_t)$, define
          \[
            F_t(w_t,r_t)=\{\text{optimal input pairs at }t\}.
          \]
          It may contain more than one pair $(k,\ell)$.
        \end{minipage}
      \end{definitionbox}
    \end{column}
  \end{columns}

  \vspace{.35em}
  Under constant returns, $F_t$ may contain many input choices. Equilibrium
  selects elements of both correspondences that are mutually compatible.
  \sources{Lecture 3, slides 24 and 26--30; Macro1 GE Theory, slides 16--24.}
\end{frame}

\begin{frame}{Definition and Characterization Answer Different Questions}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[DEFINITION]
        \begin{minipage}[t][4.55cm][t]{\dimexpr\linewidth-.76em\relax}
          \begin{itemize}
            \item Names the objects that constitute an equilibrium.
            \item States optimization and market clearing.
            \item Tests whether a proposed price-allocation path qualifies.
            \item Does not require derivatives or first-order conditions.
          \end{itemize}
        \end{minipage}
      \end{definitionbox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{propositionbox}[CHARACTERIZATION]
        \begin{minipage}[t][4.55cm][t]{\dimexpr\linewidth-.76em\relax}
          \begin{itemize}
            \item Derives equations every equilibrium must satisfy.
            \item Uses first-order conditions or KKT under their hypotheses.
            \item Reduces the objects that must be solved for.
            \item Needs extra conditions for sufficiency, existence, or uniqueness.
          \end{itemize}
        \end{minipage}
      \end{propositionbox}
    \end{column}
  \end{columns}
  \sources{AdvancedMacro Book, printed pp. 70--74; Macro1 GE Theory,
  slides 16--17 and 24--25.}
\end{frame}

\begin{frame}{DEFINITION: Competitive Equilibrium}
  \begin{definitionbox}[Competitive Equilibrium]
    Given $a_0$, a competitive equilibrium is a collection
    \[
      \left((c_t,a_{t+1}),(k_t,\ell_t),(w_t,r_t)\right)_{t=0}^{\infty}
    \]
    such that:
    \begin{enumerate}
      \item $(c,a)\in H(w,r)$;
      \item $(k_t,\ell_t)\in F_t(w_t,r_t)$ for every $t$;
      \item markets clear at every date:
      \[
        a_t=k_t,
        \qquad
        \ell_t=1,
        \qquad
        c_t+k_{t+1}=A_t f(k_t,\ell_t)+k_t.
      \]
    \end{enumerate}
  \end{definitionbox}

  \sources{AdvancedMacro Book, Definition 14, printed p. 70, specialized to no
  depreciation; Macro1 GE Theory, slides 23--24; Lecture 3, slides 24--30.}
\end{frame}

\begin{frame}{Walras' Law Makes One Clearing Condition Redundant}
  Suppose the household and firm optimize and the asset and labor markets clear.

  \statementitem{1}{Substitute $a_t=k_t$ and $a_{t+1}=k_{t+1}$ into the
  household budget:
  \[
    c_t+k_{t+1}=w_t+(1+r_t)k_t.
  \]}
  \statementitem{2}{Subtract the capital carried into date $t$:
  \[
    c_t+k_{t+1}-k_t=w_t+r_t k_t.
  \]}
  \statementitem{3}{Since $\ell_t=1$ and constant returns imply zero profit,
  \[
    w_t\ell_t+r_t k_t=A_t f(k_t,\ell_t).
  \]}
  \statementitem{4}{Therefore the goods market clears:
  \[
    c_t+k_{t+1}=A_t f(k_t,\ell_t)+k_t.
  \]}
  \sources{AdvancedMacro Book, Theorem 15, printed pp. 71--72.}
\end{frame}

\begin{frame}{CHARACTERIZATION: Prices Become Functions of Capital}
  To characterize equilibrium, set $A_t=1$ and use $\ell_t=1$.

  \begin{columns}[t,onlytextwidth]

      \begin{block}{General Technology}
        \begin{minipage}[t][2.05cm][t]{\dimexpr\linewidth-.76em\relax}

            w_t=f_\ell(k_t,1),
          \]
          \[
            r_t=f_k(k_t,1).

        \end{minipage}

\begin{frame}{CHARACTERIZATION: Feasibility Eliminates Consumption}

  \]
  Hence,
  \[

  Once the capital path is known, the allocation and prices can be recovered.
  \sources{AdvancedMacro Book, printed pp. 72--73.}
\end{frame}

\begin{frame}{CHARACTERIZATION: A Difference Equation for Capital}
  For $t=0,1,\ldots$, substitute equilibrium prices and consumption into
  the household Euler equation:
  \[
    \begin{aligned}
      &u'\!\left(
        k_t^\alpha+k_t-k_{t+1}
      \right)\\
      &\quad=
      \beta\left[
        1+\alpha k_{t+1}^{\alpha-1}
      \right]
      u'\!\left(
        k_{t+1}^\alpha+k_{t+1}-k_{t+2}
      \right).
    \end{aligned}
  \]

  This is a second-order difference equation in
  $(k_t,k_{t+1},k_{t+2})$. Combine it with
  \[
    k_0=a_0,
    \qquad
    c_t\geq0,\ k_t\geq0,
    \qquad
    \lim_{t\to\infty}\beta^t u'(c_t)k_{t+1}=0.
  \]

  \sources{AdvancedMacro Book, equation (4.40), printed pp. 72--73.}
\end{frame}

\begin{frame}{A Characterization Is Not an Existence Theorem}
  The capital equation is useful because it removes prices and consumption from the
  search. But the logical claims remain separate:

  \begin{itemize}
    \item Every interior competitive equilibrium satisfies the equation.
    \item The converse also requires all feasibility and boundary conditions,
    together with conditions making the first-order conditions sufficient.
    \item Writing the equation does not prove that a solution exists.
    \item It also does not prove that the solution is unique.
  \end{itemize}

  \vspace{.45em}
  \begin{block}{The Next Step}
    The social planner problem supplies a simpler route to existence,
    allocation uniqueness, and the welfare interpretation.
  \end{block}
  \sources{AdvancedMacro Book, printed pp. 72--75 and 86.}
\end{frame}

\begin{frame}{DEFINITION: Efficiency Uses Only Feasible Allocations}
  \begin{definitionbox}[Efficiency]
    A feasible allocation
    $z^*=\left(c_t^*,k_{t+1}^*,\ell_t^*\right)_{t=0}^{\infty}$ is efficient if
    there is no feasible $\widetilde z$ such that
    $U(\widetilde c)>U(c^*)$.
  \end{definitionbox}

  Efficiency uses only preferences and physical feasibility: no prices and no
  ownership rule. With one household, the planner selects the efficient path by
  maximizing the household's lifetime utility over feasible allocations.
  \sources{Lecture 3, slides 21--22; Macro1 GE Theory, slides 34--40.}
\end{frame}

\begin{frame}{DEFINITION: The Social Planner Problem}
  With $A_t=1$, the planner solves
  \[
    \max_{\left(c_t,k_{t+1},\ell_t\right)_{t=0}^{\infty}}
    \sum_{t=0}^{\infty}\beta^t u(c_t)
  \]
  subject to
  \[
    c_t+k_{t+1}
    =
    k_t^\alpha\ell_t^{1-\alpha}+k_t,
    \qquad t=0,1,\ldots,
  \]
  \[
    c_t\geq0,
    \qquad
    k_{t+1}\geq0,
    \qquad
    0\leq\ell_t\leq1,
    \qquad
    k_0>0\ \text{given}.
  \]

  Since leisure does not enter utility and production increases in labor,
  $\ell_t=1$. At an interior optimum,
  \[
    \lim_{t\to\infty}\beta^t u'(c_t)k_{t+1}=0.
  \]
  \sources{AdvancedMacro Book, printed pp. 75--78, specialized to no
  depreciation; Macro1 GE Theory, slides 24--26.}
\end{frame}

\begin{frame}{CHARACTERIZATION: The Planner Euler Equation}
  For the reduced problem, define
  \[
    L=
    \sum_{t=0}^{\infty}\beta^t u(c_t)
    +
    \sum_{t=0}^{\infty}
    \lambda_t
    \left[
      k_t^\alpha+k_t-c_t-k_{t+1}
    \right].
  \]

  The first-order conditions give
  \[
    \lambda_t=\beta^t u'(c_t),
    \qquad t=0,1,\ldots,
  \]
  \[
    \lambda_t
    =
    \lambda_{t+1}
    \left[
      \alpha k_{t+1}^{\alpha-1}+1
    \right],
    \qquad t=0,1,\ldots.
  \]
  Therefore,
  \[
    \boxed{
      u'(c_t)
      =
      \beta
      \left[
        \alpha k_{t+1}^{\alpha-1}+1
      \right]
      u'(c_{t+1}).
    }
  \]
  \sources{AdvancedMacro Book, printed pp. 77--78.}
\end{frame}

\begin{frame}{The Welfare Theorems Connect Two Definitions}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{theorembox}[First Welfare Theorem]
        \begin{minipage}[t][3.25cm][t]{\dimexpr\linewidth-.76em\relax}
          Every equilibrium allocation is a planner optimum:
          \[
            \text{equilibrium}
            \Longrightarrow
            \text{planner optimum}.
          \]
          Markets attain an efficient allocation.
        \end{minipage}
      \end{theorembox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{theorembox}[Second Welfare Theorem]
        \begin{minipage}[t][3.25cm][t]{\dimexpr\linewidth-.76em\relax}
          In this model, every planner optimum can be supported by prices:
          \[
            \text{planner optimum}
            \Longrightarrow
            \text{equilibrium}.
          \]
          Marginal products construct the prices.
        \end{minipage}
      \end{theorembox}
    \end{column}
  \end{columns}

  \vspace{.35em}
  Under the maintained infinite-horizon assumptions, verification uses
  feasibility, identical Euler equations, transversality, and concavity.
  \sources{AdvancedMacro Book, Theorems 18--19, printed pp. 78--79.}
\end{frame}

\begin{frame}{Supporting Prices Decentralize the Planner Allocation}
  Start from a planner solution $(c_t^*,k_{t+1}^*)_{t=0}^{\infty}$ and define
  \[
    w_t^*=(1-\alpha)(k_t^*)^\alpha,
    \qquad
    r_t^*=\alpha(k_t^*)^{\alpha-1}.
  \]

  \statementitem{1}{Set $a_t^*=k_t^*$ and $\ell_t^*=1$, so the asset and labor
  markets clear.}
  \statementitem{2}{The supporting prices satisfy the firm's first-order
  conditions; concavity makes them sufficient.}
  \statementitem{3}{The planner Euler equation becomes the household Euler
  equation; concavity and transversality make it sufficient.}
  \statementitem{4}{Planner feasibility is exactly goods-market clearing.}

  \sources{AdvancedMacro Book, proof of Theorem 19, printed pp. 78--79.}
\end{frame}

\begin{frame}{Infinite Horizon Needs a Separate Existence Argument}
  For every $t=0,1,\ldots$, feasibility can be written as
  \[
    c_t+k_{t+1}
    \leq
    k_t^\alpha+k_t.
  \]
  This is a problem over sequences, so finite-dimensional compactness is not
  enough. Under bounded feasibility, continuity, discounting, and the maintained
  tail conditions, a limit of truncated planner problems yields a solution.

  \begin{block}{Two Separate Conclusions}
    \[
      \begin{aligned}
        \text{planner existence}+\text{supporting prices}
        &\Longrightarrow \text{equilibrium existence},\\[.35em]
        \text{strict concavity}+\text{First Welfare Theorem}
        &\Longrightarrow \text{allocation uniqueness}.
      \end{aligned}
    \]
  \end{block}
  \sources{AdvancedMacro Book, Appendix A, printed p. 86; Macro1 GE Theory,
  slides 3 and 24--26.}
\end{frame}

\begin{frame}{Definition Comes First; Characterization Makes It Usable}
  \begin{enumerate}
    \item List the primitives and every endogenous sequence.
    \item Write household and firm optimization at given prices.
    \item Define equilibrium by adding market clearing at every date.
    \item Derive the characterization and keep all boundary conditions.
    \item Use the planner problem and welfare theorems to recover allocations
    and supporting prices.
  \end{enumerate}

  \vspace{.45em}
  \begin{block}{The Full Logic}
    \centering
    \textbf{Definition} identifies the object.
    \qquad
    \textbf{Characterization} reduces the search.
    \qquad
    \textbf{Theorems} justify the connection.
  \end{block}

  \vspace{.25em}
  \centering
  \textcolor{slideblue}{\bfseries General equilibrium is optimization made
  mutually consistent.}
  \sources{AdvancedMacro Book, Chapters 4--5; Lecture 3; Macro1 GE Theory.}
\end{frame}
