\begin{frame}{From a Point to a Path}
  Day 4 studied a scalar choice
  \[
    x\in D\subseteq\mathbb{R}.
  \]
  Today the choice is an infinite sequence:
  \[
    z=(x_0,x_1,x_2,\ldots).
  \]

    \vspace{.55em}
  In the model that follows, a candidate path satisfies one Euler equation
  at every date and one transversality condition at infinity.
\end{frame}

\begin{frame}{Dated Commodities}
  A commodity is identified not only by its physical features, but also by its
  date and state. In this deterministic model, date is the key label.

    \vspace{.50em}
  \begin{block}{Different Date, Different Good}
    Consumption at $t$ and $t+1$ are distinct commodities. Saving exchanges
    the current commodity for a future one; the interest rate will determine
    their relative price.
  \end{block}
\end{frame}

\begin{frame}{Two Market Structures}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Arrow--Debreu}
        \begin{itemize}
          \item Markets open only at date $0$.
          \item Every dated good is traded there, at prices $\{p_t\}$.
          \item A single lifetime budget constraint.
          \item Perfect commitment: contracts signed at $0$ are honored.
        \end{itemize}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Sequential Markets}
        \begin{itemize}
          \item Markets reopen at every date.
          \item Spot goods and one-period assets, at prices $\{w_t,r_t\}$.
          \item One budget constraint per date.
          \item Debt must be bounded by a no-Ponzi condition.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.4em}
  \centering Both assume perfect information and perfect competition, and both
  deliver the same consumption allocations. We work with sequential markets.
\end{frame}

\begin{frame}{The Environment}
  \begin{definitionbox}[Dynamic Production Economy]
    The environment is
    \[
      E=
      \left(
        \mathbb{N}_0,\beta,u,f,(A_t)_{t=0}^{\infty},k_0
      \right).
    \]
    These primitives are fixed before prices and allocations are determined.
    They define the \textbf{neoclassical growth model}, our canonical example
    for the rest of the day.
  \end{definitionbox}

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Primitives}
        \begin{minipage}[t][1.95cm][t]{\dimexpr\linewidth-.76em\relax}
          \begin{itemize}
            \item Dates $t=0,1,\ldots$
            \item Preferences: $\beta$ and $u$
            \item Technology: $f$ and $A_t$
          \end{itemize}
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Initial Resources}
        \begin{minipage}[t][1.95cm][t]{\dimexpr\linewidth-.76em\relax}
          \begin{itemize}
            \item The household owns $a_0=k_0>0$.
            \item Labor supply is one unit each date.
            \item Capital depreciates at rate $\delta\in(0,1]$.
          \end{itemize}
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Lifetime Utility}
  \begin{definitionbox}[Lifetime Utility]
    For $c=(c_0,c_1,c_2,\ldots)$,
    \[
      U(c)=\sum_{t=0}^{\infty}\beta^t u(c_t),
      \qquad 0<\beta<1.
    \]
    The discount factor $\beta$ gives the relative weight on later utility.
    We assume that this sum is well-defined and finite on admissible paths.
  \end{definitionbox}

  \vspace{.2em}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \textbf{Monotonicity}
      \[
        u'(c_t)>0
      \]
      More consumption is preferred.
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \textbf{Strict concavity}
      \[
        u''(c_t)<0
      \]
      Marginal utility falls with consumption.
    \end{column}
  \end{columns}
\end{frame}

    \[
      c_t+a_{t+1}=w_t+(1+r_t)a_t.

\begin{frame}{The Household Problem}
  \begin{definitionbox}[Given $\{w_t,r_t\}_{t=0}^{\infty}$ and $a_0$]
    \[
      \max_{\{c_t,a_{t+1}\}_{t=0}^{\infty}}\ \sum_{t=0}^{\infty}\beta^t u(c_t)
    \]
    subject to, for every $t$,
    \[
      c_t+a_{t+1}=w_t+(1+r_t)a_t,
      \qquad
      c_t\geq0 .
    \]
  \end{definitionbox}

  \begin{block}{State and Control}
    The \textbf{state} $a_t$ is predetermined at date $t$; the \textbf{control}
    $c_t$ is chosen there. The budget constraint is the law of motion
    $a_{t+1}=w_t+(1+r_t)a_t-c_t$.
  \end{block}
\end{frame}

\begin{frame}{Boundary Conditions}
  \begin{definitionbox}[An Infinite Path Needs Both Ends]
    \[
      a_0\ \text{given},
      \qquad
      \lim_{t\to\infty}\frac{a_{t+1}}{\prod_{i=1}^{t}(1+r_i)}=0 .
    \]
  \end{definitionbox}

  \begin{block}{Where the Terminal Condition Comes From}
    \textbf{No Ponzi}, imposed on the problem:
    $\lim a_{t+1}/\prod(1+r_i)\geq0$. Without it the household rolls debt over
    forever and no maximum exists.\par\smallskip
    \textbf{Transversality}, implied by optimality:
    $\lim a_{t+1}/\prod(1+r_i)\leq0$. Leaving value unspent is never optimal.
  \end{block}
\end{frame}

\begin{frame}{Shadow Values}
  There is one budget constraint at each date, so the Lagrangean carries one
  multiplier at each date:
  \[
    L=\sum_{t=0}^{\infty}\beta^t u(c_t)
     +\sum_{t=0}^{\infty}\lambda_t
      \bigl[\,\underbrace{w_t+(1+r_t)a_t}_{\text{resources at }t}
      -c_t-a_{t+1}\,\bigr].
  \]

  \begin{block}{What $\lambda_t$ Measures}
    $\lambda_t$ is the shadow value of wealth at date $t$: the gain in lifetime
    utility from one more unit of the good available at $t$. Differentiating in
    $c_t$ gives
    \[
      \beta^t u'(c_t)=\lambda_t .
    \]
  \end{block}
\end{frame}

\begin{frame}{The State Across Two Dates}
  The state $a_{t+1}$ is chosen at $t$ and carried into $t+1$, so it appears in
  two constraints:
  \[
    \underbrace{c_t+a_{t+1}=w_t+(1+r_t)a_t}_{\text{date }t,\ \text{multiplier }\lambda_t},
    \qquad
    \underbrace{c_{t+1}+a_{t+2}=w_{t+1}+(1+r_{t+1})a_{t+1}}_{\text{date }t+1,\ \text{multiplier }\lambda_{t+1}} .
  \]

  Setting $\partial L/\partial a_{t+1}=0$ therefore couples the two shadow
  values:
  \[
    \lambda_t=(1+r_{t+1})\,\lambda_{t+1}.
  \]

  \begin{block}{Why This Is the Euler Equation}
    Saving one unit costs $\lambda_t$ today and returns $(1+r_{t+1})\lambda_{t+1}$
    tomorrow. Substituting $\lambda_t=\beta^tu'(c_t)$ eliminates the multipliers.
  \end{block}
\end{frame}

\begin{frame}{Tangency Across Dates}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.45\textwidth}
      The household treats consumption at $t$ and $t+1$ as two different goods.
      At an interior optimum,
      \[
        \frac{\beta u'(c_{t+1})}{u'(c_t)}
        =
        \frac{1}{1+r_{t+1}}.
      \]

      \vspace{.45em}
      \begin{block}{Same Mathematics as Day 4}
        The MRS equals the relative price. The only new feature is that the
        goods carry date labels.
      \end{block}
    \end{column}

    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Saving One More Unit: Marginal Cost Equals Marginal Benefit}
  Combine the two first-order conditions:
  \[
    \boxed{
      u'(c_t)
      =
      \beta(1+r_{t+1})u'(c_{t+1}).
    }
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Cost of Saving One More Unit}
        \begin{minipage}[t][1.75cm][t]{\dimexpr\linewidth-.76em\relax}
          One unit is forgone today, reducing lifetime utility by
          \[
            \beta^t u'(c_t).
          \]
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Benefit One Date Later}
        \begin{minipage}[t][1.75cm][t]{\dimexpr\linewidth-.76em\relax}
          Tomorrow it yields $1+r_{t+1}$ units. Lifetime utility rises by
          \[
            \beta^{t+1}(1+r_{t+1})u'(c_{t+1}).
          \]
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.25em}
  At an interior optimum, these two marginal effects are equal.
\end{frame}

\begin{frame}{The Euler Equation and the Tail}
  For the interior candidate path in this model, impose the model-specific
  transversality condition
  \[
    \boxed{
      \lim_{t\to\infty}\beta^t u'(c_t)a_{t+1}=0.
    }
  \]

  Same condition as on \emph{Boundary Conditions}: $\lambda_t=\beta^tu'(c_t)$ and
  $\lambda_t=\lambda_0/\prod_{i=1}^{t}(1+r_i)$, so the two limits differ only by the
  constant $\lambda_0$.

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Euler Equation}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          Compares the marginal cost and benefit of moving one unit between two
          adjacent dates.
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Transversality Condition}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          Rules out tails that retain resources with positive shadow value.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.35em}
  Neither condition proves existence. Feasibility, concavity, and a separate
  existence argument remain necessary.
\end{frame}

\begin{frame}{Technology}
  In every date $t$, the firm rents capital $k_t$, hires labor $n_t$, and
  produces the single final good:
  \[
    y_t=A_t f(k_t,n_t)
    =
    A_t k_t^\alpha n_t^{1-\alpha},
    \qquad 0<\alpha<1.
  \]

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Technology}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          \raggedright
          $A_t$ is total factor productivity. It scales output at every input
          pair; $f$ remains the standard Cobb--Douglas function.
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Constant Returns}
        \begin{minipage}[t][1.65cm][t]{\dimexpr\linewidth-.76em\relax}
          \raggedright
          Under constant returns, any finite profit-maximizing plan earns zero
          profit. Positive profit would scale without bound.
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.35em}
\end{frame}

\begin{frame}{The Firm Problem}
  \begin{definitionbox}[Given $w_t$ and $r_t$, at each date separately]
    \[
      \max_{k_t,n_t\geq0}\
      \bigl\{k_t^\alpha(A_tn_t)^{1-\alpha}-w_tn_t-(r_t+\delta)k_t\bigr\}.
    \]
  \end{definitionbox}

  \begin{block}{Static, and Prices Equal Marginal Products}
    There is no intertemporal link: the firm rents, produces and sells within
    date $t$. At an interior optimum,
    \[
      w_t=A_tf_2(k_t,n_t),
      \qquad
      r_t+\delta=A_tf_1(k_t,n_t).
    \]
  \end{block}
\end{frame}

\begin{frame}{Factor Prices and the Return to Capital}
  For Cobb--Douglas production,
  \[
    w_t=(1-\alpha)A_t k_t^\alpha n_t^{-\alpha},
    \qquad
    r_t+\delta=\alpha A_t k_t^{\alpha-1}n_t^{1-\alpha}.
  \]

  A fraction $\delta$ of the machine wears out in production, so the rental must cover
  depreciation on top of the net return. Save one unit today, receive $1+r_t$
  tomorrow.

  Constant returns and the first-order conditions imply
  \[
    A_t f(k_t,n_t)=w_tn_t+(r_t+\delta)k_t,
  \]
  so competitive profits are zero.
\end{frame}

\begin{frame}{Feasible Allocations}
  \begin{definitionbox}[Allocation and Feasibility]
    A physical allocation is a sequence
    \[
      z=\left(c_t,k_{t+1},n_t\right)_{t=0}^{\infty}.
    \]
  \end{definitionbox}

  It is \textbf{feasible} if, for every $t=0,1,\ldots$,
  \[
    c_t\geq0,
    \qquad
    k_{t+1}\geq0,
    \qquad
    0\leq n_t\leq1,
  \]
  and
  \[
    \boxed{
      c_t+k_{t+1}
      \leq
      A_t k_t^\alpha n_t^{1-\alpha}+(1-\delta)k_t.
    }
  \]

  Feasibility is physical and contains no prices. Monotonicity makes the
  resource inequality bind at an optimum.
\end{frame}

\begin{frame}{Correspondences}
  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Household Correspondence]
        \begin{minipage}[t][2.75cm][t]{\dimexpr\linewidth-.76em\relax}
          At prices $(w,r)$, write
          \[
            H(w,r)=\{\text{household-optimal paths}\}.
          \]
          It may contain more than one path $(c,a)$.
        \end{minipage}
      \end{definitionbox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{definitionbox}[Firm Correspondence]
        \begin{minipage}[t][2.75cm][t]{\dimexpr\linewidth-.76em\relax}
          At prices $(w_t,r_t)$, write
          \[
            F_t(w_t,r_t)=\{\text{optimal input pairs at }t\}.
          \]
          It may contain more than one pair $(k,n)$.
        \end{minipage}
      \end{definitionbox}
    \end{column}
  \end{columns}

  \vspace{.35em}
  Under constant returns,
  \[
    \text{positive profit}\Longrightarrow F_t(w_t,r_t)=\varnothing,
    \qquad
    \text{zero profit}\Longrightarrow F_t(w_t,r_t)\text{ may be set-valued}.
  \]
  Equilibrium selects mutually compatible elements of $H$ and $F_t$.
\end{frame}

\begin{frame}{Equilibrium: The General Definition}
  \begin{definitionbox}[The Shape of Any Equilibrium Definition]
    Given the primitives and the initial state, an equilibrium is a list of
    \emph{quantities} and \emph{prices} such that
    \begin{enumerate}
      \item households \textbf{optimize}, taking prices as given;
      \item firms \textbf{optimize}, taking prices as given;
      \item markets \textbf{clear}.
    \end{enumerate}
  \end{definitionbox}

  A definition \emph{names} the objects and \emph{states} the conditions. It
  uses no derivatives, and it does not claim that such a list exists.

  \vspace{.35em}
  It is a \emph{test}, not an adjustment story: it says which lists qualify,
  not how an economy would find one.
\end{frame}

\begin{frame}{Equilibrium: The Actual Definition}
  \emph{The general definition, filled in for the growth model.}

  \vspace{.2em}
  \begin{definitionbox}[Competitive Equilibrium]
    Given $a_0$, a competitive equilibrium consists of household allocations
    $\{c_t,a_{t+1}\}_{t=0}^{\infty}$, firm allocations
    $\{k_t,n_t\}_{t=0}^{\infty}$ and prices $\{w_t,r_t\}_{t=0}^{\infty}$
    such that
    \begin{enumerate}
      \item given $\{w_t,r_t\}$, the household allocation solves the
      \textbf{household problem};
      \item given $\{w_t,r_t\}$, the firm allocation solves the
      \textbf{firm problem} at every date;
      \item \textbf{markets clear} at every date.
    \end{enumerate}
  \end{definitionbox}

  The firm pays $r_t+\delta$ per unit of capital, so the wage and the interest
  rate describe the whole price system.
\end{frame}

\begin{frame}{Market Clearing}
  \begin{definitionbox}[Three Markets, Every Date]
    \[
      \underbrace{a_t=k_t}_{\text{capital}},
      \qquad
      \underbrace{n_t=1}_{\text{labor}},
      \qquad
      \underbrace{c_t+k_{t+1}=k_t^\alpha(A_tn_t)^{1-\alpha}+(1-\delta)k_t}_{\text{goods}} .
    \]
  \end{definitionbox}

  \begin{block}{Walras' Law}
    If both problems are solved and two of the three markets clear, the third
    clears automatically. Goods-market clearing is implied by the other two.
  \end{block}
\end{frame}

\begin{frame}{Equilibrium: Characterization}
  \begin{definitionbox}[Definition versus Characterization]
    \textbf{The definition} \emph{states} conditions: solve, solve, clear.\par
    \smallskip
    \textbf{The characterization} \emph{derives} what the conditions imply:
    the first-order conditions of each problem, joined to the market-clearing
    equations.
  \end{definitionbox}

  This is where derivatives enter, and nowhere earlier.

  \vspace{.35em}
  Existence and uniqueness are separate questions; a characterization settles
  neither.
\end{frame}

\begin{frame}{The Equilibrium System}
  \begin{propositionbox}[Every Equilibrium Path Satisfies, at Every Date]
    \[
      \begin{array}{l@{\qquad}l@{\qquad}l}
        \text{\emph{household}}
          & u'(c_t)=\beta(1+r_{t+1})u'(c_{t+1}),
          & c_t+a_{t+1}=w_t+(1+r_t)a_t,\\[.3em]
        \text{\emph{firm}}
          & w_t=(1-\alpha)A_tk_t^\alpha n_t^{-\alpha},
          & r_t+\delta=\alpha A_tk_t^{\alpha-1}n_t^{1-\alpha},\\[.3em]
        \text{\emph{markets}}
          & a_t=k_t,
          & n_t=1 .
      \end{array}
    \]
  \end{propositionbox}

  Six equations each date for the six unknown sequences
  $(c_t,a_{t+1},k_t,n_t,w_t,r_t)$. The initial condition $a_0=k_0$ and the
  transversality condition select the path; goods-market clearing then follows
  by Walras' law.
\end{frame}

\begin{frame}{Prices as Functions of Capital}
  To reduce the system, set $A_t=1$ and substitute $n_t=1$.

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{block}{General Technology}
        \begin{minipage}[t][2.05cm][t]{\dimexpr\linewidth-.76em\relax}
          \[
            w_t=f_n(k_t,1),
          \]
          \[
            r_t=f_k(k_t,1)-\delta.
          \]
        \end{minipage}
      \end{block}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{block}{Cobb--Douglas Technology}
        \begin{minipage}[t][2.05cm][t]{\dimexpr\linewidth-.76em\relax}
          \[
            w_t=(1-\alpha)k_t^\alpha,
          \]
          \[
            r_t=\alpha k_t^{\alpha-1}-\delta.
          \]
        \end{minipage}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{.45em}
  Firm optimality and market clearing remove prices as independent unknowns.
\end{frame}

\begin{frame}{Consumption from Capital}
  With $n_t=1$ and $A_t=1$, the resource constraint is
  \[
    c_t+k_{t+1}=k_t^\alpha+(1-\delta)k_t.
  \]
  Hence,
  \[
    \boxed{
      c_t
      =
      k_t^\alpha+(1-\delta)k_t-k_{t+1}.
    }
  \]

\begin{frame}{Three Substitutions Produce the Capital Recurrence}
  \setlength{\abovedisplayskip}{.28em}
  \setlength{\belowdisplayskip}{.28em}
  \statementitem{1}{Firm optimality replaces the return:
  \[
    r_{t+1}=\alpha k_{t+1}^{\alpha-1}-\delta.
  \]}
  \statementitem{2}{Feasibility replaces consumption at both dates:
  \[
    c_j=k_j^\alpha+(1-\delta)k_j-k_{j+1},
    \qquad j=t,t+1.
  \]}
  \statementitem{3}{Insert both expressions into the household Euler equation:
  \[
    \boxed{
      \begin{aligned}
        u'\!\left(k_t^\alpha+(1-\delta)k_t-k_{t+1}\right)
        ={}&\beta\left[1+\alpha k_{t+1}^{\alpha-1}-\delta\right]\\
        &\times u'\!\left(k_{t+1}^\alpha+(1-\delta)k_{t+1}-k_{t+2}\right).
      \end{aligned}
    }
  \]}

  The result is a second-order difference equation in
  $(k_t,k_{t+1},k_{t+2})$.
\end{frame}

\begin{frame}{Necessity and Sufficiency}
  \begin{block}{Necessary Direction}
    \[
      \text{interior equilibrium}
      \quad\Longrightarrow\quad
      \text{capital recurrence}.
    \]
  \end{block}

  \begin{block}{Converse Direction}
    \[
      \begin{gathered}
        \text{capital recurrence}+\text{feasibility}+\text{initial condition}\\
        {}+\text{transversality}+\text{sufficient optimality conditions}
      \end{gathered}
      \quad\Longrightarrow\quad
      \text{equilibrium}.
    \]
  \end{block}

  \vspace{.30em}
  \centering
  Neither implication proves that a solution exists or that it is unique.
\end{frame}

\begin{frame}{Pareto Efficiency}
  \begin{definitionbox}[Efficiency]
    A feasible allocation
    $z^*=\left(c_t^*,k_{t+1}^*,n_t^*\right)_{t=0}^{\infty}$ is efficient if
    there is no feasible $\widetilde z$ such that
    $U(\widetilde c)>U(c^*)$.
  \end{definitionbox}

  Efficiency uses only preferences and physical feasibility: no prices and no
  ownership rule. With one household, the planner selects the efficient path by
  maximizing the household's lifetime utility over feasible allocations.
\end{frame}

\begin{frame}{The Planner Problem}
  With $A_t=1$, the planner solves
  \[
    \max_{\left(c_t,k_{t+1},n_t\right)_{t=0}^{\infty}}
    \sum_{t=0}^{\infty}\beta^t u(c_t)
  \]
  subject to
  \[
    c_t+k_{t+1}
    =
    k_t^\alpha n_t^{1-\alpha}+(1-\delta)k_t,
    \qquad t=0,1,\ldots,
  \]
  \[
    c_t\geq0,
    \qquad
    k_{t+1}\geq0,
    \qquad
    0\leq n_t\leq1,
    \qquad
    k_0>0\ \text{given}.
  \]

  Since leisure does not enter utility and production increases in labor,
  $n_t=1$. At an interior optimum,
  \[
    \lim_{t\to\infty}\beta^t u'(c_t)k_{t+1}=0.
  \]
\end{frame}

\begin{frame}{The Planner Euler Equation}
  For the reduced problem, define
  \[
    L=
    \sum_{t=0}^{\infty}\beta^t u(c_t)
    +
    \sum_{t=0}^{\infty}
    \lambda_t
    \left[
      k_t^\alpha+(1-\delta)k_t-c_t-k_{t+1}
    \right].
  \]

  The first-order conditions give
  \[
    \lambda_t=\beta^t u'(c_t),
    \qquad t=0,1,\ldots,
  \]
  \[
    \lambda_t
    =
    \lambda_{t+1}
    \left[
      \alpha k_{t+1}^{\alpha-1}+1-\delta
    \right],
    \qquad t=0,1,\ldots.
  \]
  Therefore,
  \[
    \boxed{
      u'(c_t)
      =
      \beta
      \left[
        \alpha k_{t+1}^{\alpha-1}+1-\delta
      \right]
      u'(c_{t+1}).
    }
  \]
\end{frame}

\begin{frame}{Comparing the Two Euler Equations}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.455\textwidth}
      \begin{block}{Competitive Equilibrium}
        \begin{minipage}[t][3.15cm][t]{\dimexpr\linewidth-.76em\relax}
          Household Euler equation:
          \[
            u'(c_t)=\beta(1+r_{t+1})u'(c_{t+1}).
          \]
          Firm return and clearing:
          \[
            1+r_{t+1}

            1+\alpha k_{t+1}^{\alpha-1}-\delta.
          \]
        \end{minipage}
      \end{block}
    \end{column}
    \begin{column}{.07\textwidth}

    \end{column}
    \begin{column}{.455\textwidth}
      \begin{block}{Social Planner}
        \begin{minipage}[t][3.15cm][t]{\dimexpr\linewidth-.76em\relax}
          Planner return:
          \[
            1+\alpha k_{t+1}^{\alpha-1}-\delta.
          \]
          Planner Euler equation:
          \[
            u'(c_t)
            =

  The equality is economically powerful only because concavity and

\end{frame}

\begin{frame}{The Market and Planner Choose the Same Allocation}
  Assume lifetime utility is well-defined, the relevant optima are interior,
  the problems are concave, and both transversality conditions hold.

  \begin{columns}[t,onlytextwidth]
    \begin{column}{\tightcolumnwidth}
      \begin{theorembox}[Equilibrium $\Longrightarrow$ Planner]
        \begin{minipage}[t][2.85cm][t]{\dimexpr\linewidth-.76em\relax}
          Every equilibrium allocation is a planner optimum:
          \[
            \text{equilibrium}
            \Longrightarrow
            \text{planner optimum}.
          \]
          Markets attain an efficient allocation.
        \end{minipage}
      \end{theorembox}
    \end{column}
    \hspace{\twocolumngutter}
    \begin{column}{\tightcolumnwidth}
      \begin{theorembox}[Planner $\Longrightarrow$ Equilibrium]
        \begin{minipage}[t][2.85cm][t]{\dimexpr\linewidth-.76em\relax}
          Every interior planner optimum can be supported by prices:
          \[
            \text{planner optimum}
            \Longrightarrow
            \text{equilibrium}.
          \]
          Marginal products construct the prices.
        \end{minipage}
      \end{theorembox}
    \end{column}
  \end{columns}

  \vspace{.20em}
  For an infinite horizon, these assumptions are part of the theorem; the Euler
  equations alone are not sufficient.
\end{frame}
