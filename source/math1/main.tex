\begin{frame}{Quantifiers and Negation}
\begin{definitionbox}[Quantifiers]
Fix a domain $X$. A quantifier binds the variable $x$:
\[
\begin{aligned}
\forall x\in X\;P(x)&:&& P(x)\text{ holds for every }x\in X,\\
\exists x\in X\;P(x)&:&& P(x)\text{ holds for at least one }x\in X.
\end{aligned}
\]
An occurrence of $x$ outside the scope of a quantifier is \emph{free}.
\end{definitionbox}

\textbf{Negation laws}\qquad($\Longleftrightarrow$ means ``logically equivalent'')
\begin{align*}
\neg(P\land Q)&\Longleftrightarrow \neg P\lor\neg Q,
&\neg(P\lor Q)&\Longleftrightarrow \neg P\land\neg Q,\\
\neg\bigl[\exists x\in X\;P(x)\bigr]&\Longleftrightarrow \forall x\in X\;\neg P(x),
&\neg\bigl[\forall x\in X\;P(x)\bigr]&\Longleftrightarrow \exists x\in X\;\neg P(x).
\end{align*}
\end{frame}

\begin{frame}{The Contrapositive Is Equivalent}
\begin{truthtable}
\begin{array}{cccccc}
\toprule
P & Q & \lnot Q & \lnot P & \lnot Q \rightarrow \lnot P & P \rightarrow Q \\
\midrule
T & T & F & F & T & T \\
T & F & T & F & F & F \\
F & T & F & T & T & T \\
F & F & T & T & T & T \\
\bottomrule
\end{array}
\end{truthtable}
\begin{center}
$P\rightarrow Q$ and $\neg Q\rightarrow\neg P$ have identical truth values.
Thus either statement may be proved by proving the other.
\end{center}
\end{frame}

\begin{frame}{Proof by Contradiction}
\begin{truthtable}
\begin{array}{cccccc}
\toprule
P & Q & \lnot Q & P \land \lnot Q & \lnot(P \land \lnot Q) & P \rightarrow Q \\
\midrule
T & T & F & F & T & T \\
T & F & T & T & F & F \\
F & T & F & F & T & T \\
F & F & T & F & T & T \\
\bottomrule
\end{array}
\end{truthtable}

\begin{propositionbox}[Method for proving $P\rightarrow Q$]
Assume $P$ and $\neg Q$. If these assumptions imply a contradiction $\bot$, then
$P\land\neg Q$ is impossible. Therefore $P\rightarrow Q$.
\end{propositionbox}
\end{frame}

\begin{frame}{Mathematical Induction}
\begin{theorembox}[Principle of Mathematical Induction]
Let $\mathbb N=\{1,2,3,\ldots\}$ and fix $n_0\in\mathbb N$.  For each natural
number $n\geq n_0$, let $P(n)$ be a statement. Suppose that:
\begin{enumerate}
    \item $P(n_0)$ is true.
    \item For every $k\geq n_0$, if $P(k)$ is true, then $P(k+1)$ is true.
\end{enumerate}
Then $P(n)$ is true for every natural number $n\geq n_0$.
\end{theorembox}
\end{frame}

\begin{definitionbox}[Definition 1.4.1]

\[
A\cap B=\{x\mid x\in A\text{ and }x\in B\}.

\[
A\setminus B=\{x\mid x\in A\text{ and }x\notin B\}.
\]
\end{definitionbox}

\begin{frame}{Set Theory}
\small
\begin{definitionbox}[Definition 1.4.5]
Suppose $A$ and $B$ are sets. We will say that $A$ is a \emph{subset} of $B$ if every element of $A$ is also an element of $B$. We write $A\subseteq B$ to mean that $A$ is a subset of $B$. $A$ and $B$ are said to be \emph{disjoint} if they have no elements in common. Note that this is the same as saying that the set of elements they have in common is the empty set, or in other words $A\cap B=\varnothing$.
\end{definitionbox}

\begin{itemize}
    \item Given $A\subseteq U$, its \emph{complement in $U$} is
    \[
    A^c=U\setminus A=\{x\in U\mid x\notin A\}.
    \]
    \item De Morgan's Laws also apply in Sets:
    \[
    \begin{aligned}
    (A \cup B)^{c} &= A^{c} \cap B^{c},\\[0.3em]
    (A \cap B)^{c} &= A^{c} \cup B^{c}.
    \end{aligned}
    \]

\end{itemize}
\end{frame}

\begin{frame}{Set Theory}
\vspace{0.55cm}
\begin{definitionbox}[Definition 4.1.1]
Suppose $A$ and $B$ are sets. Then the \emph{Cartesian product} of $A$ and $B$, denoted $A\times B$, is the set of all ordered pairs in which the first coordinate is an element of $A$ and the second is an element of $B$. In other words,
\[
A\times B=\{(a,b)\mid a\in A\text{ and }b\in B\}.
\]
\end{definitionbox}
\end{frame}

\begin{frame}{Binary Relations}
\begin{definitionbox}[Definition 4.2.1]
Suppose $A$ and $B$ are sets. A set $R\subseteq A\times B$ is a
\emph{relation from $A$ to $B$}. A \emph{relation on $A$} is a set
$R\subseteq A\times A$.
\end{definitionbox}

\begin{definitionbox}[Properties of a Relation on $A$]
For a relation $R$ on $A$:
\begin{enumerate}
    \item \emph{Reflexive}: $xRx$ for every $x\in A$.
    \item \emph{Complete}: $xRy$ or $yRx$ (or both) for every $x,y\in A$.
    \item \emph{Symmetric}: $xRy\rightarrow yRx$ for every $x,y\in A$.
    \item \emph{Transitive}: $(xRy\land yRz)\rightarrow xRz$ for every $x,y,z\in A$.
\end{enumerate}
\end{definitionbox}
\end{frame}

\begin{frame}{Preference Is the Primitive Object}
\begin{definitionbox}[A Preference Relation Is the Starting Point]
A preference relation $\succeq$ on $X$ records the decision maker's ranking:
\[
x\succeq y
\quad\Longleftrightarrow\quad
x\text{ is at least as good as }y.
\]
\end{definitionbox}

\begin{itemize}
    \item The ranking is taken as given rather than inferred from observed menus.
    \item \textbf{Rationality} restricts the primitive relation: it must be complete
    and transitive.
    \item A \textbf{utility function} is a numerical representation of that same
    ranking, when such a representation exists.
\end{itemize}

\begin{block}{Preference-Based Logic}
Start with preferences, impose rationality, and derive choice behavior.
\end{block}
\end{frame}

\begin{frame}{Choice Behavior Is the Primitive Object}
\begin{definitionbox}[A Choice Rule Is the Starting Point]
For each observed menu $B$, the rule records a nonempty choice set:
\[
C:\mathcal B\to 2^X\setminus\{\varnothing\},
\qquad \varnothing\neq C(B)\subseteq B.
\]
\end{definitionbox}

\begin{itemize}
    \item Choices are taken as given rather than derived from an unobserved ranking.
    \item \textbf{WARP} restricts the primitive rule by ruling out revealed reversals.
    \item Revealed preference is constructed from the observed choice behavior.
\end{itemize}

\begin{block}{Choice-Based Logic}
\textit{Do not waste any attention} speculating about rationality (yet!):
first record choices, then ask whether one rational relation explains them.
\end{block}
\end{frame}

\begin{frame}{What Is a Rich Domain of Menus?}
\begin{definitionbox}[Every Small Menu Is Observable]
The domain is \emph{rich} if every nonempty $B\subseteq X$ with $|B|\leq 3$
belongs to $\mathcal B$. Thus, for distinct $x,y,z\in X$, it contains
\[
\{x\},\{y\},\{z\};\quad
\{x,y\},\{x,z\},\{y,z\};\quad
\{x,y,z\}.
\]
\end{definitionbox}

\begin{itemize}
    \item \textbf{Singletons} anchor choice: $C(\{x\})=\{x\}$.
    \item \textbf{Pairs} recover complete pairwise comparisons.
    \item \textbf{Triples} bring three comparisons together and test transitivity.
\end{itemize}

\begin{block}{Why ``At Most Three''?}
Pairs build comparisons; triples test them. Larger menus are allowed.
\end{block}
\end{frame}

\begin{frame}{Why the Proof Starts with Pairs}
\begin{propositionbox}[Pairs Recover Revealed Preference]
Suppose every pair belongs to $\mathcal B$ and choice is nonempty. Under WARP,
\[
x\succeq^* y
\quad\Longleftrightarrow\quad
x\in C(\{x,y\}).
\]
\end{propositionbox}

\begin{itemize}
    \item $\Leftarrow$ is immediate: the pair itself is an observed menu.
    \item For $\Rightarrow$, $x$ was chosen somewhere while $y$ was available.
    If the pair chooses $y$, WARP forces it to choose $x$ as well.
    \item Since every pair has a choice, either $x\succeq^*y$ or
    $y\succeq^*x$. Hence $\succeq^*$ is complete.
\end{itemize}

\end{frame}

\begin{frame}{Why the Proof Needs Triples}
\begin{definitionbox}[A Cycle Hidden in Pair Menus]
\[
C(\{x,y\})=\{x\},\qquad
C(\{y,z\})=\{y\},\qquad
C(\{x,z\})=\{z\}.
\]
\end{definitionbox}

With pairs alone, WARP does not connect these observations: no two distinct
pair menus contain the same two alternatives. Yet the choices form the cycle
\[
x\text{ over }y,\qquad y\text{ over }z,\qquad z\text{ over }x.
\]

\begin{block}{Add the Triple $T=\{x,y,z\}$}
If $T$ chooses $x$, $y$, or $z$, that alternative loses in one of the pair
menus. WARP would require it to be chosen from that pair too. Thus a cycle
would force $C(T)=\varnothing$, contradicting nonempty choice.
\end{block}
\end{frame}

\begin{frame}{Rationalizing Choice}
\begin{definitionbox}[Rationalization]
A preference relation $\succeq$ \emph{rationalizes} a choice rule $C$ on
$\mathcal B$ if, for every $B\in\mathcal B$,
\[
C(B)=\{x\in B:x\succeq y\text{ for every }y\in B\}.
\]
The choice rule is \emph{rationalizable} if some rational preference relation
rationalizes it.
\end{definitionbox}

Thus, observed choice contains exactly the best available alternatives according
to one complete and transitive relation.
\end{frame}

\begin{frame}{How the Rationalization Proof Works}
\begin{propositionbox}[WARP and Rationalizability]
Suppose $\mathcal{B}$ contains every nonempty subset of $X$ with at most three
alternatives, and $C(B)\neq\varnothing$ for every $B\in\mathcal B$. Then $C$ is
rationalizable if and only if it satisfies WARP.
\end{propositionbox}

\begin{enumerate}
    \item Pair menus recover $\succeq^*$ and make it complete.
    \item Triple menus and WARP rule out cycles, making $\succeq^*$ transitive.
    \item WARP then links pairwise comparisons back to every observed menu:
\end{enumerate}
\[
C(B)=\{x\in B:x\succeq^* y\text{ for every }y\in B\}.
\]

\begin{block}{The Key Step}
If $y\in C(B)$ and $x\succeq^*y$, the pair menu chooses $x$; WARP therefore
forces $x\in C(B)$ too.
\end{block}
\end{frame}
