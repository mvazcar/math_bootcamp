\begin{frame}{Quantifiers and De Morgan's Laws}
\vspace{0.2cm}
\begin{center}
\begin{minipage}{0.82\linewidth}
\large
\textbf{De Morgan's laws}
\begin{align*}
\neg(P\land Q) &\text{ is equivalent to } \neg P\lor\neg Q.\\
\neg(P\lor Q) &\text{ is equivalent to } \neg P\land\neg Q.
\end{align*}

\textbf{Quantifier Negation laws}
\begin{align*}
\neg\exists x\,P(x) &\text{ is equivalent to } \forall x\,\neg P(x).\\
\neg\forall x\,P(x) &\text{ is equivalent to } \exists x\,\neg P(x).
\end{align*}
\end{minipage}
\end{center}
\end{frame}

\begin{frame}{4. Proofs by Induction}

{\small
\textbf{1.2.2 Principle of Mathematical Induction}\quad
\emph{Let $S$ be a subset of $\mathbb{N}$ that possesses the two properties:}
\begin{enumerate}
    \setlength{\itemsep}{0.15em}
    \setlength{\topsep}{0.2em}
    \item \emph{The number $1\in S$.}
    \item \emph{For every $k\in\mathbb{N}$, if $k\in S$, then $k+1\in S$.}
\end{enumerate}
\par\noindent\hspace*{2em}\emph{Then we have $S=\mathbb{N}$.}

\vspace{0.45cm}
\textbf{1.2.3 Principle of Mathematical Induction (second version)}\quad
\emph{Let $n_0\in\mathbb{N}$ and let $P(n)$ be a statement for each natural number $n\geq n_0$. Suppose that:}
\begin{enumerate}
    \setlength{\itemsep}{0.15em}
    \setlength{\topsep}{0.2em}
    \item \emph{The statement $P(n_0)$ is true.}
    \item \emph{For all $k\geq n_0$, the truth of $P(k)$ implies the truth of $P(k+1)$.}
\end{enumerate}
\par\noindent\hspace*{2em}\emph{Then $P(n)$ is true for all $n\geq n_0$.}
}

\end{frame}


\begin{frame}{From Logic to Set Theory}
\small
\begin{definitionbox}[Definition 1.4.1]
The \emph{intersection} of two sets $A$ and $B$ is the set $A\cap B$ defined as follows:
\[
A\cap B=\{x\mid x\in A\text{ and }x\in B\}.
\]

The \emph{union} of $A$ and $B$ is the set $A\cup B$ defined as follows:
\[
A\cup B=\{x\mid x\in A\text{ or }x\in B\}.
\]

The \emph{difference} of $A$ and $B$ is the set $A\setminus B$ defined as follows:
\[
A\setminus B=\{x\mid x\in A\text{ and }x\notin B\}.
\]
\end{definitionbox}
\end{frame}

\begin{frame}{Set Theory}
\small
\begin{definitionbox}[Definition 1.4.5]
Suppose $A$ and $B$ are sets. We will say that $A$ is a \emph{subset} of $B$ if every element of $A$ is also an element of $B$. We write $A\subseteq B$ to mean that $A$ is a subset of $B$. $A$ and $B$ are said to be \emph{disjoint} if they have no elements in common. Note that this is the same as saying that the set of elements they have in common is the empty set, or in other words $A\cap B=\varnothing$.
\end{definitionbox}

\begin{itemize}
    \item Given a set $A \subseteq U$, define the \textit{complement} $A^C$
    \item De Morgan's Laws also apply in Sets:
    \[
    \begin{aligned}
    (A \cup B)^{c} &= A^{c} \cap B^{c},\\[0.3em]
    (A \cap B)^{c} &= A^{c} \cup B^{c}.
    \end{aligned}
    \]

\end{itemize}
\end{frame}

\begin{frame}{Set Theory}
\vspace{0.55cm}
\begin{definitionbox}[Definition 4.1.1]
Suppose $A$ and $B$ are sets. Then the \emph{Cartesian product} of $A$ and $B$, denoted $A\times B$, is the set of all ordered pairs in which the first coordinate is an element of $A$ and the second is an element of $B$. In other words,
\[
A\times B=\{(a,b)\mid a\in A\text{ and }b\in B\}.
\]
\end{definitionbox}
\end{frame}

\begin{frame}{Binary Relations}
\small
\begin{definitionbox}[Definition 4.2.1]
Suppose $A$ and $B$ are sets. Then a set $R\subseteq A\times B$ is called a \emph{relation from $A$ to $B$}.
\end{definitionbox}

\begin{definitionbox}[Definition 4.3.2]
Suppose $R$ is a relation on $A$.
\begin{enumerate}
    \setlength{\itemsep}{0.35em}
    \item $R$ is said to be \emph{reflexive on $A$} (or just \emph{reflexive}, if $A$ is clear from context) if $\forall x\in A\,(xRx)$, or in other words $\forall x\in A\,((x,x)\in R)$.
    \item $R$ is \emph{symmetric} if $\forall x\in A\,\forall y\in A\,(xRy\rightarrow yRx)$.
    \item $R$ is \emph{transitive} if $\forall x\in A\,\forall y\in A\,\forall z\in A\,((xRy\land yRz)\rightarrow xRz)$.
\end{enumerate}
\end{definitionbox}
\end{frame}

\begin{frame}{From Preferences to Choice}
\begin{propositionbox}[WARP and Rationalizability]
Suppose $\mathcal{B}$ contains every subset of $X$ with at most three elements.
Then a nonempty choice rule $C$ is rationalizable if and only if it satisfies WARP.
\end{propositionbox}

\begin{itemize}
    \item Preferences generate choices.
    \item On this rich choice domain, WARP tells us whether observed choices can be explained by rational preferences.
\end{itemize}
\end{frame}
