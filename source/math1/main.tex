\begin{frame}{Quantifiers and Negation}
\begin{definitionbox}[Quantifiers]
Fix a domain $X$. A quantifier binds the variable $x$:
\[
\begin{aligned}
\forall x\in X\;P(x)&:&& P(x)\text{ holds for every }x\in X,\\
\exists x\in X\;P(x)&:&& P(x)\text{ holds for at least one }x\in X.
\end{aligned}
\]
An occurrence of $x$ outside the scope of a quantifier is \emph{free}.
\end{definitionbox}

\textbf{Negation laws}\qquad($\Longleftrightarrow$ means ``logically equivalent'')
\begin{align*}
\neg(P\land Q)&\Longleftrightarrow \neg P\lor\neg Q,
&\neg(P\lor Q)&\Longleftrightarrow \neg P\land\neg Q,\\
\neg\bigl[\exists x\in X\;P(x)\bigr]&\Longleftrightarrow \forall x\in X\;\neg P(x),
&\neg\bigl[\forall x\in X\;P(x)\bigr]&\Longleftrightarrow \exists x\in X\;\neg P(x).
\end{align*}
\end{frame}

\begin{propositionbox}[Method for proving $P\rightarrow Q$]
Assume $P$ and $\neg Q$. If these assumptions imply a contradiction $\bot$, then
$P\land\neg Q$ is impossible. Therefore $P\rightarrow Q$.
\end{propositionbox}
\end{frame}

\begin{frame}{Mathematical Induction}
\begin{theorembox}[Bartle 1.2.3 Principle of Mathematical Induction]
Let $\mathbb N=\{1,2,3,\ldots\}$ and fix $n_0\in\mathbb N$.  For each natural
number $n\geq n_0$, let $P(n)$ be a statement. Suppose that:
\begin{enumerate}
    \item $P(n_0)$ is true.
    \item For every $k\geq n_0$, if $P(k)$ is true, then $P(k+1)$ is true.
\end{enumerate}
Then $P(n)$ is true for every natural number $n\geq n_0$.
\end{theorembox}
\end{frame}


\begin{frame}{From Logic to Set Theory}
\small
\begin{definitionbox}[Definition 1.4.1]
The \emph{intersection} of two sets $A$ and $B$ is the set $A\cap B$ defined as follows:
\[
A\cap B=\{x\mid x\in A\text{ and }x\in B\}.
\]

The \emph{union} of $A$ and $B$ is the set $A\cup B$ defined as follows:
\[
A\cup B=\{x\mid x\in A\text{ or }x\in B\}.
\]

The \emph{difference} of $A$ and $B$ is the set $A\setminus B$ defined as follows:
\[
A\setminus B=\{x\mid x\in A\text{ and }x\notin B\}.
\]
\end{definitionbox}
\end{frame}

\begin{frame}{Set Theory}
\small
\begin{definitionbox}[Definition 1.4.5]
Suppose $A$ and $B$ are sets. $A$ is a \emph{subset} of $B$, written
$A\subseteq B$, if every element of $A$ is also an element of $B$. $A$ and $B$
are \emph{disjoint} if they have no elements in common, that is, if
$A\cap B=\varnothing$.
\end{definitionbox}

\begin{itemize}
    \item Given $A\subseteq U$, its \emph{complement in $U$} is
    \[
    A^c=U\setminus A=\{x\in U\mid x\notin A\}.
    \]
    \item De Morgan's Laws also apply in Sets:
    \[
    \begin{aligned}
    (A \cup B)^{c} &= A^{c} \cap B^{c},\\[0.3em]
    (A \cap B)^{c} &= A^{c} \cup B^{c}.
    \end{aligned}
    \]

\end{itemize}
\end{frame}

\begin{frame}{Set Theory}
\begin{definitionbox}[Definition 4.1.1]
For sets $A$ and $B$, the \emph{Cartesian product} $A\times B$ is the set of
ordered pairs whose first coordinate lies in $A$ and second in $B$:
\[
A\times B=\{(a,b)\mid a\in A\text{ and }b\in B\}.
\]
\end{definitionbox}
\end{frame}

\begin{frame}{Binary Relations}
\begin{definitionbox}[Definition 4.2.1]
Suppose $A$ and $B$ are sets. A set $R\subseteq A\times B$ is a
\emph{relation from $A$ to $B$}. A \emph{relation on $A$} is a set
$R\subseteq A\times A$.
\end{definitionbox}

\begin{definitionbox}[Properties of a Relation on $A$]
For a relation $R$ on $A$:
\begin{enumerate}
    \item \emph{Reflexive}: $xRx$ for every $x\in A$.
    \item \emph{Complete}: $xRy$ or $yRx$ (or both) for every $x,y\in A$.
    \item \emph{Symmetric}: $xRy\rightarrow yRx$ for every $x,y\in A$.
    \item \emph{Transitive}: $(xRy\land yRz)\rightarrow xRz$ for every $x,y,z\in A$.
\end{enumerate}
\end{definitionbox}
\end{frame}

\begin{frame}{Preference as the Primitive}
\begin{definitionbox}[A Preference Relation Is the Starting Point]
A preference relation $\succeq$ on $X$ records the decision maker's ranking:
\[
x\succeq y
\quad\Longleftrightarrow\quad
x\text{ is at least as good as }y.
\]
\end{definitionbox}

\begin{itemize}
    \item \textbf{Rationality} restricts it: complete and transitive.
    \item A \textbf{utility function} represents it: $x\succeq y\iff u(x)\geq u(y)$.
\end{itemize}

\end{frame}

\begin{frame}{WARP and Revealed Preference}
\begin{definitionbox}[Definition 1.C.1: Weak Axiom of Revealed Preference]
If $x,y\in B$ and $x\in C(B)$, then for every $B'\in\mathbb B$ with $x,y\in B'$,
\[
y\in C(B')
\quad\Longrightarrow\quad
x\in C(B').
\]
\end{definitionbox}

\begin{definitionbox}[Definition 1.C.2: Revealed Preference]
\[
x\succeq^* y
\quad\Longleftrightarrow\quad
x,y\in B\ \text{and}\ x\in C(B)\ \text{for some }B\in\mathbb B.
\]
Read: $x$ is \emph{revealed at least as good as} $y$. Informally, $x$ is
\emph{revealed preferred} to $y$ if in addition $y\notin C(B)$.
\end{definitionbox}

\end{frame}

\begin{frame}{A Rich Domain of Menus}
\begin{definitionbox}[Every Small Menu Is Observable]
The domain is \emph{rich} if every nonempty $B\subseteq X$ with $|B|\leq 3$
belongs to $\mathbb B$. Thus, for distinct $x,y,z\in X$, it contains
\[
\{x\},\{y\},\{z\};\quad
\{x,y\},\{x,z\},\{y,z\};\quad
\{x,y,z\}.
\]
\end{definitionbox}

\begin{itemize}
    \item \textbf{Pairs} recover pairwise comparisons.
    \item \textbf{Triples} test them for transitivity.
\end{itemize}
\end{frame}

\begin{frame}{Rationalizing Choice}
\begin{definitionbox}[Rationalization]
A preference relation $\succeq$ \emph{rationalizes} a choice rule $C$ on
$\mathbb B$ if, for every $B\in\mathbb B$,
\[
C(B)=\{x\in B:x\succeq y\text{ for every }y\in B\}.
\]
The choice rule is \emph{rationalizable} if some rational preference relation
rationalizes it.
\end{definitionbox}

\end{frame}

\begin{frame}{WARP and Rationalizability}
\begin{theorembox}
Suppose $\mathbb B$ contains every nonempty subset of $X$ with at most three
alternatives, and $C(B)\neq\varnothing$ for every $B\in\mathbb B$. Then $C$ is
rationalizable if and only if it satisfies WARP.
\end{theorembox}

\begin{itemize}
    \item \textbf{Pairs} make $\succeq^*$ complete.
    \item \textbf{Triples} make $\succeq^*$ transitive.
\end{itemize}
\end{frame}

\begin{frame}{Demand on Walrasian Menus}
\begin{definitionbox}[Restrict the Menu Domain]
Consumer demand evaluates the same choice rule on Walrasian menus:
\[
B(p,I)=\{x\in\mathbb{R}_+^L:p\cdot x\leq I\},
\qquad
x(p,I):=C\bigl(B(p,I)\bigr).
\]
When $C$ is rationalized by $\succeq$ and represented by $u$,
\[
x(p,I)
=C_{\succeq}\bigl(B(p,I)\bigr)
=\operatorname*{arg\,max}_{x\in B(p,I)}u(x).
\]
\end{definitionbox}

\vspace{.25em}
\textbf{Important.}\quad
The converse above used a rich domain. Walrasian budgets are restricted, so
WARP alone does not reproduce it.

\end{frame}
