\begin{frame}{Quantifiers and Negation}
\begin{definitionbox}[Quantifiers]
Fix a domain $X$. A quantifier binds the variable $x$:
\[
\begin{aligned}
\forall x\in X\;P(x)&:&& P(x)\text{ holds for every }x\in X,\\
\exists x\in X\;P(x)&:&& P(x)\text{ holds for at least one }x\in X.
\end{aligned}
\]
An occurrence of $x$ outside the scope of a quantifier is \emph{free}.
\end{definitionbox}

\textbf{Negation laws}\qquad($\Longleftrightarrow$ means ``logically equivalent'')
\begin{align*}
\neg(P\land Q)&\Longleftrightarrow \neg P\lor\neg Q,
&\neg(P\lor Q)&\Longleftrightarrow \neg P\land\neg Q,\\
\neg\bigl[\exists x\in X\;P(x)\bigr]&\Longleftrightarrow \forall x\in X\;\neg P(x),
&\neg\bigl[\forall x\in X\;P(x)\bigr]&\Longleftrightarrow \exists x\in X\;\neg P(x).
\end{align*}
\end{frame}

\begin{frame}{The Contrapositive Is Equivalent}
\begin{truthtable}
\begin{array}{cccccc}
\toprule
P & Q & \lnot Q & \lnot P & \lnot Q \rightarrow \lnot P & P \rightarrow Q \\
\midrule
T & T & F & F & T & T \\
T & F & T & F & F & F \\
F & T & F & T & T & T \\
F & F & T & T & T & T \\
\bottomrule
\end{array}
\end{truthtable}
\begin{center}
$P\rightarrow Q$ and $\neg Q\rightarrow\neg P$ have identical truth values.
Thus either statement may be proved by proving the other.
\end{center}

\begin{frame}{Proof by Contradiction}
\begin{truthtable}
\begin{array}{cccccc}
\toprule
P & Q & \lnot Q & P \land \lnot Q & \lnot(P \land \lnot Q) & P \rightarrow Q \\
\midrule
T & T & F & F & T & T \\
T & F & T & T & F & F \\
F & T & F & F & T & T \\
F & F & T & F & T & T \\
\bottomrule
\end{array}
\end{truthtable}

\begin{propositionbox}[Method for proving $P\rightarrow Q$]
Assume $P$ and $\neg Q$. If these assumptions imply a contradiction $\bot$, then
$P\land\neg Q$ is impossible. Therefore $P\rightarrow Q$.
\end{propositionbox}
\end{frame}

\begin{frame}{Mathematical Induction}
\begin{theorembox}[Principle of Mathematical Induction]
Let $\mathbb N=\{1,2,3,\ldots\}$ and fix $n_0\in\mathbb N$.  For each natural
number $n\geq n_0$, let $P(n)$ be a statement. Suppose that:
\begin{enumerate}
    \item $P(n_0)$ is true.
    \item For every $k\geq n_0$, if $P(k)$ is true, then $P(k+1)$ is true.
\end{enumerate}
Then $P(n)$ is true for every natural number $n\geq n_0$.
\end{theorembox}
\end{frame}


\begin{frame}{From Logic to Set Theory}
\small
\begin{definitionbox}[Definition 1.4.1]
The \emph{intersection} of two sets $A$ and $B$ is the set $A\cap B$ defined as follows:
\[
A\cap B=\{x\mid x\in A\text{ and }x\in B\}.
\]

The \emph{union} of $A$ and $B$ is the set $A\cup B$ defined as follows:
\[
A\cup B=\{x\mid x\in A\text{ or }x\in B\}.
\]

The \emph{difference} of $A$ and $B$ is the set $A\setminus B$ defined as follows:
\[
A\setminus B=\{x\mid x\in A\text{ and }x\notin B\}.
\]
\end{definitionbox}
\end{frame}

\begin{frame}{Set Theory}
\small
\begin{definitionbox}[Definition 1.4.5]
Suppose $A$ and $B$ are sets. We will say that $A$ is a \emph{subset} of $B$ if every element of $A$ is also an element of $B$. We write $A\subseteq B$ to mean that $A$ is a subset of $B$. $A$ and $B$ are said to be \emph{disjoint} if they have no elements in common. Note that this is the same as saying that the set of elements they have in common is the empty set, or in other words $A\cap B=\varnothing$.
\end{definitionbox}

\begin{itemize}
    \item Given $A\subseteq U$, its \emph{complement in $U$} is
    \[
    A^c=U\setminus A=\{x\in U\mid x\notin A\}.
    \]
    \item De Morgan's Laws also apply in Sets:
    \[
    \begin{aligned}
    (A \cup B)^{c} &= A^{c} \cap B^{c},\\[0.3em]
    (A \cap B)^{c} &= A^{c} \cup B^{c}.
    \end{aligned}
    \]

\end{itemize}
\end{frame}

\begin{frame}{Set Theory}
\vspace{0.55cm}
\begin{definitionbox}[Definition 4.1.1]
Suppose $A$ and $B$ are sets. Then the \emph{Cartesian product} of $A$ and $B$, denoted $A\times B$, is the set of all ordered pairs in which the first coordinate is an element of $A$ and the second is an element of $B$. In other words,
\[
A\times B=\{(a,b)\mid a\in A\text{ and }b\in B\}.
\]
\end{definitionbox}
\end{frame}

\begin{frame}{Binary Relations}
\begin{definitionbox}[Definition 4.2.1]
Suppose $A$ and $B$ are sets. A set $R\subseteq A\times B$ is a
\emph{relation from $A$ to $B$}. A \emph{relation on $A$} is a set
$R\subseteq A\times A$.
\end{definitionbox}

\begin{definitionbox}[Properties of a Relation on $A$]
For a relation $R$ on $A$:
\begin{enumerate}
    \item \emph{Reflexive}: $xRx$ for every $x\in A$.
    \item \emph{Complete}: $xRy$ or $yRx$ (or both) for every $x,y\in A$.
    \item \emph{Symmetric}: $xRy\rightarrow yRx$ for every $x,y\in A$.
    \item \emph{Transitive}: $(xRy\land yRz)\rightarrow xRz$ for every $x,y,z\in A$.
\end{enumerate}
\end{definitionbox}
\end{frame}

\begin{frame}{Choice Behavior Is the Primitive Object}
\begin{definitionbox}[A Choice Rule as a Set-Valued Map]
\[
C:\mathcal B\to 2^X\setminus\{\varnothing\},
\qquad B\mapsto C(B)\subseteq B.
\]
\end{definitionbox}

\begin{itemize}
    \item $B$ is the set of alternatives available to the decision maker.
    \item $C(B)$ is the set of acceptable choices from $B$; it may contain more than one alternative.
    \item $C$ has exactly one output for each $B$, but that output is itself a set.
\end{itemize}

\begin{block}{Order of the Argument}
\textit{Do not waste any attention} speculating about rationality (yet!):
first record choices, then ask whether one rational relation explains them.
\end{block}
\end{frame}

\begin{frame}{WARP Rules Out a Revealed Reversal}
\begin{definitionbox}[Two Observed Menus]
\[
B=\{a,b\},\quad C(B)=\{a\},
\qquad
B'=\{a,b,c\},\quad C(B')=\{b\}.
\]
\end{definitionbox}

Both $a$ and $b$ are available in both menus. We first observe $a$ chosen
when $b$ is available; we then observe $b$ chosen while $a$ disappears from
the choice set. This violates WARP.

\begin{block}{What WARP Does}
WARP detects a revealed reversal across menus. It requires no story about the
decision maker's motives.
\end{block}
\end{frame}

\begin{frame}{Rationalizing Choice}
\begin{definitionbox}[Rationalization]
A preference relation $\succeq$ \emph{rationalizes} a choice rule $C$ on
$\mathcal B$ if, for every $B\in\mathcal B$,
\[
C(B)=\{x\in B:x\succeq y\text{ for every }y\in B\}.
\]
The choice rule is \emph{rationalizable} if some rational preference relation
rationalizes it.
\end{definitionbox}

Thus, observed choice contains exactly the best available alternatives according
to one complete and transitive relation.
\end{frame}

\begin{frame}{From Choice to Rational Preferences}
\begin{propositionbox}[WARP and Rationalizability]
Suppose $\mathcal{B}$ contains every nonempty subset of $X$ with at most three
alternatives, and $C(B)\neq\varnothing$ for every $B\in\mathcal B$. Then $C$ is
rationalizable if and only if it satisfies WARP.
\end{propositionbox}

\begin{itemize}
    \item Menus with two alternatives recover pairwise revealed comparisons.
    \item Menus with three alternatives test whether those comparisons can be transitive.
\end{itemize}

\begin{block}{Scope}
This equivalence uses a rich domain of menus. Observed price-income budgets
require a different revealed-preference theorem.
\end{block}
\end{frame}

\begin{frame}{Choice, Preferences, and Utility Are Distinct}
\begin{itemize}
    \item \textbf{Choice $\to$ preferences:} Can one complete and transitive
    relation reproduce every observed choice set?
    \item \textbf{Preferences $\to$ utility:} Can those rankings be represented
    by real numbers?
    \item \textbf{Preferences and a menu $\to$ choice:} Does the menu contain
    a best alternative?
\end{itemize}

\begin{block}{Different Questions, Different Assumptions}
WARP and a rich menu domain address rationalization. Days 2 and 3 study
utility representation. Existence of a best alternative requires additional
assumptions on the menu and the representation.
\end{block}
\end{frame}
