\begin{frame}{Preference Relations}
  \begin{itemize}
    \item A \textbf{preference relation} $\succeq$ on $X$ is a subset of $X\times X$.
    \item $x \succeq y$ reads: $x$ is \textit{at least as good} as y (x is \textit{weakly better} than y).
    \vspace{1em}
    \item From $\succeq$ we derive:
    \begin{itemize}
      \item Strict preference: $x \succ y \iff (x \succeq y \ \text{and not } y \succeq x)$.
      \item Indifference: $x \sim y \iff (x \succeq y \ \text{and } y \succeq x)$.
    \end{itemize}
  \end{itemize}

\end{frame}

% --- Slide 8 ---
\begin{frame}{Properties of Indifference}
  \begin{block}{Indifference Relation}
    If $\succeq$ is rational (i.e., complete and transitive), then the induced indifference relation $\sim$ is:
    \begin{enumerate}
      \item \textbf{Reflexive}: $x \sim x$ for all $x \in X$.
      \item \textbf{Symmetric}: $x \sim y \;\implies\; y \sim x$.
      \item \textbf{Transitive}: $x \sim y \;\wedge\; y \sim z \;\implies\; x \sim z$.
    \end{enumerate}
  \end{block}

    \begin{block}{Equivalence Relation}
    An equivalence relation is a binary relation that is reflexive, symmetric and transitive. \\
    \end{block}  

\end{frame}

% --- Slide 9 ---
\begin{frame}{Equivalence Classes}
  \begin{block}{Equivalence Classes}  
    An equivalence relation induces a \textbf{partition} of the set: mutually disjoint subsets whose union equals the entire set.
  \end{block}

  \vspace{0.3cm}
  If $\succeq$ is rational, its induced indifference relation $\sim$
  \begin{enumerate}
    \item is an equivalence relation;
    \item generates equivalence classes of indifferent elements.
  \end{enumerate}

  \vspace{0.3cm}
  For each $x \in X$, the equivalence class of $x$ is denoted by
  \[
    [x] = \{ y \in X \mid x \sim y \}.
  \]
\end{frame}


\begin{frame}{Equivalence Classes and Utility}
  \begin{block}{Utility representation}
    A function $u:X\to\mathbb{R}$ \emph{represents} $\succeq$ if
    \[
      x\succeq y \;\;\Leftrightarrow\;\; u(x)\ge u(y).
    \]
  \end{block}

  \begin{block}{What the Representation Implies}
    If $u$ represents $\succeq$, then
    \[
      x\sim y \;\;\Leftrightarrow\;\; u(x) = u(y).
    \]
    Every equivalence class maps to one utility level; utility is
    \textbf{ordinal}.
  \end{block}

\end{frame}

\begin{frame}{Monotonicity: More is Better}

\begin{definitionbox}[MWG 3.B.2: Monotonicity]
Assume consuming more is always feasible: $x\in X$ and $y\geq x$ imply $y\in X$.
The preference relation $\succeq$ on $X$ is\\
\quad \emph{monotone} if $y\gg x$ implies $y\succ x$;\\
\quad \emph{strongly monotone} if $y\geq x$ and $y\ne x$ imply $y\succ x$.
\end{definitionbox}

\begin{propositionbox}
If $u:X\to\mathbb{R}$ represents $\succeq$, then\\
\quad $\succeq$ is monotone \emph{iff} $u(y)>u(x)$ whenever $y\gg x$;\\
\quad $\succeq$ is strongly monotone \emph{iff} $u$ is \emph{strictly
increasing}: $u(y)>u(x)$ whenever $y\geq x$ and $y\ne x$.
\end{propositionbox}
\end{frame}

\begin{frame}{Local Nonsatiation: No Thick Indifference Curves}
\begin{definitionbox}[Definition: Local Nonsatiation]
A preference relation $\succeq$ on $X$ is \emph{locally nonsatiated} if for every $x \in X$ and
every $\varepsilon > 0$, there exists $y \in X$ such that $\lVert x - y\rVert \le \varepsilon$ and $y \succ x$.
\end{definitionbox}

\centering
Euclidean distance:
$\lVert x - y\rVert=\sqrt{\sum_{i=1}^L (x_i - y_i)^2}$.
\end{frame}

\begin{frame}{Convexity: Taste for Variety}

\begin{definitionbox}[Convex Sets and Preferences]
A set $X\subseteq\mathbb{R}^L$ is \emph{convex} if $x,y\in X$ and
$\alpha\in[0,1]$ imply $\alpha x+(1-\alpha)y\in X$.

Let $X$ be convex and $U(x):=\{y\in X:y\succeq x\}$. Then $\succeq$ is
\emph{convex} if
\[
y,z\in U(x),\ \alpha\in[0,1]
\quad\Longrightarrow\quad
\alpha y+(1-\alpha)z\in U(x),
\]
and \emph{strictly convex} if that conclusion is $\succ x$ for $y\ne z$,
$\alpha\in(0,1)$.
\end{definitionbox}

\vspace{.5em}
\centering Every upper contour set $U(x)$ is convex.

\end{frame}


\begin{frame}{Level Sets and Quasiconcavity}

\begin{block}{Definitions}
Let $X\subseteq\mathbb{R}^L$ be convex. A function $f:X\to\mathbb{R}$ is
\emph{quasiconcave} if, for $x,y\in X$ and $\alpha\in[0,1]$,
\[
f(\alpha x+(1-\alpha)y)\geq\min\{f(x),f(y)\}.
\]
It is \emph{strictly quasiconcave} if, for $x \ne y$ and $\alpha \in (0,1)$,
\[
f(\alpha x + (1-\alpha)y) > \min\{f(x),f(y)\}.
\]
\end{block}

\vspace{.5em}
\textbf{Equivalently.}\quad $f$ is quasiconcave iff every upper level set
$\{\,x\in X \mid f(x)\geq c\,\}$ is convex.
\end{frame}

\begin{frame}{Convex Preferences and Quasiconcave Utility}

\begin{propositionbox}[Kreps 2.8(b)]
Let $X$ be convex and let $u:X\to\mathbb{R}$ represent $\succeq$. Then
\[
\begin{aligned}
\succeq\ \text{is convex}
&\quad\Longleftrightarrow\quad
u\ \text{is quasiconcave},\\
\succeq\ \text{is strictly convex}
&\quad\Longleftrightarrow\quad
u\ \text{is strictly quasiconcave}.
\end{aligned}
\]
\end{propositionbox}


\end{frame}

\begin{frame}{Topology on the Choice Set}

\begin{definitionbox}[Relative Topology on $X\subseteq\mathbb{R}^L$]
For $x\in X$ and $\varepsilon>0$, the relative open ball is
\[
B^X_\varepsilon(x)=\{y\in X:\lVert y-x\rVert<\varepsilon\}.
\]
\begin{itemize}
    \item A set $G\subseteq X$ is \emph{open relative to $X$} if every $x\in G$
    has some $\varepsilon>0$ with $B^X_\varepsilon(x)\subseteq G$.
    \item A set $F\subseteq X$ is \emph{closed relative to $X$} if $X\setminus F$ is open relative to $X$.
\end{itemize}
\end{definitionbox}


\end{frame}

\begin{frame}{Closed Sets Contain Their Limits}
\begin{theorembox}[Sequential Test for Closed Sets]
Let $F\subseteq X\subseteq\mathbb{R}^L$. The set $F$ is closed relative to
$X$ if and only if
\[
x_n\in F\ \text{for every }n,\qquad x_n\to x\in X
\quad\Longrightarrow\quad x\in F.
\]
\end{theorembox}

\end{frame}

\begin{frame}{Lower and Upper Contour Sets}
\begin{definitionbox}[Contour Sets at $x$]
\[
U(x) := \{ y \in X \mid y \succeq x \} \qquad L(x) := \{ y \in X \mid x \succeq y \}.
\]
The indifference set through $x$ is always
\[
I(x):=\{y\in X:y\sim x\}=U(x)\cap L(x).
\]
\end{definitionbox}

\begin{propositionbox}[Weak and Strict Sections]
If $\succeq$ is complete, then
\[
\{y\in X:y\succ x\}=X\setminus L(x),
\qquad
\{y\in X:x\succ y\}=X\setminus U(x).
\]
\end{propositionbox}
\end{frame}

\begin{frame}{Lexicographic Preferences}
\begin{definitionbox}[Example 3.C.1: The Lexicographic Preference Relation]
Let $X=\mathbb{R}_+^2$. Define $x\succeq y$ if either
\[
x_1>y_1,
\qquad\text{or}\qquad
x_1=y_1\ \text{and}\ x_2\geq y_2.
\]
It is complete, transitive, strongly monotone, and strictly convex --- yet no
utility function represents it.
\end{definitionbox}

\end{frame}

\begin{frame}{Continuity: Limits of Rankings}
\begin{definitionbox}[MWG 3.C.1: Continuity]
Let $X\subseteq\mathbb{R}^L$ and let $\succeq$ be complete. Then $\succeq$ is
\emph{continuous} if it is preserved under limits:
\[
(x_n,y_n)\to(x,y),\quad x_n\succeq y_n\ \text{for every }n
\quad\Longrightarrow\quad x\succeq y.
\]
\end{definitionbox}


\end{frame}

\begin{frame}{Continuity: Equivalent Tests}
\begin{propositionbox}[Kreps 1.14: Equivalent Characterizations]
For complete, transitive $\succeq$, continuity is equivalent to each of:
\begin{enumerate}
    \item $U(x)$ and $L(x)$ are closed relative to $X$, for every $x$.
    \item $\{y\in X:y\succ x\}$ and $\{y\in X:x\succ y\}$ are open relative to
    $X$, for every $x$.
    \item If $x\succ y$, there is $\varepsilon>0$ with $x'\succ y'$ whenever
    $\lVert x'-x\rVert<\varepsilon$ and $\lVert y'-y\rVert<\varepsilon$.
\end{enumerate}
\end{propositionbox}

\centering
Kreps takes (3) as the definition; MWG takes the limit form.
\end{frame}

\begin{frame}{Continuous Representations and Closed Contours}
\begin{theorembox}[Closed Level Sets]
If $f:X\to\mathbb{R}$ is continuous, then for every $c\in\mathbb{R}$ the sets
\[
f^{-1}\bigl([c,\infty)\bigr)
\qquad\text{and}\qquad
f^{-1}\bigl((-\infty,c]\bigr)
\]
are closed relative to $X$.
\end{theorembox}

\vspace{.5em}
\centering A continuous representation therefore implies continuous
preferences.
\end{frame}

\begin{frame}{Debreu's Representation Theorem}
\begin{theorembox}[Kreps 2.9: Debreu's Theorem]
Let $X=\mathbb{R}_+^L$. If $\succeq$ is complete, transitive, and continuous on
$X$, then there exists a \emph{continuous} utility function $u:X\to\mathbb{R}$ such that
\[
x\succeq y \quad\Longleftrightarrow\quad u(x)\geq u(y).
\]
\end{theorembox}

\begin{itemize}
    \item It gives \emph{at least one} continuous representation --- not every
    representation is continuous.
\end{itemize}

\end{frame}
