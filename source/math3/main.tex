\begin{frame}{Preference Relations}
  \begin{itemize}
    \item A \textbf{preference relation} $\succeq$ is a binary relation on $X$, that is, a subset of $X \times X$ that describes how the DM ranks any two elements in X.
    \item $x \succeq y$ reads: $x$ is \textit{at least as good} as y (x is \textit{weakly better} than y).
    \vspace{1em}
    \item From $\succeq$ we derive:
    \begin{itemize}
      \item Strict preference: $x \succ y \iff (x \succeq y \ \text{and not } y \succeq x)$.
      \item Indifference: $x \sim y \iff (x \succeq y \ \text{and } y \succeq x)$.
    \end{itemize}
  \end{itemize}

\end{frame}

% --- Slide 8 ---
\begin{frame}{Properties of Indifference}
  \begin{block}{Indifference Relation}
    If $\succeq$ is rational (i.e., complete and transitive), then the induced indifference relation $\sim$ is:
    \begin{enumerate}
      \item \textbf{Reflexive}: $x \sim x$ for all $x \in X$.
      \item \textbf{Symmetric}: $x \sim y \;\implies\; y \sim x$.
      \item \textbf{Transitive}: $x \sim y \;\wedge\; y \sim z \;\implies\; x \sim z$.
    \end{enumerate}
  \end{block}

    \begin{block}{Equivalence Relation}
    An equivalence relation is a binary relation that is reflexive, symmetric and transitive. \\
    \end{block}  

\end{frame}

% --- Slide 9 ---
\begin{frame}{Equivalence Classes}
  \begin{block}{Equivalence Classes}  
    An equivalence relation induces a \textbf{partition} of the set: mutually disjoint subsets whose union equals the entire set.
  \end{block}

  \vspace{0.3cm}
  Note that the indifference relation $\sim$
  \begin{enumerate}
    \item Is an equivalence relation.
    \item It generates equivalence classes of indifferent elements.
  \end{enumerate}

  \vspace{0.3cm}
  For each $x \in X$, the equivalence class of $x$ is denoted by
  \[
    [x] = \{ y \in X \mid x \sim y \}.
  \]
\end{frame}


\begin{frame}{Equivalence Classes and Utility}
  \begin{block}{Utility representation}
    A function $u:X\to\mathbb{R}$ \emph{represents} $\succeq$ if
    \[
      x\succeq y \;\;\Leftrightarrow\;\; u(x)\ge u(y).
    \]
  \end{block}

  \begin{block}{What the Representation Implies}
    If $u$ represents $\succeq$, then
    \[
      x\sim y \;\;\Leftrightarrow\;\; u(x) = u(y).
    \]
    Thus, every equivalence class is mapped to a single utility level. Moreover, if
    $\phi:\mathbb{R}\to\mathbb{R}$ is strictly increasing, then $\phi\circ u$
    represents the same preferences: utility is \textbf{ordinal}.
  \end{block}

\end{frame}

\begin{frame}{Monotonicity: More is Better}

\begin{definitionbox}[Definitions]
A preference relation $\succeq$ is\\
\quad \emph{weakly monotone} if $x \ge y$ implies $x \succeq y$.\\
\quad \emph{strictly monotone} if $x \ge y$ and $x \ne y$ imply
$x \succ y$.\par

A function $f: \mathbb{R}^L \to \mathbb{R}$ is\\
\quad \emph{non-decreasing} if $x \ge y$ implies $f(x) \ge f(y)$ for all $x,y \in \mathbb{R}^L$;\\
\quad \emph{strictly increasing} if $x \ge y$ and $x \ne y$ implies $f(x) > f(y)$ $ \forall x,y \in \mathbb{R}^L$.
\end{definitionbox}

\begin{propositionbox}
If $u$ represents $\succeq$ then\\
\quad $\succeq$ is weakly monotone \emph{iff} $u$ is non-decreasing;\\
\quad $\succeq$ is strictly monotone \emph{iff} $u$ is strictly increasing.
\end{propositionbox}
\end{frame}

\begin{frame}{Local Nonsatiation: No Thick Indifference Curves}
\begin{definitionbox}[Definition: Local Nonsatiation]
A preference relation $\succeq$ on $X$ is \emph{locally nonsatiated} if for every $x \in X$ and
every $\varepsilon > 0$, there exists $y \in X$ such that $\lVert x - y\rVert \le \varepsilon$ and $y \succ x$.
\end{definitionbox}

Thus, preferences are locally nonsatiated if for each $x$ you can find an
arbitrarily close $y$ that is strictly better.\\[0.5em]
Recall that we are in a Euclidean space, so
\[
\lVert x - y\rVert=\sqrt{\sum_{i=1}^L (x_i - y_i)^2}.
\]
Local nonsatiation rules out ``thick'' indifference sets.
\end{frame}

\begin{frame}{Monotonicity and Local Nonsatiation}
\begin{propositionbox}
On $X=\mathbb{R}_+^L$, strict monotonicity implies local nonsatiation.
\end{propositionbox}

\textbf{Proof.}\quad
Suppose $\succeq$ is strictly monotone, i.e. $x \ge y$ and $x \ne y$ imply $x \succ y$.\\
For each $x \in \mathbb{R}_+^L$ and every $\varepsilon > 0$, let
$y = x + \frac{\varepsilon}{L}\mathbf{1}$. Then $y\in\mathbb{R}_+^L$ and
\[
\lVert x - y\rVert=\sqrt{\sum_{i=1}^L \left(\frac{\varepsilon}{L}\right)^2}
=\sqrt{L\left(\frac{\varepsilon}{L}\right)^2}=\frac{\varepsilon}{\sqrt{L}}\le \varepsilon
\]
Strict monotonicity implies that $y \succ x$.\\[0.5em]
However, local nonsatiation does not imply monotonicity.
\end{frame}

\begin{frame}{Convexity: Taste for Variety}

\begin{definitionbox}[Convex Sets and Preferences]
A set $X\subseteq\mathbb{R}^L$ is \emph{convex} if $x,y\in X$ and
$\alpha\in[0,1]$ imply $\alpha x+(1-\alpha)y\in X$.

Let $X$ be convex. For each $x\in X$, define
$U(x):=\{y\in X:y\succeq x\}$. A preference relation $\succeq$ on $X$ is
\emph{convex} if
\[
y,z\in U(x),\ \alpha\in[0,1]
\quad\Longrightarrow\quad
\alpha y+(1-\alpha)z\in U(x).
\]
It is \emph{strictly convex} if $y\ne z$ and $\alpha\in(0,1)$ imply
$\alpha y+(1-\alpha)z\succ x$ whenever $y,z\in U(x)$.
\end{definitionbox}

\begin{block}{Precise Geometric Statement}
Convex preferences mean that every upper contour set $U(x)$ is convex. This is
the precise statement behind the informal ``taste for variety'' description.
\end{block}

\end{frame}


\begin{frame}{Level Sets and Quasiconcavity}

\begin{block}{Definitions}
Let $X\subseteq\mathbb{R}^L$ be convex. A function $f:X\to\mathbb{R}$ is
\emph{quasiconcave} if, for $x,y\in X$ and $\alpha\in[0,1]$,
\[
f(\alpha x+(1-\alpha)y)\geq\min\{f(x),f(y)\}.
\]
It is \emph{strictly quasiconcave} if, for $x \ne y$ and $\alpha \in (0,1)$,
\[
f(\alpha x + (1-\alpha)y) > \min\{f(x),f(y)\}.
\]
\end{block}

\begin{block}{A More Transparent Definition}
A function $f : X \to \mathbb{R}$ is \emph{quasiconcave} if every upper level set of $f$ is convex. That is, $\{\,x \in X \mid f(x) \geq c\,\}$ is convex for every $c \in \mathbb{R}$.
\end{block}
\end{frame}

\begin{frame}{Convex Preferences and Quasiconcave Utility}

\begin{propositionbox}[Exact Equivalence]
Let $X$ be convex and let $u:X\to\mathbb{R}$ represent $\succeq$. Then
\[
\succeq\ \text{is convex}
\quad\Longleftrightarrow\quad
u\ \text{is quasiconcave}.
\]
\end{propositionbox}

\begin{block}{Why}
For every $x\in X$,
\[
U(x)=\{y\in X:y\succeq x\}
=\{y\in X:u(y)\geq u(x)\}.
\]
Upper contour sets of preferences are upper level sets of any representing
utility function. Their convexity is therefore the same property.
\end{block}

\end{frame}

\begin{frame}{Topology on the Choice Set}

\begin{definitionbox}[Relative Topology on $X\subseteq\mathbb{R}^L$]
For $x\in X$ and $\varepsilon>0$, the relative open ball is
\[
B^X_\varepsilon(x)=\{y\in X:\lVert y-x\rVert<\varepsilon\}.
\]
\begin{itemize}
    \item A set $G\subseteq X$ is \emph{open relative to $X$} if every $x\in G$
    has some $\varepsilon>0$ with $B^X_\varepsilon(x)\subseteq G$.
    \item A set $F\subseteq X$ is \emph{closed relative to $X$} if $X\setminus F$ is open relative to $X$.
\end{itemize}
\end{definitionbox}

\begin{block}{Why ``Relative'' Matters}
At a boundary point of $X$---for example, where one commodity is zero---the ball is intersected with the feasible choice set $X$.
\end{block}
\end{frame}

\begin{frame}{Lower and Upper Contour Sets}
\begin{definitionbox}[Contour Sets at $x$]
\[
U(x) := \{ y \in X \mid y \succeq x \} \qquad L(x) := \{ y \in X \mid x \succeq y \}.
\]
The indifference set through $x$ is always
\[
I(x):=\{y\in X:y\sim x\}=U(x)\cap L(x).
\]
\end{definitionbox}

\begin{propositionbox}[What Continuity Adds]
If preferences are continuous, $U(x)$ and $L(x)$ are closed relative to $X$,
while $\{y\in X:y\succ x\}$ and $\{y\in X:x\succ y\}$ are open relative to $X$.
\end{propositionbox}
\end{frame}

\begin{frame}{Continuity: Equivalent Tests}
\begin{propositionbox}[Continuity of Preferences]
Let $X\subseteq\mathbb{R}^L$ and let $\succeq$ be complete. The following are equivalent:
\begin{enumerate}
    \item $\succeq$ is continuous.
    \item For every $x\in X$, $U(x)$ and $L(x)$ are closed relative to $X$.
    \item For every $x\in X$, $\{y\in X:y\succ x\}$ and
    $\{y\in X:x\succ y\}$ are open relative to $X$.
\end{enumerate}
\end{propositionbox}

\begin{block}{Interpretation}
Small changes in a bundle cannot cause a strict ranking to reverse abruptly.
\end{block}
\end{frame}

\begin{frame}{Continuity: Rankings Survive Limits}
\begin{propositionbox}[Sequence Test]
Let $x_n\to x$ in $X$.
\begin{enumerate}
    \item If $x_n\succeq y$ for every $n$, then $x\succeq y$.
    \item If $y\succeq x_n$ for every $n$, then $y\succeq x$.
\end{enumerate}
\end{propositionbox}

\begin{block}{Equivalent Local Test}
If $x\succ y$, then every bundle $x'$ sufficiently close to $x$ also satisfies
$x'\succ y$. Likewise, if $y\succ x$, then $y\succ x'$ for all $x'$ sufficiently
close to $x$.
\end{block}
\end{frame}

\begin{frame}{Debreu's Representation Theorem}
\begin{theorembox}[Continuous Representation]
Let $X=\mathbb{R}_+^L$. If $\succeq$ is complete, transitive, and continuous on
$X$, then there exists a \emph{continuous} utility function $u:X\to\mathbb{R}$ such that
\[
x\succeq y \quad\Longleftrightarrow\quad u(x)\geq u(y).
\]
\end{theorembox}

\begin{itemize}
    \item A continuous utility representation implies continuous preferences.
    \item The theorem guarantees at least one continuous representation; it does not make every utility representation continuous.
    \item Rationality is necessary. On $\mathbb{R}_+^L$, continuity supplies enough additional structure for a continuous representation.
\end{itemize}
    
\end{frame}
